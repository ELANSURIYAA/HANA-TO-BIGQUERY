FROM "CVS_FRIP"."FIN_BASE_COMPARISON_WEEKS_VT_S4" 
WHERE "WEEK_CY" IN ( SELECT "ZZWEEK"  FROM "_SYS_BIC"."CVS_FRIP.Base.Master/CV_BASE_MD_RCALWEEK_S4"

					 AND TO_DATE(ZRWENDDATE) >= TO_DATE(ADD_DAYS(CURRENT_DATE,-7)) );
