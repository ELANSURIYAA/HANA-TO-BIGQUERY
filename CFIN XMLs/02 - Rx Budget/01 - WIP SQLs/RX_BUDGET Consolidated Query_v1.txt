-- @ip_fc_count: STRING (frozen cube flag, precomputed externally)
-- @v_week: STRING (prior fiscal week, precomputed externally)
-- =====================================================================
with WEEKDATES AS (
    SELECT
        CAST(CURR.RCLNT AS STRING)      AS MANDT,
        CAST(CURR.ZZWEEK AS STRING)     AS CURR_WEEK_NBR,
        CAST(CURR.ZRWSTRTDATE AS DATE)  AS CURR_WEEK_START,
        CAST(CURR.ZRWENDDATE AS DATE)   AS CURR_WEEK_END,
        CAST(PREV.ZZWEEK AS STRING)     AS PREV_WEEK_NBR,
        CAST(PREV.ZRWSTRTDATE AS DATE)  AS PREV_WEEK_START,
        CAST(PREV.ZRWENDDATE AS DATE)   AS PREV_WEEK_END
    FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek` AS CURR
    LEFT JOIN `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek` AS PREV
        ON CAST(PREV.RCLNT AS STRING) = CAST(CURR.RCLNT AS STRING)
       AND CAST(PREV.ZRWENDDATE AS DATE) =
           DATE_SUB(CAST(CURR.ZRWSTRTDATE AS DATE), INTERVAL 1 DAY)
    WHERE CURRENT_DATE('America/New_York')
          BETWEEN CAST(CURR.ZRWSTRTDATE AS DATE)
              AND CAST(CURR.ZRWENDDATE AS DATE)
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY CAST(CURR.RCLNT AS STRING)
        ORDER BY CAST(CURR.ZRWSTRTDATE AS DATE) DESC,
                 CAST(CURR.ZZWEEK AS STRING) DESC
    ) = 1),

    IP_FC_COUNT as (
      SELECT COUNT(*) FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds041` WHERE bic_zio_sweek  = (Select WEEKDATES.PREV_WEEK_NBR from WEEKDATES)
																		   AND bic_zio_ver = 'LEDGER_BUD'
    ),
