-- =====================================================================
-- CONSOLIDATED BIGQUERY SQL FOR CV_COMP_SKF_BUDGET_STATIC
-- Final Consumer: CV_COMP_SKF_BUDGET_STATIC
-- Runtime Parameters: @ip_version, @ip_week_ending_from, @ip_week_ending_to
-- =====================================================================

-- =====================================================================
-- CTE 1: Physical_AZSRP_DS092
-- Logic: Source frozen cube data from AZSRP_DS092_VT_S4 with client filter and frozen cube flag condition
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds092
-- Filters: mandt IN ('110','200') AND @ip_fc_count != '0'
-- =====================================================================
WITH Physical_AZSRP_DS092 AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds092`
  WHERE
    mandt IN ('110', '200')
    AND @ip_fc_count != '0'
),

-- =====================================================================
-- CTE 2: Physical_AZSRP_DS031
-- Logic: Source live cube data from AZSRP_DS031_VT_S4 with client filter and live cube flag condition
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds031
-- Filters: mandt IN ('110','200') AND @ip_fc_count = '0'
-- =====================================================================
Physical_AZSRP_DS031 AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_compcode,
    bic_zio_cctr,
    bic_zio_pctr,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds031`
  WHERE
    mandt IN ('110', '200')
    AND @ip_fc_count = '0'
),

-- =====================================================================
-- CTE 3: WEEKLY_SNAPSHOT_DS05
-- Logic: Union frozen and live cube data with flag indicator, apply fiscal variant and version filters, map live cube columns to frozen cube structure
-- Source: Physical_AZSRP_DS092 (frozen) and Physical_AZSRP_DS031 (live)
-- Filters: FISCVARNT='K4', bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to, bic_zio_ver=@ip_version, bic_zio_saudt IN ('1','10'), bic_zio_pskf IN ('804013','801000')
-- Calculation: Conditional measure selection based on FLAG (frozen vs live), column mapping for live cube
-- =====================================================================
WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'FC' AS flag
  FROM
    Physical_AZSRP_DS092
  UNION ALL
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    CAST(NULL AS STRING) AS b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    CAST(NULL AS STRING) AS b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'LC' AS flag
  FROM
    Physical_AZSRP_DS031
),

-- =====================================================================
-- CTE 4: WEEKLY_SNAPSHOT_AGGREGATED
-- Logic: Aggregate weekly snapshot data with conditional measure selection (frozen cube takes precedence over live cube)
-- Source: WEEKLY_SNAPSHOT_DS05
-- Aggregation: GROUP BY all non-measure columns, conditional SUM based on FLAG
-- Calculation: IF(SUM(CASE WHEN flag='FC' THEN bic_zio_skfq END) IS NULL, SUM(CASE WHEN flag='LC' THEN bic_zio_skfq END), SUM(CASE WHEN flag='FC' THEN bic_zio_skfq END))
-- =====================================================================
WEEKLY_SNAPSHOT_AGGREGATED AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    IF(
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq ELSE NULL END) IS NULL,
      SUM(CASE WHEN flag = 'LC' THEN bic_zio_skfq ELSE NULL END),
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq ELSE NULL END)
    ) AS bic_zio_skfq
  FROM
    WEEKLY_SNAPSHOT_DS05
  GROUP BY
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit
),

-- =====================================================================
-- CTE 5: WEEKLY_SNAPSHOT_FILTERED
-- Logic: Apply final filters on aggregated weekly snapshot
-- Source: WEEKLY_SNAPSHOT_AGGREGATED
-- Filters: fiscvarnt='K4', bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to, bic_zio_ver=@ip_version, bic_zio_saudt IN ('1','10'), bic_zio_pskf IN ('804013','801000')
-- =====================================================================
WEEKLY_SNAPSHOT_FILTERED AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM
    WEEKLY_SNAPSHOT_AGGREGATED
  WHERE
    fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 6: HIER_NODE
-- Logic: Filter hierarchy nodes for parent node pattern and validity date
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node
-- Filters: REGEXP_CONTAINS(parnode, r'CORERET$') AND hryvalto='99991231', mandt IN ('120','200')
-- =====================================================================
HIER_NODE AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node`
  WHERE
    mandt IN ('120', '200')
    AND REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 7: Join_1
-- Logic: Inner join weekly snapshot with hierarchy nodes on cost center
-- Source: WEEKLY_SNAPSHOT_FILTERED and HIER_NODE
-- Join: INNER JOIN on b631_s_costcntr = nodevalue
-- =====================================================================
Join_1 AS (
  SELECT
    wsd.fiscper,
    wsd.fiscvarnt,
    wsd.fiscyear,
    wsd.fiscper3,
    wsd.mandt,
    wsd.recordmode,
    wsd.bic_zio_sweek,
    wsd.b631_s_co_area,
    wsd.bic_zio_cmpcd,
    wsd.b631_s_proftctr,
    hn.nodevalue AS b631_s_costcntr,
    wsd.bic_zio_ver,
    wsd.bic_zio_saudt,
    wsd.bic_zio_psty,
    wsd.bic_zio_pskf,
    wsd.unit,
    wsd.bic_zio_skfq
  FROM
    WEEKLY_SNAPSHOT_FILTERED wsd
  INNER JOIN HIER_NODE hn
    ON wsd.b631_s_costcntr = hn.nodevalue
),

-- =====================================================================
-- CTE 8: Physical_ZTSRP_ATTR_ACT
-- Logic: Source store/profit center attributes from physical table with client filter
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_attr_act
-- Filters: mandt IN ('120','200')
-- =====================================================================
Physical_ZTSRP_ATTR_ACT AS (
  SELECT
    mandt,
    strnum,
    prctr,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    city,
    state,
    fs_open_dat,
    rx_open_dat,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_attr_act`
  WHERE
    mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 9: Join_2
-- Logic: Left join with store attributes on profit center
-- Source: Join_1 and Physical_ZTSRP_ATTR_ACT
-- Join: LEFT JOIN on b631_s_proftctr = prctr
-- =====================================================================
Join_2 AS (
  SELECT
    j.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM
    Join_1 j
  LEFT JOIN Physical_ZTSRP_ATTR_ACT s
    ON j.b631_s_proftctr = s.prctr
),

-- =====================================================================
-- CTE 10: Physical_ZTSRP_COMPFL
-- Logic: Source comp flag data from physical table with client and version filters
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_compfl
-- Filters: mandt IN ('110','200'), comp_ver=@ip_version
-- =====================================================================
Physical_ZTSRP_COMPFL AS (
  SELECT
    mandt,
    prctr,
    strnum,
    rx_comp_wk,
    zweek,
    zmonth,
    zyear,
    comp_ver
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_compfl`
  WHERE
    mandt IN ('110', '200')
    AND comp_ver = @ip_version
),

-- =====================================================================
-- CTE 11: Join_3
-- Logic: Left join with comp flags on profit center and week
-- Source: Join_2 and Physical_ZTSRP_COMPFL
-- Join: LEFT JOIN on b631_s_proftctr=prctr AND bic_zio_sweek=zweek
-- =====================================================================
Join_3 AS (
  SELECT
    jw.*,
    c.rx_comp_wk
  FROM
    Join_2 jw
  LEFT JOIN Physical_ZTSRP_COMPFL c
    ON jw.b631_s_proftctr = c.prctr
    AND jw.bic_zio_sweek = c.zweek
),

-- =====================================================================
-- CTE 12: CAL_WEEK
-- Logic: Source calendar week attributes from physical table with client filter
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek
-- Filters: rclnt IN ('120','200')
-- =====================================================================
CAL_WEEK AS (
  SELECT
    zrwstrtdate,
    zrwenddate,
    zzweek,
    rclnt
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek`
  WHERE
    rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 13: Join_4
-- Logic: Left join with calendar week on week number
-- Source: Join_3 and CAL_WEEK
-- Join: LEFT JOIN on bic_zio_sweek=zzweek
-- =====================================================================
Join_4 AS (
  SELECT
    j.*,
    cw.zrwenddate,
    cw.zrwstrtdate,
    cw.rclnt
  FROM
    Join_3 j
  LEFT JOIN CAL_WEEK cw
    ON j.bic_zio_sweek = cw.zzweek
),

-- =====================================================================
-- CTE 14: PROFIT_CENTER_TEXT
-- Logic: Source profit center text from physical table with client filter
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Filters: mandt IN ('120','200')
-- =====================================================================
PROFIT_CENTER_TEXT AS (
  SELECT
    mandt,
    prctr,
    ltext AS profit_center_text
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE
    mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 15: Join_5
-- Logic: Left join with profit center text on profit center
-- Source: Join_4 and PROFIT_CENTER_TEXT
-- Join: LEFT JOIN on b631_s_proftctr=prctr
-- =====================================================================
Join_5 AS (
  SELECT
    jw.*,
    pct.profit_center_text
  FROM
    Join_4 jw
  LEFT JOIN PROFIT_CENTER_TEXT pct
    ON jw.b631_s_proftctr = pct.prctr
),

-- =====================================================================
-- CTE 16: FLAGS
-- Logic: Final projection with calculated columns
-- Source: Join_5
-- Calculation: CAL_WEEK_NUMBER (last 2 digits of week), CAL_BIC_ZIO_SKFQ (conditional doubling for SKF account 804013), CAL_RX_COMP_WK (null-to-zero conversion)
-- =====================================================================
FLAGS AS (
  SELECT
    bic_zio_skfq,
    unit,
    bic_zio_pskf,
    bic_zio_psty,
    rx_comp_wk,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    strnum,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    RIGHT(CAST(bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS NUMERIC) * 2 
      ELSE CAST(bic_zio_skfq AS NUMERIC) 
    END AS cal_bic_zio_skfq,
    CASE 
      WHEN rx_comp_wk IS NULL THEN '0' 
      ELSE rx_comp_wk 
    END AS cal_rx_comp_wk
  FROM
    Join_5
)

-- =====================================================================
-- FINAL SELECT: CV_COMP_SKF_BUDGET_STATIC Output
-- Logic: Aggregate final output with all dimensions and measures
-- Source: FLAGS
-- Aggregation: GROUP BY all non-measure columns, SUM for measures
-- Measures: bic_zio_skfq_negative (SUM of raw bic_zio_skfq), bic_zio_skfq (SUM of calculated cal_bic_zio_skfq)
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  SUM(bic_zio_skfq) AS bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS bic_zio_skfq
FROM
  FLAGS
GROUP BY
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code
