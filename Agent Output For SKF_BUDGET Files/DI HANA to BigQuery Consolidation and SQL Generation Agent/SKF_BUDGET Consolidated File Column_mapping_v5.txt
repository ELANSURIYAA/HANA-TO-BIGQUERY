-- =====================================================================
-- CTE 12: WEEKLY_SNAPSHOT_DS05
-- Logic: Filter weekly budget snapshot for fiscal variant K4, week range, 
--        version, audit trail, and SKF accounts
-- Source: CV_BASE_SKF_WEEKLY_BUDGET_S4
-- Filters: FISCVARNT='K4', week range, version, audit, SKF accounts
-- =====================================================================
WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    wb._bic_zio_pctr,
    wb._bic_zio_pskf,
    wb._bic_zio_saudt,
    wb._bic_zio_sweek,
    wb._bic_zio_skfq
  FROM
    CV_BASE_SKF_WEEKLY_BUDGET_S4 wb
  INNER JOIN
    CV_BASE_MD_RCALWEEK_S4 cal
    ON wb._bic_zio_sweek = cal.rcalweek
  WHERE
    cal.rcaldate BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND wb._bic_zio_saudt IN ('1', '10')
    AND wb._bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 13: HIER_NODE
-- Logic: Filter hierarchy nodes for CORERET pattern with valid-to date
-- Source: CV_BASE_MD_HRRP_NODE_S4
-- Filters: PARNODE matches '*CORERET', HRYVALTO='99991231'
-- =====================================================================
HIER_NODE AS (
  SELECT
    nodeid,
    parnode,
    hryvalto
  FROM
    CV_BASE_MD_HRRP_NODE_S4
  WHERE
    REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 14: JOIN_SNAPSHOT_HIER
-- Logic: Join weekly snapshot with hierarchy nodes
-- Source: WEEKLY_SNAPSHOT_DS05 + HIER_NODE
-- Join: _BIC_ZIO_PCTR = NODEID
-- =====================================================================
JOIN_SNAPSHOT_HIER AS (
  SELECT
    ws._bic_zio_pctr,
    ws._bic_zio_pskf,
    ws._bic_zio_saudt,
    ws._bic_zio_sweek,
    ws._bic_zio_skfq,
    hn.parnode
  FROM
    WEEKLY_SNAPSHOT_DS05 ws
  INNER JOIN
    HIER_NODE hn
    ON ws._bic_zio_pctr = hn.nodeid
),

-- =====================================================================
-- CTE 15: JOIN_STORE_ATTR
-- Logic: Join with store attributes
-- Source: JOIN_SNAPSHOT_HIER + CV_COMP_MD_SRPACT_STATIC
-- Join: _BIC_ZIO_PCTR = ZSTORE, _BIC_ZIO_SWEEK = ZWEEK
-- =====================================================================
JOIN_STORE_ATTR AS (
  SELECT
    jsh._bic_zio_pctr,
    jsh._bic_zio_pskf,
    jsh._bic_zio_saudt,
    jsh._bic_zio_sweek,
    jsh._bic_zio_skfq,
    jsh.parnode,
    sa.zregion,
    sa.zdistrict,
    sa.zarea,
    sa.zmarket,
    sa.zstate,
    sa.zstore_typ,
    sa.zstore_frmt,
    sa.zrx_typ,
    sa.zbanner_cd,
    sa.zbanner_nm
  FROM
    JOIN_SNAPSHOT_HIER jsh
  INNER JOIN
    CV_COMP_MD_SRPACT_STATIC sa
    ON jsh._bic_zio_pctr = sa.zstore
    AND jsh._bic_zio_sweek = sa.zweek
),

-- =====================================================================
-- CTE 16: JOIN_COMP_FLAG
-- Logic: Join with comparison flags
-- Source: JOIN_STORE_ATTR + CV_COMP_MD_COMPFL_STATIC
-- Join: _BIC_ZIO_PCTR = ZSTORE, _BIC_ZIO_SWEEK = ZWEEK
-- =====================================================================
JOIN_COMP_FLAG AS (
  SELECT
    jsa._bic_zio_pctr,
    jsa._bic_zio_pskf,
    jsa._bic_zio_saudt,
    jsa._bic_zio_sweek,
    jsa._bic_zio_skfq,
    jsa.parnode,
    jsa.zregion,
    jsa.zdistrict,
    jsa.zarea,
    jsa.zmarket,
    jsa.zstate,
    jsa.zstore_typ,
    jsa.zstore_frmt,
    jsa.zrx_typ,
    jsa.zbanner_cd,
    jsa.zbanner_nm,
    cf.zrx_comp_wk,
    cf.zfs_comp_wk
  FROM
    JOIN_STORE_ATTR jsa
  INNER JOIN
    CV_COMP_MD_COMPFL_STATIC cf
    ON jsa._bic_zio_pctr = cf.zstore
    AND jsa._bic_zio_sweek = cf.zweek
),

-- =====================================================================
-- CTE 17: JOIN_CAL_WEEK
-- Logic: Join with calendar week data
-- Source: JOIN_COMP_FLAG + CV_BASE_MD_RCALWEEK_S4
-- Join: _BIC_ZIO_SWEEK = RCALWEEK
-- =====================================================================
JOIN_CAL_WEEK AS (
  SELECT
    jcf._bic_zio_pctr,
    jcf._bic_zio_pskf,
    jcf._bic_zio_saudt,
    jcf._bic_zio_sweek,
    jcf._bic_zio_skfq,
    jcf.parnode,
    jcf.zregion,
    jcf.zdistrict,
    jcf.zarea,
    jcf.zmarket,
    jcf.zstate,
    jcf.zstore_typ,
    jcf.zstore_frmt,
    jcf.zrx_typ,
    jcf.zbanner_cd,
    jcf.zbanner_nm,
    jcf.zrx_comp_wk,
    jcf.zfs_comp_wk,
    cw.rcalmonth,
    cw.rcalyear,
    cw.rcalquart
  FROM
    JOIN_COMP_FLAG jcf
  INNER JOIN
    CV_BASE_MD_RCALWEEK_S4 cw
    ON jcf._bic_zio_sweek = cw.rcalweek
),

-- =====================================================================
-- CTE 18: JOIN_PROFIT_CENTER
-- Logic: Join with profit center text
-- Source: JOIN_CAL_WEEK + CV_BASE_MD_CEPCT_S4
-- Join: _BIC_ZIO_PCTR = PRCTR
-- =====================================================================
JOIN_PROFIT_CENTER AS (
  SELECT
    jcw._bic_zio_pctr,
    jcw._bic_zio_pskf,
    jcw._bic_zio_saudt,
    jcw._bic_zio_sweek,
    jcw._bic_zio_skfq,
    jcw.parnode,
    jcw.zregion,
    jcw.zdistrict,
    jcw.zarea,
    jcw.zmarket,
    jcw.zstate,
    jcw.zstore_typ,
    jcw.zstore_frmt,
    jcw.zrx_typ,
    jcw.zbanner_cd,
    jcw.zbanner_nm,
    jcw.zrx_comp_wk,
    jcw.zfs_comp_wk,
    jcw.rcalmonth,
    jcw.rcalyear,
    jcw.rcalquart,
    pc.ktext AS prctr_text
  FROM
    JOIN_CAL_WEEK jcw
  LEFT JOIN
    CV_BASE_MD_CEPCT_S4 pc
    ON jcw._bic_zio_pctr = pc.prctr
),

-- =====================================================================
-- CTE 19: FLAGS
-- Logic: Calculate derived columns and apply business rules
-- Source: JOIN_PROFIT_CENTER
-- Calculation: CAL_WEEK_NUMBER, CAL_BIC_ZIO_SKFQ, CAL_RX_COMP_WK
-- =====================================================================
FLAGS AS (
  SELECT
    _bic_zio_pctr,
    _bic_zio_pskf,
    _bic_zio_saudt,
    _bic_zio_sweek,
    parnode,
    zregion,
    zdistrict,
    zarea,
    zmarket,
    zstate,
    zstore_typ,
    zstore_frmt,
    zrx_typ,
    zbanner_cd,
    zbanner_nm,
    zrx_comp_wk,
    zfs_comp_wk,
    rcalmonth,
    rcalyear,
    rcalquart,
    prctr_text,
    -- Calculated columns
    RIGHT(CAST(_bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN _bic_zio_pskf = '804013' THEN _bic_zio_skfq * 2 
      ELSE _bic_zio_skfq 
    END AS cal_bic_zio_skfq,
    COALESCE(zrx_comp_wk, '0') AS cal_rx_comp_wk,
    -- Measures
    -1 * CASE 
      WHEN _bic_zio_pskf = '804013' THEN _bic_zio_skfq * 2 
      ELSE _bic_zio_skfq 
    END AS _bic_zio_skfq_negative,
    CASE 
      WHEN _bic_zio_pskf = '804013' THEN _bic_zio_skfq * 2 
      ELSE _bic_zio_skfq 
    END AS _bic_zio_skfq
  FROM
    JOIN_PROFIT_CENTER
)

-- =====================================================================
-- FINAL AGGREGATION
-- Logic: Aggregate by all dimensions, SUM measures
-- Source: FLAGS
-- Aggregation: GROUP BY all attributes, SUM on measures
-- Output: Final SKF Budget Static result set
-- =====================================================================
SELECT
  _bic_zio_pctr,
  _bic_zio_pskf,
  _bic_zio_saudt,
  _bic_zio_sweek,
  parnode,
  zregion,
  zdistrict,
  zarea,
  zmarket,
  zstate,
  zstore_typ,
  zstore_frmt,
  zrx_typ,
  zbanner_cd,
  zbanner_nm,
  zrx_comp_wk,
  zfs_comp_wk,
  rcalmonth,
  rcalyear,
  rcalquart,
  prctr_text,
  cal_week_number,
  cal_rx_comp_wk,
  SUM(_bic_zio_skfq_negative) AS _bic_zio_skfq_negative,
  SUM(_bic_zio_skfq) AS _bic_zio_skfq
FROM
  FLAGS
GROUP BY
  _bic_zio_pctr,
  _bic_zio_pskf,
  _bic_zio_saudt,
  _bic_zio_sweek,
  parnode,
  zregion,
  zdistrict,
  zarea,
  zmarket,
  zstate,
  zstore_typ,
  zstore_frmt,
  zrx_typ,
  zbanner_cd,
  zbanner_nm,
  zrx_comp_wk,
  zfs_comp_wk,
  rcalmonth,
  rcalyear,
  rcalquart,
  prctr_text,
  cal_week_number,
  cal_rx_comp_wk;
