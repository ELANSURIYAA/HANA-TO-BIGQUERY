/*******************************************************************************
 * CONSOLIDATED BIGQUERY SQL FOR SKF_BUDGET
 * 
 * Final Consumer: CV_COMP_SKF_BUDGET_STATIC
 * 
 * Purpose: This consolidated query implements the complete SKF Budget logic
 * by inlining all dependent calculation views and applying HANA-to-BigQuery
 * column mappings from the provided mapping file.
 * 
 * Dependencies Consolidated (in order of consumption):
 *   1. CV_BASE_SKF_WEEKLY_BUDGET_S4 (Weekly budget snapshot from frozen/live cubes)
 *   2. CV_BASE_MD_HRRP_NODE_S4 (Hierarchy nodes)
 *   3. CV_COMP_MD_SRPACT_STATIC (Store attributes - sourced from TBL_WSS_SRP_ATTR_ACT)
 *   4. CV_COMP_MD_COMPFL_STATIC (Completion flags - sourced from TBL_WSS_SRP_COMPFLAG)
 *   5. CV_BASE_MD_RCALWEEK_S4 (Retail calendar week)
 *   6. CV_BASE_MD_CEPCT_S4 (Profit center text)
 * 
 * Parameters:
 *   @IP_VERSION - Version identifier (default: 'LEDGER_BUD')
 *   @IP_WEEK_ENDING_FROM - Start week for budget period
 *   @IP_WEEK_ENDING_TO - End week for budget period
 *   @IP_FC_COUNT - Frozen cube count flag (derived from IP_VERSION via SFN_SKF_FC_FLAG)
 * 
 * Notes:
 *   - All HANA column names have been mapped to BigQuery equivalents
 *   - Table references use fully qualified BigQuery naming
 *   - MANDT/RCLNT filters applied as per source views (120, 200)
 *   - Frozen/Live cube logic preserved with FLAG column
 *   - All joins, filters, and calculated columns preserved exactly
 * 
 * Validation Items:
 *   - SFN_PRIOR_FISCAL_WEEK() and SFN_SKF_FC_FLAG() functions need BigQuery implementation
 *   - Verify @IP_FC_COUNT derivation logic
 *   - Validate REGEXP_CONTAINS pattern for PARNODE suffix match
 *   - Confirm SESSION_USER() context in BigQuery environment
 * 
 ******************************************************************************/

-- =============================================================================
-- CTE 1: FROZEN_CUBE
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092 (AZSRP_DS092_VT_S4)
-- Purpose: Frozen budget cube data filtered by MANDT and FC flag
-- Logic: Direct projection with MANDT filter and IP_FC_COUNT != 0
-- =============================================================================
WITH frozen_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    _bic_zio_sweek,
    fiscyear,
    fiscper3,
    _b631_s_chrtacct,
    _bic_zio_psty,
    _bic_zio_pskf,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _b631_s_proftctr,
    _b631_s_costcntr,
    _b631_s_funcarea,
    _bic_zio_ver,
    _bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    _bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092`
  WHERE mandt IN ('110', '200')
    AND @IP_FC_COUNT != '0'
),

-- =============================================================================
-- CTE 2: LIVE_CUBE
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031 (AZSRP_DS031_VT_S4)
-- Purpose: Live budget cube data filtered by MANDT and LC flag
-- Logic: Direct projection with MANDT filter and IP_FC_COUNT = 0
-- Column Mapping: _bic_zio_pctr → _b631_s_proftctr, _bic_zio_cctr → _b631_s_costcntr
-- =============================================================================
live_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    _bic_zio_sweek,
    fiscyear,
    fiscper3,
    _b631_s_compcode,
    _bic_zio_cctr,
    _bic_zio_pctr,
    _bic_zio_psty,
    _bic_zio_pskf,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _bic_zio_ver,
    _bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    _bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031`
  WHERE mandt IN ('110', '200')
    AND @IP_FC_COUNT = '0'
),

-- =============================================================================
-- CTE 3: COMBINED_FROZEN_LIVE_DATA (CV_BASE_SKF_WEEKLY_BUDGET_S4 logic)
-- Purpose: Union frozen and live cubes with FLAG column
-- Logic: UNION ALL with column alignment and FLAG='FC'/'LC' marker
-- Calculated Measure: IF(ISNULL(RES_QUANT_FC), RES_QUANT_LC, RES_QUANT_FC)
-- =============================================================================
combined_frozen_live_data AS (
  SELECT
    fiscper,
    fiscvarnt,
    _bic_zio_sweek,
    fiscyear,
    fiscper3,
    _b631_s_chrtacct,
    _bic_zio_psty,
    _bic_zio_pskf,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _b631_s_proftctr,
    _b631_s_costcntr,
    _b631_s_funcarea,
    _bic_zio_ver,
    _bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    _bic_zio_skfq,
    'FC' AS flag
  FROM frozen_cube
  
  UNION ALL
  
  SELECT
    fiscper,
    fiscvarnt,
    _bic_zio_sweek,
    fiscyear,
    fiscper3,
    '' AS _b631_s_chrtacct,
    _bic_zio_psty,
    _bic_zio_pskf,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _bic_zio_pctr AS _b631_s_proftctr,
    _bic_zio_cctr AS _b631_s_costcntr,
    '' AS _b631_s_funcarea,
    _bic_zio_ver,
    _bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    _bic_zio_skfq,
    'LC' AS flag
  FROM live_cube
),

-- =============================================================================
-- CTE 4: WEEKLY_SNAPSHOT_DS05
-- Purpose: Aggregated weekly budget with frozen/live cube logic
-- Logic: GROUP BY all attributes, calculate measure as FC if available, else LC
-- Filters: FISCVARNT='K4', week range, version, audit trail, SKF account
-- =============================================================================
weekly_snapshot_ds05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    _bic_zio_sweek,
    fiscyear,
    fiscper3,
    _b631_s_chrtacct,
    _bic_zio_psty,
    _bic_zio_pskf,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _b631_s_proftctr,
    _b631_s_costcntr,
    _b631_s_funcarea,
    _bic_zio_ver,
    _bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    flag,
    -- Calculated measure: prefer frozen cube quantity, fallback to live cube
    IFNULL(
      SUM(CASE WHEN flag = 'FC' THEN _bic_zio_skfq END),
      SUM(CASE WHEN flag = 'LC' THEN _bic_zio_skfq END)
    ) AS _bic_zio_skfq
  FROM combined_frozen_live_data
  WHERE fiscvarnt = 'K4'
    AND _bic_zio_sweek BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
    AND _bic_zio_ver = @IP_VERSION
    AND _bic_zio_saudt IN ('1', '10')
    AND _bic_zio_pskf IN ('804013', '801000')
  GROUP BY
    fiscper, fiscvarnt, _bic_zio_sweek, fiscyear, fiscper3, _b631_s_chrtacct,
    _bic_zio_psty, _bic_zio_pskf, _b631_s_co_area, _bic_zio_cmpcd,
    _b631_s_proftctr, _b631_s_costcntr, _b631_s_funcarea, _bic_zio_ver,
    _bic_zio_saudt, mandt, recordmode, unit, flag
),

-- =============================================================================
-- CTE 5: FILTERED_HIERARCHY_NODES (CV_BASE_MD_HRRP_NODE_S4 logic)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node
-- Purpose: Hierarchy nodes filtered for CORERET suffix and validity date
-- Logic: MANDT filter + PARNODE pattern match + HRYVALTO filter
-- =============================================================================
filtered_hierarchy_nodes AS (
  SELECT
    mandt,
    hryid,
    hryver,
    nodecls,
    hrynode,
    parnode,
    hryvalto,
    hryvalfrom,
    balind,
    nodetype,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE mandt IN (120, 200)
    AND REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =============================================================================
-- CTE 6: JOINED_WEEKLY_SNAPSHOT_WITH_HIERARCHY
-- Purpose: Join weekly snapshot with hierarchy nodes on cost center
-- Join Type: INNER JOIN
-- Join Key: _b631_s_costcntr = nodevalue
-- =============================================================================
joined_weekly_snapshot_with_hierarchy AS (
  SELECT
    ws.fiscper,
    ws.fiscvarnt,
    ws.fiscyear,
    ws.fiscper3,
    ws.mandt,
    ws.recordmode,
    ws._bic_zio_sweek,
    ws._b631_s_co_area,
    ws._bic_zio_cmpcd,
    ws._b631_s_proftctr,
    hn.nodevalue AS _b631_s_costcntr,
    ws._bic_zio_ver,
    ws._bic_zio_saudt,
    ws._bic_zio_psty,
    ws._bic_zio_pskf,
    ws.unit,
    ws._bic_zio_skfq,
    ws.flag
  FROM weekly_snapshot_ds05 ws
  INNER JOIN filtered_hierarchy_nodes hn
    ON ws._b631_s_costcntr = hn.nodevalue
),

-- =============================================================================
-- CTE 7: STORE_ATTR_ACTUAL (CV_COMP_MD_SRPACT_STATIC logic)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act
--         (via TBL_WSS_SRP_ATTR_ACT populated by STP_WSS_SRP_ATTRIBUTES)
-- Purpose: Store profile attributes aggregated by profit center
-- Logic: SUM measures, GROUP BY attributes, MANDT filter
-- Note: In production, this would reference TBL_WSS_SRP_ATTR_ACT populated
--       by the stored procedure; for consolidation we inline the base logic
-- =============================================================================
store_attr_actual AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE mandt IN ('120', '200')
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat, fs_open_dat,
    fs_close_dat, rep_mkt_code, rep_mkt_desc, address, city, state, zipcode,
    division_code, division_desc, division_mgr, area_code, area_desc, area_mgr,
    region_code, region_desc, region_mgr, district_code, district_desc, zone_code,
    district_mgr, zone_desc, major_region_code, major_region, store_type,
    store_type_desc, rx_store_desc, distribution_cntr, distribution_cntr_desc,
    store_24_hr, budget_open_dat, budget_closed_dat, budget_reloc_dat,
    const_open_dat, const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc, original_fs_open_dat,
    original_rx_open_dat, group_one, group_two, ard_sort_key, area_cost_center,
    regiion_cost_center, county_code, county_desc, sqft_bracket_desc, sqft_sort_key,
    acquisition_code, acquisition_desc, photo_lab_add_dat, photo_lab_close_dat,
    min_hrs_ind, createdby, created_on, rx_division_code, rx_area_code,
    rx_region_code, rx_district_code
),

-- =============================================================================
-- CTE 8: JOINED_WEEKLY_WITH_STORE_ATTRS
-- Purpose: Enrich weekly snapshot with store attributes
-- Join Type: LEFT JOIN
-- Join Key: _b631_s_proftctr = prctr
-- =============================================================================
joined_weekly_with_store_attrs AS (
  SELECT
    j.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM joined_weekly_snapshot_with_hierarchy j
  LEFT JOIN store_attr_actual s
    ON j._b631_s_proftctr = s.prctr
),

-- =============================================================================
-- CTE 9: COMP_FLAG_BUDGET (CV_COMP_MD_COMPFL_STATIC logic)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl
--         (via TBL_WSS_SRP_COMPFLAG populated by STP_WSS_SRP_ATTRIBUTES)
-- Purpose: Completion flags filtered by version
-- Logic: Direct projection with COMP_VER filter
-- Note: In production, this would reference TBL_WSS_SRP_COMPFLAG populated
--       by the stored procedure; for consolidation we inline the base logic
-- =============================================================================
comp_flag_budget AS (
  SELECT
    mandt,
    prctr,
    strnum,
    comp_ver,
    zweek,
    zmonth,
    zyear,
    fs_comp_pre,
    rx_comp_pre,
    fs_comp_mon,
    rx_comp_mon,
    fs_comp_wk,
    rx_comp_wk
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE comp_ver = @IP_VERSION
),

-- =============================================================================
-- CTE 10: JOINED_WITH_COMP_FLAGS
-- Purpose: Enrich with completion flags
-- Join Type: LEFT JOIN
-- Join Keys: _b631_s_proftctr = prctr AND _bic_zio_sweek = zweek
-- =============================================================================
joined_with_comp_flags AS (
  SELECT
    jw.*,
    c.rx_comp_wk
  FROM joined_weekly_with_store_attrs jw
  LEFT JOIN comp_flag_budget c
    ON jw._b631_s_proftctr = c.prctr
    AND jw._bic_zio_sweek = c.zweek
),

-- =============================================================================
-- CTE 11: CAL_WEEK (CV_BASE_MD_RCALWEEK_S4 logic)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek
-- Purpose: Retail calendar week reference data
-- Logic: Direct projection with RCLNT filter
-- =============================================================================
cal_week AS (
  SELECT
    rclnt,
    zzweek,
    zcalyrp,
    zryear,
    zrperiod,
    zryrp,
    zryrqtr,
    zrwstrtdate,
    zrwenddate
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE rclnt IN (120, 200)
),

-- =============================================================================
-- CTE 12: JOINED_WITH_CALENDAR_WEEK
-- Purpose: Enrich with calendar week dates
-- Join Type: LEFT JOIN
-- Join Key: _bic_zio_sweek = zzweek
-- =============================================================================
joined_with_calendar_week AS (
  SELECT
    j.*,
    cw.zrwenddate,
    cw.zrwstrtdate,
    cw.rclnt
  FROM joined_with_comp_flags j
  LEFT JOIN cal_week cw
    ON j._bic_zio_sweek = cw.zzweek
),

-- =============================================================================
-- CTE 13: PROFIT_CENTER_TEXT (CV_BASE_MD_CEPCT_S4 logic)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Purpose: Profit center text descriptions
-- Logic: Direct projection with MANDT filter
-- =============================================================================
profit_center_text AS (
  SELECT
    mandt,
    spras,
    prctr,
    datbi,
    kokrs,
    ktext,
    ltext,
    mctxt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN (120, 200)
),

-- =============================================================================
-- CTE 14: JOINED_WITH_PROFIT_CENTER_TEXT
-- Purpose: Enrich with profit center text
-- Join Type: LEFT JOIN
-- Join Key: _b631_s_proftctr = prctr
-- =============================================================================
joined_with_profit_center_text AS (
  SELECT
    jw.*,
    pct.ltext AS profit_center_text
  FROM joined_with_calendar_week jw
  LEFT JOIN profit_center_text pct
    ON jw._b631_s_proftctr = pct.prctr
),

-- =============================================================================
-- CTE 15: FLAGS
-- Purpose: Final projection with calculated columns
-- Calculated Columns:
--   - CAL_WEEK_NUMBER: RIGHT(CAST(_bic_zio_sweek AS STRING), 2)
--   - CAL_BIC_ZIO_SKFQ: CASE when _bic_zio_pskf='804013' then _bic_zio_skfq * 2 else _bic_zio_skfq
--   - CAL_RX_COMP_WK: IF(rx_comp_wk IS NULL, '0', rx_comp_wk)
-- =============================================================================
flags AS (
  SELECT
    _bic_zio_skfq,
    unit,
    _bic_zio_pskf,
    _bic_zio_psty,
    rx_comp_wk,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    _bic_zio_sweek,
    _b631_s_co_area,
    _bic_zio_cmpcd,
    _b631_s_proftctr,
    _b631_s_costcntr,
    _bic_zio_ver,
    _bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    strnum,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    flag,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    -- Calculated columns
    RIGHT(CAST(_bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN _bic_zio_pskf = '804013' THEN CAST(_bic_zio_skfq AS NUMERIC) * 2 
      ELSE CAST(_bic_zio_skfq AS NUMERIC) 
    END AS cal_bic_zio_skfq,
    CASE 
      WHEN rx_comp_wk IS NULL THEN '0' 
      ELSE rx_comp_wk 
    END AS cal_rx_comp_wk
  FROM joined_with_profit_center_text
)

-- =============================================================================
-- FINAL OUTPUT: CV_COMP_SKF_BUDGET_STATIC
-- Purpose: Aggregated budget view with all enrichments
-- Logic: GROUP BY all attributes, SUM measures
-- Output Measures:
--   - _bic_zio_skfq_negative: SUM(_bic_zio_skfq)
--   - _bic_zio_skfq: SUM(cal_bic_zio_skfq)
-- =============================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  _bic_zio_sweek,
  _b631_s_co_area,
  _bic_zio_cmpcd,
  _b631_s_proftctr,
  _b631_s_costcntr,
  _bic_zio_ver,
  _bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  _bic_zio_psty,
  _bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  flag AS skf_bud_cube,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  -- Measures
  SUM(_bic_zio_skfq) AS _bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS _bic_zio_skfq
FROM flags
GROUP BY
  mandt, fiscper, fiscvarnt, fiscyear, fiscper3, recordmode, _bic_zio_sweek,
  _b631_s_co_area, _bic_zio_cmpcd, _b631_s_proftctr, _b631_s_costcntr,
  _bic_zio_ver, _bic_zio_saudt, rep_mkt_code, rep_mkt_desc, emerg_mkt_ind,
  _bic_zio_psty, _bic_zio_pskf, unit, division_code, division_desc, area_code,
  area_desc, district_code, district_desc, region_code, region_desc,
  cal_week_number, strnum, zrwenddate, zrwstrtdate, city, state,
  profit_center_text, fs_open_dat, rx_open_dat, cal_rx_comp_wk, flag,
  rx_division_code, rx_area_code, rx_region_code, rx_district_code
;
