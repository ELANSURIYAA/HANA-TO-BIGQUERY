-- =====================================================================
-- CTE 1: base_weekly_budget_frozen
-- Logic: Frozen cube data from AZSRP_DS092_VT_S4, filtered by MANDT and IP_FC_COUNT
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092
-- Filters: MANDT IN ('110','200') AND @ip_fc_count != '0'
-- =====================================================================
WITH base_weekly_budget_frozen AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count != '0'
),

-- =====================================================================
-- CTE 2: base_weekly_budget_live
-- Logic: Live cube data from AZSRP_DS031_VT_S4, filtered by MANDT and IP_FC_COUNT
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031
-- Filters: MANDT IN ('110','200') AND @ip_fc_count = '0'
-- =====================================================================
base_weekly_budget_live AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_compcode,
    bic_zio_cctr,
    bic_zio_pctr,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count = '0'
),

-- =====================================================================
-- CTE 3: weekly_snapshot_union
-- Logic: Union of frozen and live cubes with FLAG indicator, harmonized columns
-- Source: base_weekly_budget_frozen UNION ALL base_weekly_budget_live
-- Calculation: Conditional selection of frozen vs live data based on FLAG
-- =====================================================================
weekly_snapshot_union AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'FC' AS flag
  FROM base_weekly_budget_frozen
  
  UNION ALL
  
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    '' AS b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    '' AS b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'LC' AS flag
  FROM base_weekly_budget_live
),

-- =====================================================================
-- CTE 4: weekly_snapshot_aggregated
-- Logic: Aggregate weekly snapshot with conditional frozen/live logic
-- Source: weekly_snapshot_union
-- Aggregation: GROUP BY all dimensions, SUM(bic_zio_skfq) with IF logic
-- Calculation: IF frozen value is NULL, use live value, else use frozen
-- =====================================================================
weekly_snapshot_aggregated AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    IF(
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq ELSE NULL END) IS NULL,
      SUM(CASE WHEN flag = 'LC' THEN bic_zio_skfq ELSE NULL END),
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq ELSE NULL END)
    ) AS bic_zio_skfq
  FROM weekly_snapshot_union
  GROUP BY
    fiscper, fiscvarnt, bic_zio_sweek, fiscyear, fiscper3,
    b631_s_chrtacct, bic_zio_psty, bic_zio_pskf, b631_s_co_area,
    bic_zio_cmpcd, b631_s_proftctr, b631_s_costcntr, b631_s_funcarea,
    bic_zio_ver, bic_zio_saudt, mandt, recordmode, unit
),

-- =====================================================================
-- CTE 5: weekly_snapshot_filtered
-- Logic: Filter aggregated snapshot by fiscal variant, week range, version, audit, and SKF account
-- Source: weekly_snapshot_aggregated
-- Filters: FISCVARNT='K4', week between parameters, version=IP_VERSION, audit IN (1,10), SKF IN (804013,801000)
-- =====================================================================
weekly_snapshot_filtered AS (
  SELECT *
  FROM weekly_snapshot_aggregated
  WHERE fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 6: hierarchy_nodes_filtered
-- Logic: Filter hierarchy nodes for CORERET parent and valid-to date
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node
-- Filters: PARNODE ends with 'CORERET' AND HRYVALTO='99991231'
-- =====================================================================
hierarchy_nodes_filtered AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
    AND mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 7: weekly_with_hierarchy
-- Logic: Join weekly snapshot with hierarchy nodes on cost center
-- Source: weekly_snapshot_filtered INNER JOIN hierarchy_nodes_filtered
-- Join: ON b631_s_costcntr = nodevalue
-- =====================================================================
weekly_with_hierarchy AS (
  SELECT
    w.fiscper,
    w.fiscvarnt,
    w.fiscyear,
    w.fiscper3,
    w.mandt,
    w.recordmode,
    w.bic_zio_sweek,
    w.b631_s_co_area,
    w.bic_zio_cmpcd,
    w.b631_s_proftctr,
    h.nodevalue AS b631_s_costcntr,
    w.bic_zio_ver,
    w.bic_zio_saudt,
    w.bic_zio_psty,
    w.bic_zio_pskf,
    w.unit,
    w.bic_zio_skfq
  FROM weekly_snapshot_filtered w
  INNER JOIN hierarchy_nodes_filtered h
    ON w.b631_s_costcntr = h.nodevalue
),

-- =====================================================================
-- CTE 8: store_attributes_base
-- Logic: Base store attributes from ZTSRP_ATTR_ACT, filtered by MANDT
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act
-- Filters: MANDT IN ('120','200')
-- Aggregation: SUM of numeric measures, GROUP BY dimensions
-- =====================================================================
store_attributes_base AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE mandt IN ('120', '200')
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat,
    fs_open_dat, fs_close_dat, rep_mkt_code, rep_mkt_desc, address,
    city, state, zipcode, division_code, division_desc, division_mgr,
    area_code, area_desc, area_mgr, region_code, region_desc, region_mgr,
    district_code, district_desc, zone_code, district_mgr, zone_desc,
    major_region_code, major_region, store_type, store_type_desc,
    rx_store_desc, distribution_cntr, distribution_cntr_desc, store_24_hr,
    budget_open_dat, budget_closed_dat, budget_reloc_dat, const_open_dat,
    const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc, original_fs_open_dat,
    original_rx_open_dat, group_one, group_two, ard_sort_key, area_cost_center,
    regiion_cost_center, county_code, county_desc, sqft_bracket_desc,
    sqft_sort_key, acquisition_code, acquisition_desc, photo_lab_add_dat,
    photo_lab_close_dat, min_hrs_ind, createdby, created_on, rx_division_code,
    rx_area_code, rx_region_code, rx_district_code
),

-- =====================================================================
-- CTE 9: weekly_with_store_attrs
-- Logic: Left join weekly data with store attributes on profit center
-- Source: weekly_with_hierarchy LEFT JOIN store_attributes_base
-- Join: ON b631_s_proftctr = prctr
-- =====================================================================
weekly_with_store_attrs AS (
  SELECT
    w.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM weekly_with_hierarchy w
  LEFT JOIN store_attributes_base s
    ON w.b631_s_proftctr = s.prctr
),

-- =====================================================================
-- CTE 10: comp_flags_base
-- Logic: Comp flags from ZTSRP_COMPFL, filtered by MANDT and version
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl
-- Filters: MANDT IN ('110','200') AND COMP_VER = @ip_version
-- =====================================================================
comp_flags_base AS (
  SELECT
    mandt,
    prctr,
    strnum,
    comp_ver,
    rx_comp_wk,
    zweek,
    zmonth,
    zyear
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE mandt IN ('110', '200')
    AND comp_ver = @ip_version
),

-- =====================================================================
-- CTE 11: weekly_with_comp_flags
-- Logic: Left join weekly data with comp flags on profit center and week
-- Source: weekly_with_store_attrs LEFT JOIN comp_flags_base
-- Join: ON b631_s_proftctr = prctr AND bic_zio_sweek = zweek
-- =====================================================================
weekly_with_comp_flags AS (
  SELECT
    w.*,
    c.rx_comp_wk
  FROM weekly_with_store_attrs w
  LEFT JOIN comp_flags_base c
    ON w.b631_s_proftctr = c.prctr
    AND w.bic_zio_sweek = c.zweek
),

-- =====================================================================
-- CTE 12: calendar_week_base
-- Logic: Calendar week attributes from ZTFIGL_RCALWEEK, filtered by RCLNT
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek
-- Filters: RCLNT IN ('120','200')
-- =====================================================================
calendar_week_base AS (
  SELECT
    zrwstrtdate,
    zrwenddate,
    zzweek,
    rclnt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 13: weekly_with_calendar
-- Logic: Left join weekly data with calendar week on week number
-- Source: weekly_with_comp_flags LEFT JOIN calendar_week_base
-- Join: ON bic_zio_sweek = zzweek
-- =====================================================================
weekly_with_calendar AS (
  SELECT
    w.*,
    c.zrwenddate,
    c.zrwstrtdate
  FROM weekly_with_comp_flags w
  LEFT JOIN calendar_week_base c
    ON w.bic_zio_sweek = c.zzweek
),

-- =====================================================================
-- CTE 14: profit_center_text_base
-- Logic: Profit center text from CEPCT, filtered by MANDT
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Filters: MANDT IN ('120','200')
-- =====================================================================
profit_center_text_base AS (
  SELECT
    mandt,
    prctr,
    ltext AS profit_center_text
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 15: weekly_with_pc_text
-- Logic: Left join weekly data with profit center text on profit center
-- Source: weekly_with_calendar LEFT JOIN profit_center_text_base
-- Join: ON b631_s_proftctr = prctr
-- =====================================================================
weekly_with_pc_text AS (
  SELECT
    w.*,
    p.profit_center_text
  FROM weekly_with_calendar w
  LEFT JOIN profit_center_text_base p
    ON w.b631_s_proftctr = p.prctr
),

-- =====================================================================
-- CTE 16: calculated_fields
-- Logic: Add calculated columns for week number, adjusted SKF quantity, and comp flag
-- Source: weekly_with_pc_text
-- Calculation: CAL_WEEK_NUMBER = RIGHT(bic_zio_sweek, 2)
-- Calculation: CAL_BIC_ZIO_SKFQ = CASE WHEN bic_zio_pskf='804013' THEN bic_zio_skfq*2 ELSE bic_zio_skfq
-- Calculation: CAL_RX_COMP_WK = IF rx_comp_wk IS NULL THEN '0' ELSE rx_comp_wk
-- =====================================================================
calculated_fields AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    bic_zio_skfq,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    strnum,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    city,
    state,
    fs_open_dat,
    rx_open_dat,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    rx_comp_wk,
    zrwenddate,
    zrwstrtdate,
    profit_center_text,
    -- Calculated columns
    RIGHT(CAST(bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS NUMERIC) * 2 
      ELSE CAST(bic_zio_skfq AS NUMERIC) 
    END AS cal_bic_zio_skfq,
    CASE 
      WHEN rx_comp_wk IS NULL THEN '0' 
      ELSE rx_comp_wk 
    END AS cal_rx_comp_wk,
    'Y' AS skf_bud_cube
  FROM weekly_with_pc_text
)

-- =====================================================================
-- FINAL SELECT: Aggregated output with all dimensions and measures
-- Logic: Final aggregation over calculated fields
-- Aggregation: GROUP BY all dimensions, SUM measures
-- Measures: bic_zio_skfq_negative (SUM of original), bic_zio_skfq (SUM of calculated)
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  skf_bud_cube,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  SUM(bic_zio_skfq) AS bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS bic_zio_skfq
FROM calculated_fields
GROUP BY
  mandt, fiscper, fiscvarnt, fiscyear, fiscper3, recordmode,
  bic_zio_sweek, b631_s_co_area, bic_zio_cmpcd, b631_s_proftctr,
  b631_s_costcntr, bic_zio_ver, bic_zio_saudt, rep_mkt_code,
  rep_mkt_desc, emerg_mkt_ind, bic_zio_psty, bic_zio_pskf, unit,
  division_code, division_desc, area_code, area_desc, district_code,
  district_desc, region_code, region_desc, cal_week_number, strnum,
  zrwenddate, zrwstrtdate, city, state, profit_center_text,
  fs_open_dat, rx_open_dat, cal_rx_comp_wk, skf_bud_cube,
  rx_division_code, rx_area_code, rx_region_code, rx_district_code
