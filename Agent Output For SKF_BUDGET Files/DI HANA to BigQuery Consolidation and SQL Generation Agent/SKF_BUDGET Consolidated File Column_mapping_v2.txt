-- =====================================================================
-- CONSOLIDATED BIGQUERY SQL FOR CV_COMP_SKF_BUDGET_STATIC
-- Source: HANA Calculation View and Stored Procedure Logic
-- Target: BigQuery Standard SQL
-- Runtime Parameters: @ip_version, @ip_week_ending_from, @ip_week_ending_to, @v_week
-- =====================================================================

-- =====================================================================
-- CTE 1: base_weekly_budget_frozen
-- Logic: Frozen cube data from AZSRP_DS092 filtered by fiscal variant, week range, version, audit trail, and SKF account
-- Source: bic_azsrp_ds092 (via CV_BASE_SKF_WEEKLY_BUDGET_S4 logic)
-- Filters: fiscvarnt='K4', zweek between parameters, version=parameter, saudt IN('1','10'), pskf IN('804013','801000'), mandt IN('110','200'), fc_count!=0
-- =====================================================================
WITH base_weekly_budget_frozen AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek AS zweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'FC' AS flag
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092`
  WHERE mandt IN ('110', '200')
    AND fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 2: base_weekly_budget_live
-- Logic: Live cube data from AZSRP_DS031 filtered by fiscal variant, week range, version, audit trail, and SKF account
-- Source: bic_azsrp_ds031 (via CV_BASE_SKF_WEEKLY_BUDGET_S4 logic)
-- Filters: fiscvarnt='K4', zweek between parameters, version=parameter, saudt IN('1','10'), pskf IN('804013','801000'), mandt IN('110','200'), fc_count=0
-- =====================================================================
base_weekly_budget_live AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek AS zweek,
    fiscyear,
    fiscper3,
    b631_s_compcode,
    bic_zio_cctr,
    bic_zio_pctr,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'LC' AS flag
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031`
  WHERE mandt IN ('110', '200')
    AND fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 3: combined_weekly_budget
-- Logic: Union of frozen and live budget data with column alignment
-- Source: base_weekly_budget_frozen UNION ALL base_weekly_budget_live
-- Calculation: Live cube maps profit center and cost center from different source columns, chart of accounts and functional area set to empty string
-- =====================================================================
combined_weekly_budget AS (
  SELECT
    fiscper,
    fiscvarnt,
    zweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    flag
  FROM base_weekly_budget_frozen
  
  UNION ALL
  
  SELECT
    fiscper,
    fiscvarnt,
    zweek,
    fiscyear,
    fiscper3,
    '' AS b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    '' AS b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    flag
  FROM base_weekly_budget_live
),

-- =====================================================================
-- CTE 4: hierarchy_nodes_filtered
-- Logic: Hierarchy nodes filtered for core retail parent nodes with validity date
-- Source: hrrp_node (via CV_BASE_MD_HRRP_NODE_S4)
-- Filters: parnode ends with 'CORERET', hryvalto='99991231', mandt IN('120','200')
-- =====================================================================
hierarchy_nodes_filtered AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE mandt IN ('120', '200')
    AND REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 5: budget_with_hierarchy
-- Logic: Join weekly budget with hierarchy to map cost centers to hierarchy nodes
-- Source: combined_weekly_budget INNER JOIN hierarchy_nodes_filtered
-- Join: b631_s_costcntr = nodevalue
-- =====================================================================
budget_with_hierarchy AS (
  SELECT
    b.fiscper,
    b.fiscvarnt,
    b.zweek,
    b.fiscyear,
    b.fiscper3,
    b.b631_s_chrtacct,
    b.bic_zio_psty,
    b.bic_zio_pskf,
    b.b631_s_co_area,
    b.bic_zio_cmpcd,
    b.b631_s_proftctr,
    h.nodevalue AS b631_s_costcntr,
    b.b631_s_funcarea,
    b.bic_zio_ver,
    b.bic_zio_saudt,
    b.mandt,
    b.recordmode,
    b.unit,
    b.bic_zio_skfq,
    b.flag
  FROM combined_weekly_budget b
  INNER JOIN hierarchy_nodes_filtered h
    ON b.b631_s_costcntr = h.nodevalue
),

-- =====================================================================
-- CTE 6: store_attributes_from_procedure
-- Logic: Store profile attributes populated by STP_WSS_SRP_ATTRIBUTES procedure from CV_BASE_MD_SRPACT_S4
-- Source: ztsrp_attr_act (via STP_WSS_SRP_ATTRIBUTES → TBL_WSS_SRP_ATTR_ACT → CV_COMP_MD_SRPACT_STATIC)
-- Filters: mandt IN('120','200') from source
-- Aggregation: SUM on measure columns, GROUP BY on attributes
-- =====================================================================
store_attributes_from_procedure AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE mandt IN ('120', '200')
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat, fs_open_dat, fs_close_dat,
    rep_mkt_code, rep_mkt_desc, address, city, state, zipcode, division_code, division_desc, division_mgr,
    area_code, area_desc, area_mgr, region_code, region_desc, region_mgr, district_code, district_desc,
    zone_code, district_mgr, zone_desc, major_region_code, major_region, store_type, store_type_desc,
    rx_store_desc, distribution_cntr, distribution_cntr_desc, store_24_hr, budget_open_dat, budget_closed_dat,
    budget_reloc_dat, const_open_dat, const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc, original_fs_open_dat, original_rx_open_dat,
    group_one, group_two, ard_sort_key, area_cost_center, regiion_cost_center, county_code, county_desc,
    sqft_bracket_desc, sqft_sort_key, acquisition_code, acquisition_desc, photo_lab_add_dat, photo_lab_close_dat,
    min_hrs_ind, createdby, created_on, rx_division_code, rx_area_code, rx_region_code, rx_district_code
),

-- =====================================================================
-- CTE 7: budget_with_store_attributes
-- Logic: Left join budget data with store attributes on profit center
-- Source: budget_with_hierarchy LEFT JOIN store_attributes_from_procedure
-- Join: b631_s_proftctr = prctr
-- =====================================================================
budget_with_store_attributes AS (
  SELECT
    b.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM budget_with_hierarchy b
  LEFT JOIN store_attributes_from_procedure s
    ON b.b631_s_proftctr = s.prctr
),

-- =====================================================================
-- CTE 8: comp_flags_from_procedure
-- Logic: Comp flag data populated by STP_WSS_SRP_ATTRIBUTES procedure from CV_BASE_MD_COMPFL_S4, filtered by week parameter
-- Source: ztsrp_compfl (via STP_WSS_SRP_ATTRIBUTES → TBL_WSS_SRP_COMPFLAG → CV_COMP_MD_COMPFL_STATIC)
-- Filters: mandt IN('110','200') from source, zweek=@v_week from procedure, comp_ver=@ip_version from final consumer
-- =====================================================================
comp_flags_from_procedure AS (
  SELECT
    mandt,
    prctr,
    strnum,
    comp_ver,
    zweek,
    zmonth,
    zyear,
    fs_comp_pre,
    rx_comp_pre,
    fs_comp_mon,
    rx_comp_mon,
    fs_comp_wk,
    rx_comp_wk
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE mandt IN ('110', '200')
    AND zweek = @v_week
    AND comp_ver = @ip_version
),

-- =====================================================================
-- CTE 9: budget_with_comp_flags
-- Logic: Left join budget with store attributes to comp flags on profit center and week
-- Source: budget_with_store_attributes LEFT JOIN comp_flags_from_procedure
-- Join: b631_s_proftctr = prctr AND zweek = zweek
-- =====================================================================
budget_with_comp_flags AS (
  SELECT
    b.*,
    c.rx_comp_wk
  FROM budget_with_store_attributes b
  LEFT JOIN comp_flags_from_procedure c
    ON b.b631_s_proftctr = c.prctr
    AND b.zweek = c.zweek
),

-- =====================================================================
-- CTE 10: calendar_week_attributes
-- Logic: Retail calendar week attributes with start and end dates
-- Source: ztfigl_rcalweek (via CV_BASE_MD_RCALWEEK_S4)
-- Filters: rclnt IN('120','200')
-- =====================================================================
calendar_week_attributes AS (
  SELECT
    rclnt,
    zzweek,
    zcalyrp,
    zryear,
    zrperiod,
    zryrp,
    zryrqtr,
    zrwstrtdate,
    zrwenddate
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 11: budget_with_calendar
-- Logic: Left join budget with comp flags to calendar week on week number
-- Source: budget_with_comp_flags LEFT JOIN calendar_week_attributes
-- Join: zweek = zzweek
-- =====================================================================
budget_with_calendar AS (
  SELECT
    b.*,
    c.zrwenddate,
    c.zrwstrtdate
  FROM budget_with_comp_flags b
  LEFT JOIN calendar_week_attributes c
    ON b.zweek = c.zzweek
),

-- =====================================================================
-- CTE 12: profit_center_text
-- Logic: Profit center text descriptions
-- Source: v_cepct (via CV_BASE_MD_CEPCT_S4)
-- Filters: mandt IN('120','200')
-- =====================================================================
profit_center_text AS (
  SELECT
    mandt,
    spras,
    prctr,
    datbi,
    kokrs,
    ktext,
    ltext,
    mctxt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 13: budget_with_profit_center_text
-- Logic: Left join budget with calendar to profit center text on profit center
-- Source: budget_with_calendar LEFT JOIN profit_center_text
-- Join: b631_s_proftctr = prctr
-- =====================================================================
budget_with_profit_center_text AS (
  SELECT
    b.*,
    p.ltext AS profit_center_text
  FROM budget_with_calendar b
  LEFT JOIN profit_center_text p
    ON b.b631_s_proftctr = p.prctr
),

-- =====================================================================
-- CTE 14: final_with_calculations
-- Logic: Apply calculated columns and prepare for final aggregation
-- Source: budget_with_profit_center_text
-- Calculation: cal_week_number = RIGHT(zweek, 2), cal_bic_zio_skfq = CASE WHEN pskf='804013' THEN skfq*2 ELSE skfq END, cal_rx_comp_wk = IFNULL(rx_comp_wk, '0')
-- =====================================================================
final_with_calculations AS (
  SELECT
    mandt,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    recordmode,
    zweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    RIGHT(zweek, 2) AS cal_week_number,
    strnum,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    IFNULL(rx_comp_wk, '0') AS cal_rx_comp_wk,
    flag AS skf_bud_cube,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    bic_zio_skfq,
    CASE 
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS NUMERIC) * 2 
      ELSE CAST(bic_zio_skfq AS NUMERIC) 
    END AS cal_bic_zio_skfq
  FROM budget_with_profit_center_text
)

-- =====================================================================
-- FINAL SELECT WITH AGGREGATION
-- Logic: Aggregate budget quantities by all dimension attributes
-- Aggregation: SUM(bic_zio_skfq) AS bic_zio_skfq_negative, SUM(cal_bic_zio_skfq) AS bic_zio_skfq
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  zweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  skf_bud_cube,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  SUM(bic_zio_skfq) AS bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS bic_zio_skfq
FROM final_with_calculations
GROUP BY
  mandt, fiscper, fiscvarnt, fiscyear, fiscper3, recordmode, zweek, b631_s_co_area, bic_zio_cmpcd,
  b631_s_proftctr, b631_s_costcntr, bic_zio_ver, bic_zio_saudt, rep_mkt_code, rep_mkt_desc, emerg_mkt_ind,
  bic_zio_psty, bic_zio_pskf, unit, division_code, division_desc, area_code, area_desc, district_code,
  district_desc, region_code, region_desc, cal_week_number, strnum, zrwenddate, zrwstrtdate, city, state,
  profit_center_text, fs_open_dat, rx_open_dat, cal_rx_comp_wk, skf_bud_cube, rx_division_code, rx_area_code,
  rx_region_code, rx_district_code
