-- =====================================================================
-- CONSOLIDATED BIGQUERY SQL FOR CV_COMP_SKF_BUDGET_STATIC
-- =====================================================================
-- Source System: SAP HANA (SAPSEP schema)
-- Target System: Google BigQuery (edp-qa-finstorage.edp_saps4_shared_cnsv)
-- Consolidation Date: 2024
-- 
-- This consolidated SQL implements CV_COMP_SKF_BUDGET_STATIC by fully
-- expanding all dependencies and applying column mappings from the
-- provided Columns_Mapping.csv.
--
-- PARAMETERS (Runtime):
--   @ip_version STRING - Version filter (default: 'LEDGER_BUD')
--   @ip_week_ending_from STRING - Start week filter
--   @ip_week_ending_to STRING - End week filter
--   @v_week STRING - Week parameter for procedure logic
-- =====================================================================

-- =====================================================================
-- CTE 1: frozen_cube
-- Logic: Selects SKF quantity data from frozen cube (DS092)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds092
-- Filters: mandt IN ('110','120','200'), frozen flag condition
-- =====================================================================
WITH frozen_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds092`
  WHERE
    mandt IN ('110', '120', '200')
),

-- =====================================================================
-- CTE 2: live_cube
-- Logic: Selects SKF quantity data from live cube (DS031)
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds031
-- Filters: mandt IN ('110','120','200'), live flag condition
-- =====================================================================
live_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_compcode,
    bic_zio_cctr,
    bic_zio_pctr,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds031`
  WHERE
    mandt IN ('110', '120', '200')
),

-- =====================================================================
-- CTE 3: weekly_snapshot_base
-- Logic: Combines frozen and live cubes with flag indicator
-- Source: frozen_cube UNION ALL live_cube
-- Calculation: FLAG column added ('FC' or 'LC')
-- =====================================================================
weekly_snapshot_base AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'FC' AS flag
  FROM frozen_cube
  
  UNION ALL
  
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    CAST(NULL AS STRING) AS b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    CAST(NULL AS STRING) AS b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'LC' AS flag
  FROM live_cube
),

-- =====================================================================
-- CTE 4: weekly_snapshot_aggregated
-- Logic: Aggregates weekly snapshot with conditional measure logic
-- Source: weekly_snapshot_base
-- Aggregation: Prioritizes frozen (FC) over live (LC) data
-- Calculation: Uses COALESCE to select FC quantity if available, else LC
-- =====================================================================
weekly_snapshot_aggregated AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    COALESCE(
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq END),
      SUM(CASE WHEN flag = 'LC' THEN bic_zio_skfq END)
    ) AS bic_zio_skfq
  FROM weekly_snapshot_base
  GROUP BY
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit
),

-- =====================================================================
-- CTE 5: weekly_snapshot_filtered
-- Logic: Applies business filters to weekly snapshot
-- Source: weekly_snapshot_aggregated
-- Filters: 
--   - fiscvarnt = 'K4'
--   - bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
--   - bic_zio_ver = @ip_version
--   - bic_zio_saudt IN ('1', '10')
--   - bic_zio_pskf IN ('804013', '801000')
-- =====================================================================
weekly_snapshot_filtered AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    bic_zio_skfq
  FROM weekly_snapshot_aggregated
  WHERE
    fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 6: hierarchy_nodes_filtered
-- Logic: Filters hierarchy nodes for retail core nodes
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node
-- Filters: 
--   - mandt IN ('120', '200')
--   - parnode LIKE '%CORERET' (ends with CORERET)
--   - hryvalto = '99991231'
-- =====================================================================
hierarchy_nodes_filtered AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node`
  WHERE
    mandt IN ('120', '200')
    AND REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 7: weekly_with_hierarchy
-- Logic: Joins weekly snapshot with hierarchy nodes
-- Source: weekly_snapshot_filtered INNER JOIN hierarchy_nodes_filtered
-- Join: ON b631_s_costcntr = nodevalue
-- =====================================================================
weekly_with_hierarchy AS (
  SELECT
    w.fiscper,
    w.fiscvarnt,
    w.fiscyear,
    w.fiscper3,
    w.mandt,
    w.recordmode,
    w.bic_zio_sweek,
    w.b631_s_co_area,
    w.bic_zio_cmpcd,
    w.b631_s_proftctr,
    h.nodevalue AS b631_s_costcntr,
    w.bic_zio_ver,
    w.bic_zio_saudt,
    w.bic_zio_psty,
    w.bic_zio_pskf,
    w.unit,
    w.bic_zio_skfq
  FROM weekly_snapshot_filtered w
  INNER JOIN hierarchy_nodes_filtered h
    ON w.b631_s_costcntr = h.nodevalue
),

-- =====================================================================
-- CTE 8: store_attributes
-- Logic: Aggregates store/profit center attributes
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_attr_act
-- Aggregation: SUM on rx_hrs_oper, fs_hrs_oper, rx_store, retail_sqft_amt, total_sqft_amt
-- =====================================================================
store_attributes AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_attr_act`
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat,
    fs_open_dat, fs_close_dat, rep_mkt_code, rep_mkt_desc, address,
    city, state, zipcode, division_code, division_desc, division_mgr,
    area_code, area_desc, area_mgr, region_code, region_desc, region_mgr,
    district_code, district_desc, zone_code, district_mgr, zone_desc,
    major_region_code, major_region, store_type, store_type_desc,
    rx_store_desc, distribution_cntr, distribution_cntr_desc, store_24_hr,
    budget_open_dat, budget_closed_dat, budget_reloc_dat, const_open_dat,
    const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc,
    original_fs_open_dat, original_rx_open_dat, group_one, group_two,
    ard_sort_key, area_cost_center, regiion_cost_center, county_code,
    county_desc, sqft_bracket_desc, sqft_sort_key, acquisition_code,
    acquisition_desc, photo_lab_add_dat, photo_lab_close_dat, min_hrs_ind,
    createdby, created_on, rx_division_code, rx_area_code, rx_region_code,
    rx_district_code
),

-- =====================================================================
-- CTE 9: weekly_with_store_attrs
-- Logic: Left joins weekly data with store attributes
-- Source: weekly_with_hierarchy LEFT JOIN store_attributes
-- Join: ON b631_s_proftctr = prctr
-- =====================================================================
weekly_with_store_attrs AS (
  SELECT
    w.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM weekly_with_hierarchy w
  LEFT JOIN store_attributes s
    ON w.b631_s_proftctr = s.prctr
),

-- =====================================================================
-- CTE 10: comp_flags
-- Logic: Selects completion flags filtered by version
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_compfl
-- Filters: comp_ver = @ip_version
-- =====================================================================
comp_flags AS (
  SELECT
    mandt,
    prctr,
    strnum,
    rx_comp_wk,
    zweek,
    zmonth,
    zyear,
    comp_ver
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_compfl`
  WHERE comp_ver = @ip_version
),

-- =====================================================================
-- CTE 11: weekly_with_comp_flags
-- Logic: Left joins weekly data with completion flags
-- Source: weekly_with_store_attrs LEFT JOIN comp_flags
-- Join: ON b631_s_proftctr = prctr AND bic_zio_sweek = zweek
-- =====================================================================
weekly_with_comp_flags AS (
  SELECT
    w.*,
    c.rx_comp_wk
  FROM weekly_with_store_attrs w
  LEFT JOIN comp_flags c
    ON w.b631_s_proftctr = c.prctr
    AND w.bic_zio_sweek = c.zweek
),

-- =====================================================================
-- CTE 12: calendar_week
-- Logic: Selects calendar week attributes
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek
-- Filters: rclnt IN ('120', '200')
-- =====================================================================
calendar_week AS (
  SELECT
    zrwstrtdate,
    zrwenddate,
    zzweek,
    rclnt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek`
  WHERE rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 13: weekly_with_calendar
-- Logic: Left joins weekly data with calendar week
-- Source: weekly_with_comp_flags LEFT JOIN calendar_week
-- Join: ON bic_zio_sweek = zzweek
-- =====================================================================
weekly_with_calendar AS (
  SELECT
    w.*,
    c.zrwenddate,
    c.zrwstrtdate
  FROM weekly_with_comp_flags w
  LEFT JOIN calendar_week c
    ON w.bic_zio_sweek = c.zzweek
),

-- =====================================================================
-- CTE 14: profit_center_text
-- Logic: Selects profit center text descriptions
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Filters: mandt IN ('110', '120', '200')
-- =====================================================================
profit_center_text AS (
  SELECT
    mandt,
    prctr,
    ltext AS profit_center_text
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN ('110', '120', '200')
),

-- =====================================================================
-- CTE 15: weekly_with_pc_text
-- Logic: Left joins weekly data with profit center text
-- Source: weekly_with_calendar LEFT JOIN profit_center_text
-- Join: ON b631_s_proftctr = prctr
-- =====================================================================
weekly_with_pc_text AS (
  SELECT
    w.*,
    p.profit_center_text
  FROM weekly_with_calendar w
  LEFT JOIN profit_center_text p
    ON w.b631_s_proftctr = p.prctr
),

-- =====================================================================
-- CTE 16: final_with_calculations
-- Logic: Adds calculated columns for final output
-- Source: weekly_with_pc_text
-- Calculation:
--   - cal_week_number: Last 2 digits of bic_zio_sweek
--   - cal_bic_zio_skfq: Doubles quantity if pskf = '804013'
--   - cal_rx_comp_wk: Defaults to '0' if rx_comp_wk is NULL
-- =====================================================================
final_with_calculations AS (
  SELECT
    bic_zio_skfq,
    unit,
    bic_zio_pskf,
    bic_zio_psty,
    rx_comp_wk,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    strnum,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    -- Calculated columns
    RIGHT(CAST(bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS FLOAT64) * 2 
      ELSE CAST(bic_zio_skfq AS FLOAT64) 
    END AS cal_bic_zio_skfq,
    COALESCE(rx_comp_wk, '0') AS cal_rx_comp_wk
  FROM weekly_with_pc_text
)

-- =====================================================================
-- FINAL SELECT: CV_COMP_SKF_BUDGET_STATIC
-- Logic: Final aggregation with GROUP BY on all attributes
-- Source: final_with_calculations
-- Aggregation: 
--   - _BIC_ZIO_SKFQ_NEGATIVE: SUM(bic_zio_skfq)
--   - _BIC_ZIO_SKFQ: SUM(cal_bic_zio_skfq)
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  -- Measures
  SUM(bic_zio_skfq) AS bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS bic_zio_skfq
FROM final_with_calculations
GROUP BY
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code
;
