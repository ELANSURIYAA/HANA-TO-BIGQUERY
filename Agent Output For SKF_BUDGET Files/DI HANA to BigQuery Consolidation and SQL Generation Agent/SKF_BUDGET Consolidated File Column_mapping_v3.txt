-- =====================================================================
-- CONSOLIDATED BIGQUERY SQL FOR SKF_BUDGET DATA PIPELINE
-- Final Consumer: CV_COMP_SKF_BUDGET_STATIC
-- =====================================================================
-- This consolidated SQL inlines all upstream dependencies and applies
-- physical column mappings from hana_to_bq_column_mapping.
-- 
-- Source Artifacts Consolidated:
-- 1. CV_BASE_MD_CEPCT_S4 (Profit Center Text)
-- 2. CV_BASE_MD_COMPFL_S4 (Store Comp Flags)
-- 3. CV_BASE_MD_HRRP_NODE_S4 (Hierarchy Nodes)
-- 4. CV_BASE_MD_RCALWEEK_S4 (Retail Calendar Week)
-- 5. CV_BASE_MD_SRPACT_S4 (Store Attributes)
-- 6. CV_BASE_SKF_WEEKLY_BUDGET_S4 (Weekly Budget - Frozen & Live)
-- 7. STP_WSS_SRP_ATTRIBUTES (Procedure - inlined as data load logic)
-- 8. CV_COMP_MD_COMPFL_STATIC (Static Comp Flags)
-- 9. CV_COMP_MD_SRPACT_STATIC (Static Store Attributes)
-- 10. CV_COMP_SKF_BUDGET_STATIC (Final Consumer)
--
-- Parameters:
--   @ip_version: Budget version (default: 'LEDGER_BUD')
--   @ip_week_ending_from: Week range start
--   @ip_week_ending_to: Week range end
--   @ip_fc_count: Frozen cube flag (derived from ip_version)
-- =====================================================================

-- =====================================================================
-- CTE 1: BASE_CEPCT
-- Logic: Profit Center text dimension from CEPCT table
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Filters: mandt IN (120, 200)
-- =====================================================================
WITH BASE_CEPCT AS (
  SELECT
    mandt,
    spras,
    prctr,
    datbi,
    kokrs,
    ktext,
    ltext,
    mctxt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 2: BASE_COMPFL
-- Logic: Store profile comp flags from ZTSRP_COMPFL table
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl
-- Filters: mandt IN (110, 200)
-- =====================================================================
BASE_COMPFL AS (
  SELECT
    mandt,
    prctr,
    strnum,
    comp_ver,
    zweek,
    zmonth,
    zyear,
    fs_comp_pre,
    rx_comp_pre,
    fs_comp_mon,
    rx_comp_mon,
    fs_comp_wk,
    rx_comp_wk
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE mandt IN ('110', '200')
),

-- =====================================================================
-- CTE 3: BASE_HRRP_NODE
-- Logic: Hierarchy node data from HRRP_NODE table
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node
-- Filters: mandt IN (120, 200)
-- =====================================================================
BASE_HRRP_NODE AS (
  SELECT
    mandt,
    hryid,
    hryver,
    nodecls,
    hrynode,
    parnode,
    hryvalto,
    hryvalfrom,
    balind,
    nodetype,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 4: BASE_RCALWEEK
-- Logic: Retail calendar week data from ZTFIGL_RCALWEEK table
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek
-- Filters: rclnt IN (120, 200)
-- =====================================================================
BASE_RCALWEEK AS (
  SELECT
    rclnt,
    zzweek,
    zcalyrp,
    zryear,
    zrperiod,
    zryrp,
    zryrqtr,
    zrwstrtdate,
    zrwenddate
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 5: BASE_SRPACT
-- Logic: Store attributes from ZTSRP_ATTR_ACT table with aggregation
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act
-- Filters: mandt IN (120, 200)
-- Aggregation: SUM on measures (rx_hrs_oper, fs_hrs_oper, retail_sqft_amt, total_sqft_amt)
-- =====================================================================
BASE_SRPACT AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE mandt IN ('120', '200')
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat, fs_open_dat, fs_close_dat,
    rep_mkt_code, rep_mkt_desc, address, city, state, zipcode, division_code, division_desc,
    division_mgr, area_code, area_desc, area_mgr, region_code, region_desc, region_mgr,
    district_code, district_desc, zone_code, district_mgr, zone_desc, major_region_code,
    major_region, store_type, store_type_desc, rx_store, rx_store_desc, distribution_cntr,
    distribution_cntr_desc, store_24_hr, budget_open_dat, budget_closed_dat, budget_reloc_dat,
    const_open_dat, const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc, original_fs_open_dat,
    original_rx_open_dat, group_one, group_two, ard_sort_key, area_cost_center,
    regiion_cost_center, county_code, county_desc, sqft_bracket_desc, sqft_sort_key,
    acquisition_code, acquisition_desc, photo_lab_add_dat, photo_lab_close_dat, min_hrs_ind,
    createdby, created_on, rx_division_code, rx_area_code, rx_region_code, rx_district_code
),

-- =====================================================================
-- CTE 6: FROZEN_CUBE
-- Logic: Frozen budget data from AZSRP_DS092_VT_S4
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092
-- Filters: mandt IN (110, 200) AND @ip_fc_count != '0'
-- =====================================================================
FROZEN_CUBE AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds092`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count != '0'
),

-- =====================================================================
-- CTE 7: LIVE_CUBE
-- Logic: Live budget data from AZSRP_DS031_VT_S4
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031
-- Filters: mandt IN (110, 200) AND @ip_fc_count = '0'
-- =====================================================================
LIVE_CUBE AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_compcode,
    bic_zio_cctr,
    bic_zio_pctr,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds031`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count = '0'
),

-- =====================================================================
-- CTE 8: WEEKLY_BUDGET_UNION
-- Logic: UNION ALL of Frozen and Live cubes with column alignment
-- Calculation: Conditional measure selection based on FLAG
-- =====================================================================
WEEKLY_BUDGET_UNION AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'FC' AS flag
  FROM FROZEN_CUBE
  
  UNION ALL
  
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    '' AS b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    '' AS b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    bic_zio_skfq,
    'LC' AS flag
  FROM LIVE_CUBE
),

-- =====================================================================
-- CTE 9: WEEKLY_BUDGET_AGGREGATED
-- Logic: Aggregate weekly budget with conditional measure calculation
-- Calculation: IF(ISNULL(RES_QUANT_FC), RES_QUANT_LC, RES_QUANT_FC)
-- Aggregation: GROUP BY all non-measure columns, SUM on bic_zio_skfq
-- =====================================================================
WEEKLY_BUDGET_AGGREGATED AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    bic_zio_psty,
    bic_zio_pskf,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    recordmode,
    unit,
    COALESCE(
      SUM(CASE WHEN flag = 'FC' THEN bic_zio_skfq END),
      SUM(CASE WHEN flag = 'LC' THEN bic_zio_skfq END)
    ) AS bic_zio_skfq
  FROM WEEKLY_BUDGET_UNION
  GROUP BY
    fiscper, fiscvarnt, bic_zio_sweek, fiscyear, fiscper3, b631_s_chrtacct,
    bic_zio_psty, bic_zio_pskf, b631_s_co_area, bic_zio_cmpcd, b631_s_proftctr,
    b631_s_costcntr, b631_s_funcarea, bic_zio_ver, bic_zio_saudt, mandt,
    recordmode, unit
),

-- =====================================================================
-- CTE 10: WEEKLY_SNAPSHOT_FILTERED
-- Logic: Apply final filters on weekly budget for SKF budget processing
-- Source: WEEKLY_BUDGET_AGGREGATED
-- Filters: fiscvarnt='K4', bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to,
--          bic_zio_ver=@ip_version, bic_zio_saudt IN (1,10), bic_zio_pskf IN (804013,801000)
-- =====================================================================
WEEKLY_SNAPSHOT_FILTERED AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    bic_zio_skfq
  FROM WEEKLY_BUDGET_AGGREGATED
  WHERE fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 11: HIERARCHY_FILTERED
-- Logic: Filter hierarchy nodes for CORERET pattern and valid-to date
-- Source: BASE_HRRP_NODE
-- Filters: parnode ends with 'CORERET', hryvalto = '99991231'
-- =====================================================================
HIERARCHY_FILTERED AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM BASE_HRRP_NODE
  WHERE REGEXP_CONTAINS(parnode, r'CORERET$')
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 12: WEEKLY_WITH_HIERARCHY
-- Logic: Join weekly snapshot with hierarchy nodes
-- Join: INNER JOIN on b631_s_costcntr = nodevalue
-- =====================================================================
WEEKLY_WITH_HIERARCHY AS (
  SELECT
    w.fiscper,
    w.fiscvarnt,
    w.fiscyear,
    w.fiscper3,
    w.mandt,
    w.recordmode,
    w.bic_zio_sweek,
    w.b631_s_co_area,
    w.bic_zio_cmpcd,
    w.b631_s_proftctr,
    h.nodevalue AS b631_s_costcntr,
    w.bic_zio_ver,
    w.bic_zio_saudt,
    w.bic_zio_psty,
    w.bic_zio_pskf,
    w.unit,
    w.bic_zio_skfq
  FROM WEEKLY_SNAPSHOT_FILTERED w
  INNER JOIN HIERARCHY_FILTERED h
    ON w.b631_s_costcntr = h.nodevalue
),

-- =====================================================================
-- CTE 13: WITH_STORE_ATTRIBUTES
-- Logic: Left join with store attributes
-- Join: LEFT JOIN on b631_s_proftctr = prctr
-- Source: BASE_SRPACT (aggregated store attributes)
-- =====================================================================
WITH_STORE_ATTRIBUTES AS (
  SELECT
    w.*,
    s.rep_mkt_code,
    s.rep_mkt_desc,
    s.emerg_mkt_ind,
    s.strnum,
    s.division_code,
    s.division_desc,
    s.area_code,
    s.area_desc,
    s.district_code,
    s.district_desc,
    s.region_code,
    s.region_desc,
    s.city,
    s.state,
    s.fs_open_dat,
    s.rx_open_dat,
    s.rx_division_code,
    s.rx_area_code,
    s.rx_region_code,
    s.rx_district_code
  FROM WEEKLY_WITH_HIERARCHY w
  LEFT JOIN BASE_SRPACT s
    ON w.b631_s_proftctr = s.prctr
),

-- =====================================================================
-- CTE 14: WITH_COMP_FLAGS
-- Logic: Left join with comp flags filtered by version
-- Join: LEFT JOIN on b631_s_proftctr = prctr AND bic_zio_sweek = zweek
-- Source: BASE_COMPFL
-- Filters: comp_ver = @ip_version
-- =====================================================================
WITH_COMP_FLAGS AS (
  SELECT
    w.*,
    c.rx_comp_wk
  FROM WITH_STORE_ATTRIBUTES w
  LEFT JOIN (
    SELECT
      mandt,
      prctr,
      strnum,
      rx_comp_wk,
      zweek,
      zmonth,
      zyear,
      comp_ver
    FROM BASE_COMPFL
    WHERE comp_ver = @ip_version
  ) c
    ON w.b631_s_proftctr = c.prctr
    AND w.bic_zio_sweek = c.zweek
),

-- =====================================================================
-- CTE 15: WITH_CALENDAR_WEEK
-- Logic: Left join with retail calendar week
-- Join: LEFT JOIN on bic_zio_sweek = zzweek
-- Source: BASE_RCALWEEK
-- =====================================================================
WITH_CALENDAR_WEEK AS (
  SELECT
    w.*,
    c.zrwenddate,
    c.zrwstrtdate
  FROM WITH_COMP_FLAGS w
  LEFT JOIN BASE_RCALWEEK c
    ON w.bic_zio_sweek = c.zzweek
),

-- =====================================================================
-- CTE 16: WITH_PROFIT_CENTER_TEXT
-- Logic: Left join with profit center text
-- Join: LEFT JOIN on b631_s_proftctr = prctr
-- Source: BASE_CEPCT
-- =====================================================================
WITH_PROFIT_CENTER_TEXT AS (
  SELECT
    w.*,
    p.ltext AS profit_center_text
  FROM WITH_CALENDAR_WEEK w
  LEFT JOIN BASE_CEPCT p
    ON w.b631_s_proftctr = p.prctr
),

-- =====================================================================
-- CTE 17: FINAL_CALCULATED_COLUMNS
-- Logic: Apply calculated columns for final output
-- Calculation: 
--   - cal_week_number: RIGHT(bic_zio_sweek, 2)
--   - cal_bic_zio_skfq: CASE WHEN bic_zio_pskf='804013' THEN bic_zio_skfq*2 ELSE bic_zio_skfq
--   - cal_rx_comp_wk: COALESCE(rx_comp_wk, '0')
-- =====================================================================
FINAL_CALCULATED_COLUMNS AS (
  SELECT
    mandt,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    division_code,
    division_desc,
    area_code,
    area_desc,
    district_code,
    district_desc,
    region_code,
    region_desc,
    strnum,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    bic_zio_skfq,
    -- Calculated columns
    RIGHT(CAST(bic_zio_sweek AS STRING), 2) AS cal_week_number,
    CASE 
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS NUMERIC) * 2 
      ELSE CAST(bic_zio_skfq AS NUMERIC) 
    END AS cal_bic_zio_skfq,
    COALESCE(rx_comp_wk, '0') AS cal_rx_comp_wk
  FROM WITH_PROFIT_CENTER_TEXT
)

-- =====================================================================
-- FINAL OUTPUT: CV_COMP_SKF_BUDGET_STATIC
-- Logic: Final aggregation with GROUP BY on all non-measure columns
-- Aggregation: 
--   - bic_zio_skfq_negative: SUM(bic_zio_skfq)
--   - bic_zio_skfq: SUM(cal_bic_zio_skfq)
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  division_code,
  division_desc,
  area_code,
  area_desc,
  district_code,
  district_desc,
  region_code,
  region_desc,
  cal_week_number,
  strnum,
  zrwenddate,
  zrwstrtdate,
  city,
  state,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_rx_comp_wk AS rx_comp_wk,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  SUM(bic_zio_skfq) AS bic_zio_skfq_negative,
  SUM(cal_bic_zio_skfq) AS bic_zio_skfq
FROM FINAL_CALCULATED_COLUMNS
GROUP BY
  mandt, fiscper, fiscvarnt, fiscyear, fiscper3, recordmode, bic_zio_sweek,
  b631_s_co_area, bic_zio_cmpcd, b631_s_proftctr, b631_s_costcntr, bic_zio_ver,
  bic_zio_saudt, rep_mkt_code, rep_mkt_desc, emerg_mkt_ind, bic_zio_psty,
  bic_zio_pskf, unit, division_code, division_desc, area_code, area_desc,
  district_code, district_desc, region_code, region_desc, cal_week_number,
  strnum, zrwenddate, zrwstrtdate, city, state, profit_center_text,
  fs_open_dat, rx_open_dat, cal_rx_comp_wk, rx_division_code, rx_area_code,
  rx_region_code, rx_district_code;
