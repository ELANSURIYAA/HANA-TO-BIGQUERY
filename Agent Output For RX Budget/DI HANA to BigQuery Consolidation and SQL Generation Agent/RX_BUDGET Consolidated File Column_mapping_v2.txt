-- =====================================================================
-- CTE 9: Combined_Budget_Data_With_Measures
-- Logic: Union of frozen and live cube data with calculated amount measure
--        Replaces: CV_BASE_FIN_WEEKLY_BUDGET_S4 final output
-- Source: Frozen_Cube_Budget_Data UNION ALL Live_Cube_Budget_Data
-- Calculation: b631_s_amount = COALESCE(b631_s_amount_negative from FC, b631_s_amount_negative from LC) * -1
--              (Implements: IF(ISNULL(RES_AMOUNT_FC), RES_AMOUNT_LC, RES_AMOUNT_FC))
-- Aggregation: SUM on all three amount measures grouped by all non-measure columns
-- =====================================================================
Combined_Budget_Data_With_Measures AS (
  SELECT
    flag,
    mandt,
    bic_zio_sweek,
    b631_s_proftctr,
    b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zwwpc_pa1,
    SUM(b631_s_amount_negative) AS b631_s_amount_negative,
    SUM(bic_zio_amt) AS bic_zio_amt,
    SUM(b631_s_amount_negative * -1) AS b631_s_amount
  FROM (
    SELECT * FROM Frozen_Cube_Budget_Data
    UNION ALL
    SELECT * FROM Live_Cube_Budget_Data
  )
  GROUP BY
    flag, mandt, bic_zio_sweek, b631_s_proftctr, b631_s_costcntr,
    bic_zio_ver, bic_zio_saudt, bic_zwwpc_pa1
),

-- Source: Combined_Budget_Data_With_Measures (fact)
-- Join: INNER JOIN Hierarchy_Nodes_Filtered ON b631_s_proftctr = nodevalue
-- Join: LEFT OUTER JOIN Composite_Store_Attributes_Static ON b631_s_proftctr = prctr
-- Join: LEFT OUTER JOIN Composite_Completion_Flags_Static ON b631_s_proftctr = prctr AND bic_zio_sweek = zweek
-- Join: LEFT OUTER JOIN Retail_Calendar_Weeks ON bic_zio_sweek = zzweek
-- Join: LEFT OUTER JOIN Profit_Center_Texts ON b631_s_proftctr = prctr
-- Filters: FISCVARNT = 'K4', bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
-- Calculation: 
--   - CAL_FS_RX_FLAG: CASE LEFT(bic_zwwpc_pa1,2) WHEN 'FS' THEN 'FS' WHEN 'RX' THEN 'RX' ELSE '' END
--   - CAL_COMP_FLAG: IF fs_comp_wk IS NULL THEN '0' ELSE CASE bic_zio_saudt WHEN '1' THEN fs_comp_wk WHEN '10' THEN rx_comp_wk ELSE '' END
--   - CAL_WEEK_NUMBER: RIGHT(bic_zio_sweek,2)
--   - FIN_BUD_CUBE: flag (from union)
-- Aggregation: SUM on b631_s_amount_negative, bic_zio_amt, b631_s_amount
-- =====================================================================
Final_Budget_Static_With_All_Joins AS (
  SELECT
    -- Dimension columns from fact table
    fact.mandt,
    fact.bic_zio_sweek,
    fact.b631_s_proftctr,
    fact.b631_s_costcntr,
    fact.bic_zio_ver,
    fact.bic_zio_saudt,
    fact.bic_zwwpc_pa1,
    
    -- Calculated columns
    CASE LEFT(fact.bic_zwwpc_pa1, 2)
      WHEN 'FS' THEN 'FS'
      WHEN 'RX' THEN 'RX'
      ELSE ''
    END AS fs_rx_flag,
    
    CASE
      WHEN comp.fs_comp_wk IS NULL THEN '0'
      ELSE CASE fact.bic_zio_saudt
        WHEN '1' THEN comp.fs_comp_wk
        WHEN '10' THEN comp.rx_comp_wk
        ELSE ''
      END
    END AS comp_flag,
    
    RIGHT(fact.bic_zio_sweek, 2) AS week_number,
    
    fact.flag AS fin_bud_cube,
    
    -- Dimension columns from hierarchy
    hier.objvers,
    hier.hryid,
    hier.hrynode,
    hier.nodevalue,
    hier.parnode,
    hier.hryvalfr,
    hier.hryvalto,
    hier.nodetype,
    hier.hrylevl,
    hier.childnum,
    
    -- Dimension columns from store attributes
    attr.strnum,
    attr.kokrs,
    attr.kostl,
    attr.rx_open_dat,
    attr.rx_close_dat,
    attr.rx_relo_dat,
    attr.fs_open_dat,
    attr.fs_close_dat,
    attr.fs_relo_dat,
    attr.rx_open_wk,
    attr.rx_close_wk,
    attr.rx_relo_wk,
    attr.fs_open_wk,
    attr.fs_close_wk,
    attr.fs_relo_wk,
    attr.rx_open_mon,
    attr.rx_close_mon,
    attr.rx_relo_mon,
    attr.fs_open_mon,
    attr.fs_close_mon,
    attr.fs_relo_mon,
    attr.rx_open_yr,
    attr.rx_close_yr,
