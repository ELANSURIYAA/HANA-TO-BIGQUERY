-- =====================================================================
-- CTE 1: base_weekly_budget_frozen_cube
-- Logic: Reads frozen cube budget data (DS052), filters by client and FC flag
-- Source: AZSRP_DS052_VT_S4 (via CV_BASE_FIN_WEEKLY_BUDGET_S4)
-- Filters: MANDT IN ('110','200'), @ip_fc_count != '0'
-- =====================================================================
WITH base_weekly_budget_frozen_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    b631_s_gl_acct,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    bic_zwwpc_pa1,
    bic_zwwsc_pa1,
    recordmode,
    currency,
    b631_s_amount,
    bic_zio_amt,
    'FC' AS flag
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds202`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count != '0'
),

-- =====================================================================
-- CTE 2: base_weekly_budget_live_cube
-- Logic: Reads live cube budget data (DS041), filters by client and FC flag, remaps profit center and cost center
-- Source: AZSRP_DS041_VT_S4 (via CV_BASE_FIN_WEEKLY_BUDGET_S4)
-- Filters: MANDT IN ('110','200'), @ip_fc_count = '0'
-- Calculation: Remap bic_zio_pctr to b631_s_proftctr, bic_zio_cctr to b631_s_costcntr
-- =====================================================================
base_weekly_budget_live_cube AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    b631_s_gl_acct,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr AS b631_s_proftctr,
    bic_zio_cctr AS b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    bic_zwwpc_pa1,
    bic_zwwsc_pa1,
    recordmode,
    currency,
    b631_s_amount,
    bic_zio_amt,
    'LC' AS flag
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds041`
  WHERE mandt IN ('110', '200')
    AND @ip_fc_count = '0'
),

-- =====================================================================
-- CTE 3: combined_weekly_budget
-- Logic: Union frozen and live cube data
-- Source: base_weekly_budget_frozen_cube UNION ALL base_weekly_budget_live_cube
-- =====================================================================
combined_weekly_budget AS (
  SELECT * FROM base_weekly_budget_frozen_cube
  UNION ALL
  SELECT * FROM base_weekly_budget_live_cube
),

-- =====================================================================
-- CTE 4: weekly_snapshot_ds05
-- Logic: Aggregates combined budget data, applies fiscal and version filters
-- Source: combined_weekly_budget
-- Filters: fiscvarnt = 'K4', bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to, bic_zio_ver = @ip_version, bic_zio_saudt IN ('1', '10')
-- Aggregation: SUM(b631_s_amount), SUM(bic_zio_amt), GROUP BY all non-measures
-- Calculation: Restricted measures for LC and FC flags, calculated measure b631_s_amount = COALESCE(FC amount, LC amount)
-- =====================================================================
weekly_snapshot_ds05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    b631_s_gl_acct,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    bic_zwwpc_pa1,
    bic_zwwsc_pa1,
    recordmode,
    currency,
    flag,
    SUM(bic_zio_amt) AS bic_zio_amt,
    SUM(CASE WHEN flag = 'LC' THEN b631_s_amount ELSE 0 END) AS res_amount_lc,
    SUM(CASE WHEN flag = 'FC' THEN b631_s_amount ELSE 0 END) AS res_amount_fc
  FROM combined_weekly_budget
  WHERE fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
  GROUP BY
    fiscper, fiscvarnt, bic_zio_sweek, fiscyear, fiscper3,
    b631_s_chrtacct, b631_s_gl_acct, b631_s_co_area, bic_zio_cmpcd,
    b631_s_proftctr, b631_s_costcntr, b631_s_funcarea, bic_zio_ver,
    bic_zio_saudt, mandt, bic_zwwpc_pa1, bic_zwwsc_pa1, recordmode,
    currency, flag
),

-- =====================================================================
-- CTE 5: weekly_snapshot_with_calculated_amount
-- Logic: Computes final b631_s_amount using restricted measures
-- Source: weekly_snapshot_ds05
-- Calculation: b631_s_amount = CASE WHEN res_amount_fc IS NULL OR res_amount_fc = 0 THEN res_amount_lc ELSE res_amount_fc END
-- =====================================================================
weekly_snapshot_with_calculated_amount AS (
  SELECT
    fiscper,
    fiscvarnt,
    bic_zio_sweek,
    fiscyear,
    fiscper3,
    b631_s_chrtacct,
    b631_s_gl_acct,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    mandt,
    bic_zwwpc_pa1,
    bic_zwwsc_pa1,
    recordmode,
    currency,
    flag,
    bic_zio_amt,
    CASE
      WHEN res_amount_fc IS NULL OR res_amount_fc = 0 THEN res_amount_lc
      ELSE res_amount_fc
    END AS b631_s_amount
  FROM weekly_snapshot_ds05
),

    fs_open_dat,

    rx_district_code,
    CASE LEFT(CAST(bic_zwwpc_pa1 AS STRING), 2)
      WHEN 'FS' THEN 'FS'
      WHEN 'RX' THEN 'RX'
      ELSE ''
    END AS cal_fs_rx_flag,
    COALESCE(

        WHEN 'RX' THEN rx_comp_wk

    ) AS cal_comp_flag,
    RIGHT(CAST(bic_zio_sweek AS STRING), 2) AS cal_week_number,
    b631_s_amount * -1 AS b631_s_amount_negative
  FROM week_number_with_pc_text
)
