-- Filters: _BIC_ZIO_SWEEK BETWEEN @ip_week_ending_from AND @ip_week_ending_to
-- Join: b.MANDT = w.RCLNT AND b._BIC_ZIO_SWEEK = w.RCALWEEK
-- Aggregation: None
-- Calculation: None

  SELECT
    b.*,
    w.rcalweek,
    w.rcalweek_txt,
    w.rcalmonth,
    w.rcalmonth_txt,
    w.rcalyear,
    w.rcalyear_txt,
    w.rcalqtr,
    w.rcalqtr_txt
  FROM
    Budget_With_Comp_Flags b
  LEFT JOIN
    CV_BASE_MD_RCALWEEK_S4 w
    ON b.mandt = w.rclnt
    AND b._bic_zio_sweek = w.rcalweek
  WHERE
    b._bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
),

-- =====================================================================

    p.datbi,
    p.kokrs,
    p.ktext,
    p.ltext,
    p.mctxt

    Budget_With_Calendar b
  LEFT JOIN
    CV_BASE_MD_CEPCT_S4 p
    ON b.mandt = p.mandt
    AND b._b631_s_proftctr = p.prctr
)
