WITH

    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _B631_S_GL_ACCT,

    _BIC_ZIO_CMPCD,

    `PROJECT.DATASET.AZSRP_DS052_VT_S4`
  WHERE
    MANDT IN ('110', '200')
    AND @IP_FC_COUNT != '0'
),

Live_Cube AS (
  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _B631_S_GL_ACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _BIC_ZIO_PCTR AS _B631_S_PROFTCTR,
    _BIC_ZIO_CCTR AS _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    RECORDMODE,
    CURRENCY,
    _B631_S_AMOUNT,
    _BIC_ZIO_AMT,
    'LC' AS FLAG
  FROM
    `PROJECT.DATASET.AZSRP_DS041_VT_S4`
  WHERE
    MANDT IN ('110', '200')
    AND @IP_FC_COUNT = '0'
),

Combined_Frozen_Live_Cube AS (
  SELECT * FROM Frozen_Cube
  UNION ALL
  SELECT * FROM Live_Cube
),

Aggregated_Weekly_Budget AS (
  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _B631_S_GL_ACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,

  FROM
    Combined_Frozen_Live_Cube
  GROUP BY
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _B631_S_GL_ACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,

    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    RECORDMODE,
    CURRENCY,
    FLAG
),

RES_AMOUNT_LC AS (
  SELECT
    *,
    CASE WHEN FLAG = 'LC' THEN _B631_S_AMOUNT_DUMMY ELSE 0 END AS RES_AMOUNT_LC
  FROM
    Aggregated_Weekly_Budget

  SELECT
    *,
    CASE WHEN FLAG = 'FC' THEN _B631_S_AMOUNT_DUMMY ELSE 0 END AS RES_AMOUNT_FC

Summed_Measures AS (

    _BIC_ZIO_SAUDT,
    MANDT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,

    -- FLAG intentionally omitted from output as per logical model (if required, add it back)
    SUM(_BIC_ZIO_AMT) AS _BIC_ZIO_AMT,
    SUM(CASE WHEN FLAG = 'LC' THEN _B631_S_AMOUNT_DUMMY ELSE 0 END) AS RES_AMOUNT_LC,

  FROM
    Aggregated_Weekly_Budget
  GROUP BY
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,

    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    RECORDMODE,
    CURRENCY
)

SELECT
  FISCPER,
  FISCVARNT,
  _BIC_ZIO_SWEEK,
  FISCYEAR,
  FISCPER3,
  _B631_S_CHRTACCT,
  _B631_S_GL_ACCT,
  _B631_S_CO_AREA,
  _BIC_ZIO_CMPCD,
  _B631_S_PROFTCTR,
  _B631_S_COSTCNTR,
  _B631_S_FUNCAREA,
  _BIC_ZIO_VER,
  _BIC_ZIO_SAUDT,

  -- Calculated measure as per logical model: IF(ISNULL(RES_AMOUNT_FC), RES_AMOUNT_LC, RES_AMOUNT_FC)
  CASE
    WHEN RES_AMOUNT_FC IS NULL OR RES_AMOUNT_FC = 0 THEN RES_AMOUNT_LC
    ELSE RES_AMOUNT_FC
  END AS _B631_S_AMOUNT
FROM
  Summed_Measures
;
