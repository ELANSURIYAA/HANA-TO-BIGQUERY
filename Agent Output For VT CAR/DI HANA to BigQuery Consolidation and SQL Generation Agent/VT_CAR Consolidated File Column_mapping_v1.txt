-- CTE 9: SCRIPT_KPIs
-- Logic: Derives calculated KPI columns from script/RX data joined with currency.
--        Implements SCRIPT_KPIs node from CV_COMP_FLASH_SALES:
--        - CAL_FS_UNITS: 1 if ZZ_CUSTTYPE IN ('F','M','R'), -1 if IN ('X','Y','Z'), else 0
--        - CAL_RX_CNT_GE84_NS: 0 if empty string, else CAST to INT64
--        - CAL_RX_CNT_GE84_RE: 0 if empty string, else CAST to INT64
--        - CAL_RX_CNT_NS: 0 if empty string, else CAST to INT64
--        - CAL_RX_CNT_RE: 0 if ZZ_RX_CNT_REFILL empty, else CAST to INT64
--        - CAL_CVD_UNITS: 1 if WORKSTATIONID='0000000555' AND ZZ_CUSTTYPE='C',
--                         -1 if WORKSTATIONID='0000000555' AND ZZ_CUSTTYPE='W', else 0
-- Source: Joined_Scripts_With_Currency
-- Calculation: CAL_FS_UNITS, CAL_RX_CNT_GE84_NS, CAL_RX_CNT_GE84_RE,
--              CAL_RX_CNT_NS, CAL_RX_CNT_RE, CAL_CVD_UNITS
-- =====================================================================
SCRIPT_KPIs AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE,
    ZZ_CUSTTYPE,
    CURRENCY,
    ZZ_UPD_TIMESTAMP AS scripts_upd_timestamp,
    WORKSTATIONID,
    CASE
      WHEN ZZ_CUSTTYPE IN ('F', 'M', 'R') THEN 1
      WHEN ZZ_CUSTTYPE IN ('X', 'Y', 'Z') THEN -1
      ELSE 0
    END AS cal_fs_units,
    CASE
      WHEN ZZ_RX_CNT_GE84_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_NS AS INT64)
    END AS cal_rx_cnt_ge84_ns,
    CASE
      WHEN ZZ_RX_CNT_GE84_RE = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_RE AS INT64)
    END AS cal_rx_cnt_ge84_re,
    CASE
      WHEN ZZ_RX_CNT_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_NS AS INT64)
    END AS cal_rx_cnt_ns,
    CASE
      WHEN ZZ_RX_CNT_REFILL = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_REFILL AS INT64)
    END AS cal_rx_cnt_re,
    CASE
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'C' THEN 1
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'W' THEN -1
      ELSE 0
    END AS cal_cvd_units
  FROM Joined_Scripts_With_Currency
),

-- =====================================================================
-- CTE 21: Combined_Flash_Sales_Inputs

--        its specific metric(s) and NULLs for all others:
--        Branch 1: SCRIPT_KPIs - script/RX counters, cal_fs_units, cal_cvd_units
--        Branch 2: Joined_FS_Sales_With_Retail_Types - fs_salesamount
--        Branch 3: Joined_Employee_Discount_With_Discount_Types - emp_reductionamount
--        Branch 4: Joined_RX_Sales_With_Retail_Types - rx_salesamount
--        Branch 5: Joined_FS_Discount_With_Discount_Types - reductionamount
--        Branch 6: Joined_COVID_Sales_With_RX_Retail_Types - cvd_amount
-- Source: All six joined/projected CTEs above
-- =====================================================================

    BUSINESSDAYDATE,

    CAST(NULL AS NUMERIC) AS emp_reductionamount,
    CAST(NULL AS NUMERIC) AS rx_salesamount,
    CAST(NULL AS NUMERIC) AS reductionamount,
    cal_rx_cnt_ns,
    cal_rx_cnt_re,
    CURRENCY AS transcurrency,
    CAST(NULL AS STRING) AS fs_upd_timestamp,
    CAST(NULL AS STRING) AS rx_upd_timestamp,
    scripts_upd_timestamp,
    scripts_upd_timestamp AS zz_upd_timestamp,
    CAST(NULL AS NUMERIC) AS cvd_amount,
    cal_cvd_units

  UNION ALL

    BUSINESSDAYDATE,

    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_RE,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_RE,
    CAST(NULL AS INT64) AS cal_fs_units,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_re,
    fs_salesamount,
    CAST(NULL AS NUMERIC) AS emp_reductionamount,
    CAST(NULL AS NUMERIC) AS rx_salesamount,
    CAST(NULL AS NUMERIC) AS reductionamount,
    CAST(NULL AS INT64) AS cal_rx_cnt_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_re,
    TRANSCURRENCY AS transcurrency,
    fs_upd_timestamp,
    CAST(NULL AS STRING) AS rx_upd_timestamp,
    CAST(NULL AS STRING) AS scripts_upd_timestamp,
    fs_upd_timestamp AS zz_upd_timestamp,
    CAST(NULL AS NUMERIC) AS cvd_amount,
    CAST(NULL AS INT64) AS cal_cvd_units
  FROM Joined_FS_Sales_With_Retail_Types

  UNION ALL

  -- Branch 3: Employee Discount
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_REFILL,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_RE,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_RE,
    CAST(NULL AS INT64) AS cal_fs_units,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_re,
    CAST(NULL AS NUMERIC) AS fs_salesamount,
    emp_reductionamount,
    CAST(NULL AS NUMERIC) AS rx_salesamount,
    CAST(NULL AS NUMERIC) AS reductionamount,
    CAST(NULL AS INT64) AS cal_rx_cnt_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_re,
    TRANSCURRENCY AS transcurrency,
    CAST(NULL AS STRING) AS fs_upd_timestamp,
    CAST(NULL AS STRING) AS rx_upd_timestamp,
    CAST(NULL AS STRING) AS scripts_upd_timestamp,
    emp_disc_upd_timestamp AS zz_upd_timestamp,
    CAST(NULL AS NUMERIC) AS cvd_amount,
    CAST(NULL AS INT64) AS cal_cvd_units
  FROM Joined_Employee_Discount_With_Discount_Types

  UNION ALL

  -- Branch 4: RX Sales
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_REFILL,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_RE,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_RE,
    CAST(NULL AS INT64) AS cal_fs_units,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_re,
    CAST(NULL AS NUMERIC) AS fs_salesamount,
    CAST(NULL AS NUMERIC) AS emp_reductionamount,
    rx_salesamount,
    CAST(NULL AS NUMERIC) AS reductionamount,
    CAST(NULL AS INT64) AS cal_rx_cnt_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_re,
    TRANSCURRENCY AS transcurrency,

    rx_upd_timestamp,
    CAST(NULL AS STRING) AS scripts_upd_timestamp,
    rx_upd_timestamp AS zz_upd_timestamp,
    CAST(NULL AS NUMERIC) AS cvd_amount,

  UNION ALL

  -- Branch 5: FS Discount
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_REFILL,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_RE,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_RE,
    CAST(NULL AS INT64) AS cal_fs_units,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_re,
    CAST(NULL AS NUMERIC) AS fs_salesamount,
    CAST(NULL AS NUMERIC) AS emp_reductionamount,
    CAST(NULL AS NUMERIC) AS rx_salesamount,
    REDUCTIONAMOUNT,
    CAST(NULL AS INT64) AS cal_rx_cnt_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_re,
    TRANSCURRENCY AS transcurrency,

  -- Branch 6: COVID Sales

    MANDT,
    RETAILSTOREID,

    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_CNT_GE84_RE,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_NS,
    CAST(NULL AS STRING) AS ZZ_RX_MCRX_GE84_RE,
    CAST(NULL AS INT64) AS cal_fs_units,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_ge84_re,
    CAST(NULL AS NUMERIC) AS fs_salesamount,
    CAST(NULL AS NUMERIC) AS emp_reductionamount,
    CAST(NULL AS NUMERIC) AS rx_salesamount,
    CAST(NULL AS NUMERIC) AS reductionamount,
    CAST(NULL AS INT64) AS cal_rx_cnt_ns,
    CAST(NULL AS INT64) AS cal_rx_cnt_re,
    TRANSCURRENCY AS transcurrency,
    CAST(NULL AS STRING) AS fs_upd_timestamp,
    CAST(NULL AS STRING) AS rx_upd_timestamp,
    CAST(NULL AS STRING) AS scripts_upd_timestamp,
    cvd_upd_timestamp AS zz_upd_timestamp,

  FROM Joined_COVID_Sales_With_RX_Retail_Types

-- =====================================================================
-- CTE 22: COMBINE_DATA
-- Logic: Aggregates the unioned six-branch dataset by the grain keys
--        MANDT, RETAILSTOREID, BUSINESSDAYDATE, TRANSCURRENCY, ZZ_UPD_TIMESTAMP.
--        Implements COMBINE_DATA aggregation node from CV_COMP_FLASH_SALES.
--        SUM for sales/discount/unit/count measures.
--        COUNT for raw ZZ_RX_* string fields (non-null count).
--        MAX for timestamp fields.
-- Source: Combined_Flash_Sales_Inputs
-- Aggregation: GROUP BY MANDT, RETAILSTOREID, BUSINESSDAYDATE, TRANSCURRENCY,
--              ZZ_UPD_TIMESTAMP; SUM/COUNT/MAX per measure
-- =====================================================================
COMBINE_DATA AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    transcurrency,
    zz_upd_timestamp,
    SUM(fs_salesamount)       AS fs_salesamount,
    SUM(reductionamount)      AS reductionamount,
    SUM(rx_salesamount)       AS rx_salesamount,
    COUNT(ZZ_RX_CNT_NS)       AS zz_rx_cnt_ns,
    COUNT(ZZ_RX_CNT_REFILL)   AS zz_rx_cnt_refill,
    COUNT(ZZ_RX_CNT_GE84_NS)  AS zz_rx_cnt_ge84_ns,
    COUNT(ZZ_RX_CNT_GE84_RE)  AS zz_rx_cnt_ge84_re,
    COUNT(ZZ_RX_MCRX_GE84_NS) AS zz_rx_mcrx_ge84_ns,
    COUNT(ZZ_RX_MCRX_GE84_RE) AS zz_rx_mcrx_ge84_re,
    SUM(cal_fs_units)          AS cal_fs_units,
    SUM(emp_reductionamount)   AS emp_reductionamount,
    SUM(cal_rx_cnt_ge84_ns)    AS cal_rx_cnt_ge84_ns,
    SUM(cal_rx_cnt_ge84_re)    AS cal_rx_cnt_ge84_re,
    SUM(cal_rx_cnt_ns)         AS cal_rx_cnt_ns,
    SUM(cal_rx_cnt_re)         AS cal_rx_cnt_re,
    MAX(fs_upd_timestamp)      AS fs_upd_timestamp,
    MAX(rx_upd_timestamp)      AS rx_upd_timestamp,
    MAX(scripts_upd_timestamp) AS scripts_upd_timestamp,
    SUM(cvd_amount)            AS cvd_amount,
    SUM(cal_cvd_units)         AS cal_cvd_units
  FROM Combined_Flash_Sales_Inputs
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    transcurrency,
    zz_upd_timestamp
),

-- =====================================================================
-- CTE 23: MERGED
-- Logic: Inner joins NAVIX (base retail-day grain) with COMBINE_DATA on
--        MANDT, RETAILSTOREID, BUSINESSDAYDATE to produce the final
--        pre-aggregation dataset. Implements MERGED node from CV_COMP_FLASH_SALES.
--        This is the output of the intermediate composite view
--        (CV_COMP_FLASH_SALES / xml_acc_cv_comp_flash_sales-VT-table-CV).
-- Source: NAVIX, COMBINE_DATA
-- Join: INNER JOIN NAVIX n ON COMBINE_DATA c ON n.MANDT=c.MANDT,
--       n.RETAILSTOREID=c.RETAILSTOREID, n.BUSINESSDAYDATE=c.BUSINESSDAYDATE
-- =====================================================================
MERGED AS (
  SELECT
    n.MANDT,
    n.RETAILSTOREID,
    n.BUSINESSDAYDATE,
    c.fs_salesamount,
    c.reductionamount,
    c.rx_salesamount,
    c.zz_rx_cnt_ns,
    c.zz_rx_cnt_refill,
    c.zz_rx_cnt_ge84_ns,
    c.zz_rx_cnt_ge84_re,
    c.zz_rx_mcrx_ge84_ns,
    c.zz_rx_mcrx_ge84_re,
    c.cal_fs_units,
    c.emp_reductionamount,
    c.cal_rx_cnt_ge84_ns,
    c.cal_rx_cnt_ge84_re,
    c.cal_rx_cnt_ns,
    c.cal_rx_cnt_re,
    c.transcurrency,
    c.rx_upd_timestamp,
    c.fs_upd_timestamp,
    c.scripts_upd_timestamp,
    c.zz_upd_timestamp,
    c.cvd_amount,
    c.cal_cvd_units
  FROM NAVIX n
  INNER JOIN COMBINE_DATA c
    ON n.MANDT = c.MANDT
   AND n.RETAILSTOREID = c.RETAILSTOREID
   AND n.BUSINESSDAYDATE = c.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 24: Filtered_FLASH_SALES_VT_CAR
-- Logic: Applies the final-consumer filter from CV_BASE_FIN_FLASH_SALES_CAR
--        (xml_acc_FLASH_SALES_VT_CAR). Filters the MERGED composite result
--        by ZZ_UPD_TIMESTAMP between the two runtime parameters
--        @ip_upd_timestamp_from and @ip_upd_timestamp_to (inclusive).
--        This replaces the Projection_1 node of the final consumer that
--        reads from FLASH_SALES_VT_CAR; since the composite view logic is
--        fully inlined, the filter is applied directly on MERGED output.
--        REQUIRES VALIDATION: The final consumer reads from CVS_FRIP.FLASH_SALES_VT_CAR
--        which is a physical reporting/staging table not present in Columns_Mapping_v3.
--        This consolidation inlines the composite view logic in place of that table.
-- Source: MERGED
-- Filters: zz_upd_timestamp >= @ip_upd_timestamp_from
--          AND zz_upd_timestamp <= @ip_upd_timestamp_to
-- =====================================================================
Filtered_FLASH_SALES_VT_CAR AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    fs_salesamount,
    reductionamount,
    rx_salesamount,
    cal_fs_units,
    emp_reductionamount,
    cal_rx_cnt_ge84_ns,
    cal_rx_cnt_ge84_re,
    cal_rx_cnt_ns,
    cal_rx_cnt_re,
    rx_upd_timestamp,
    fs_upd_timestamp,
    scripts_upd_timestamp,
    zz_upd_timestamp,
    transcurrency,
    cvd_amount,
    cal_cvd_units
  FROM MERGED
  WHERE zz_upd_timestamp >= @ip_upd_timestamp_from
    AND zz_upd_timestamp <= @ip_upd_timestamp_to
    -- REQUIRES VALIDATION: zz_upd_timestamp datatype vs parameter type alignment
)

-- =====================================================================
-- Final SELECT
-- Logic: Aggregates the filtered flash sales data by the final-consumer
--        logical model grain: MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--        ZZ_UPD_TIMESTAMP, TRANSCURRENCY.
--        Applies SUM aggregation to all sales/discount/unit/count measures
--        and MAX aggregation to all timestamp measures.
--        Implements the CUBE/Aggregation output of CV_BASE_FIN_FLASH_SALES_CAR
--        (xml_acc_FLASH_SALES_VT_CAR).
-- Source: Filtered_FLASH_SALES_VT_CAR
-- Aggregation: GROUP BY MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--              ZZ_UPD_TIMESTAMP, TRANSCURRENCY
-- =====================================================================
SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  zz_upd_timestamp,
  transcurrency,
  SUM(fs_salesamount)       AS fs_salesamount,
  SUM(reductionamount)      AS reductionamount,
  SUM(rx_salesamount)       AS rx_salesamount,
  SUM(cal_fs_units)         AS cal_fs_units,
  SUM(emp_reductionamount)  AS emp_reductionamount,
  SUM(cal_rx_cnt_ge84_ns)   AS cal_rx_cnt_ge84_ns,
  SUM(cal_rx_cnt_ge84_re)   AS cal_rx_cnt_ge84_re,
  SUM(cal_rx_cnt_ns)        AS cal_rx_cnt_ns,
  SUM(cal_rx_cnt_re)        AS cal_rx_cnt_re,
  MAX(rx_upd_timestamp)     AS rx_upd_timestamp,
  MAX(fs_upd_timestamp)     AS fs_upd_timestamp,
  MAX(scripts_upd_timestamp) AS scripts_upd_timestamp,
  SUM(cvd_amount)           AS cvd_amount,
  SUM(cal_cvd_units)        AS cal_cvd_units
FROM Filtered_FLASH_SALES_VT_CAR
GROUP BY
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  zz_upd_timestamp,
  transcurrency;
