-- =====================================================================
-- CTE 4: GL_Hierarchy
-- Logic: G/L account hierarchy nodes filtered for parent node 181
--        and active validity date, with leading zeros trimmed from nodevalue
-- Source: edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node

-- =====================================================================

  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue,
    hryid,
    LTRIM(nodevalue, '0') AS cal_node_value
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node`
  WHERE mandt IN ('120', '200')
    AND parnode LIKE '%181'
    AND hryvalto = DATE '9999-12-31'
    AND hryid = 'CVS2'
),

-- =====================================================================

-- =====================================================================

  FROM Weekly_Actuals_With_PC_Hierarchy wa

-- =====================================================================
Weekly_Actuals_With_Profit_Center_Text AS (
  SELECT
    wa.*,
    pct.profit_center_text,
    CASE
      WHEN LEFT(wa.bic_zwwpc_pa1, 2) = 'FS' THEN 'FS'
      WHEN LEFT(wa.bic_zwwpc_pa1, 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS cal_fs_rx_flag,
    CASE
      WHEN LEFT(wa.bic_zwwpc_pa1, 2) = 'FS' THEN wa.fs_comp_wk
      WHEN LEFT(wa.bic_zwwpc_pa1, 2) = 'RX' THEN wa.rx_comp_wk
      ELSE wa.fs_comp_wk
    END AS cal_comp_flag,
    RIGHT(@ip_week_cy, 2) AS cal_week_number
  FROM Weekly_Actuals_With_Comparison_Flags wa
  LEFT JOIN Profit_Center_Text pct
    ON wa.b631_s_proftctr = pct.prctr
    AND wa.mandt = pct.mandt
)
