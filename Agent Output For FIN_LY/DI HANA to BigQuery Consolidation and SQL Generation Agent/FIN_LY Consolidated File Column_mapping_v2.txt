-- =====================================================================
-- CONSOLIDATED BIGQUERY SQL FOR CV_COMP_FIN_ACTUAL_STATIC
-- Final Consumer: CV_COMP_FIN_ACTUAL_STATIC

-- Consolidation Date: 2024

-- Runtime Parameters (must be supplied at execution):
-- @ip_version: STRING - Version filter for weekly actuals
-- @ip_week_ending_from: STRING - Week ending from (not used in current logic)
-- @ip_week_ending_to: STRING - Week ending to (not used in current logic)
-- @ip_version_comp_flag: STRING - Comparison flag version
-- @ip_week_ly: STRING - Last year week filter
-- @ip_week_cy: STRING - Current year week filter
-- @v_week: STRING - Prior fiscal week (replaces SFN_PRIOR_FISCAL_WEEK function)

-- =====================================================================
-- CTE 1: WEEKLY_SNAPSHOT_DS05
-- Logic: Filter weekly financial actuals by fiscal variant, version, week, and audit type
-- Source: CV_BASE_FIN_WEEKLY_ACTUAL_S4 → AZSRP_DS072_VT_S4
-- Filters: FISCVARNT = 'K4', _BIC_ZIO_VER = @ip_version, _BIC_ZIO_SWEEK = @ip_week_ly, _BIC_ZIO_SAUDT IN ('1', '10'), MANDT IN ('110', '200')
-- =====================================================================
WITH WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    bic_zio_sweek AS _BIC_ZIO_SWEEK,
    b631_s_chrtacct AS _B631_S_CHRTACCT,
    b631_s_co_area AS _B631_S_CO_AREA,
    bic_zio_cmpcd AS _BIC_ZIO_CMPCD,
    b631_s_proftctr AS _B631_S_PROFTCTR,
    b631_s_costcntr AS _B631_S_COSTCNTR,
    b631_s_funcarea AS _B631_S_FUNCAREA,
    bic_zio_ver AS _BIC_ZIO_VER,
    bic_zio_saudt AS _BIC_ZIO_SAUDT,
    b631_s_gl_acct AS _B631_S_GL_ACCT,
    bic_zwwpc_pa1 AS _BIC_ZWWPC_PA1,
    bic_zwwsc_pa1 AS _BIC_ZWWSC_PA1,
    b631_s_amount AS _B631_S_AMOUNT,
    currency,
    bic_zio_d1amt AS _BIC_ZIO_D1AMT,
    bic_zio_d2amt AS _BIC_ZIO_D2AMT,
    bic_zio_d3amt AS _BIC_ZIO_D3AMT,
    bic_zio_d4amt AS _BIC_ZIO_D4AMT,
    bic_zio_d5amt AS _BIC_ZIO_D5AMT,
    bic_zio_d6amt AS _BIC_ZIO_D6AMT,
    bic_zio_d7amt AS _BIC_ZIO_D7AMT
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds212`
  WHERE
    mandt IN ('110', '200')
    AND fiscvarnt = 'K4'
    AND bic_zio_ver = @ip_version
    AND bic_zio_sweek = @ip_week_ly
    AND bic_zio_saudt IN ('1', '10')
),

-- =====================================================================
-- CTE 2: PC_HIER
-- Logic: Filter profit center hierarchy nodes for CORE_RET parent and valid-to date
-- Source: CV_BASE_MD_HRRP_NODE_S4 → HRRP_NODE
-- Filters: PARNODE LIKE '%CORE_RET%', HRYVALTO = '99991231', MANDT IN ('120', '200')
-- =====================================================================
PC_HIER AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node`
  WHERE
    mandt IN ('120', '200')
    AND parnode LIKE '%CORE_RET%'
    AND hryvalto = DATE '9999-12-31'
),

-- =====================================================================
-- CTE 3: Join_WeeklySnapshot_PC_Hier
-- Logic: Join weekly actuals with profit center hierarchy
-- Join: INNER JOIN on _B631_S_PROFTCTR = PC_HIER.NODEVALUE and MANDT
-- =====================================================================
Join_WeeklySnapshot_PC_Hier AS (
  SELECT
    ws.*
  FROM
    WEEKLY_SNAPSHOT_DS05 ws
    INNER JOIN PC_HIER ph
      ON ws._B631_S_PROFTCTR = ph.nodevalue
      AND ws.mandt = ph.mandt
),

-- =====================================================================
-- CTE 4: GL_HIER
-- Logic: Filter G/L account hierarchy nodes for parent 181, hierarchy ID CVS2, and valid-to date
-- Source: CV_BASE_MD_HRRP_NODE_S4 → HRRP_NODE
-- Filters: PARNODE LIKE '%181', HRYVALTO = '99991231', HRYID = 'CVS2', MANDT IN ('120', '200')
-- Calculation: CAL_NODE_VALUE = LTRIM(NODEVALUE, '0')
-- =====================================================================
GL_HIER AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue,
    hryid,
    LTRIM(nodevalue, '0') AS cal_node_value
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_hrrp_node`
  WHERE
    mandt IN ('120', '200')
    AND parnode LIKE '%181'
    AND hryvalto = DATE '9999-12-31'
    AND hryid = 'CVS2'
),

-- =====================================================================
-- CTE 5: Join_GL_Hier
-- Logic: Join with G/L account hierarchy
-- Join: INNER JOIN on _B631_S_GL_ACCT = GL_HIER.NODEVALUE and MANDT
-- =====================================================================
Join_GL_Hier AS (
  SELECT
    j.*
  FROM
    Join_WeeklySnapshot_PC_Hier j
    INNER JOIN GL_HIER gh
      ON j._B631_S_GL_ACCT = gh.nodevalue
      AND j.mandt = gh.mandt
),

-- =====================================================================
-- CTE 6: Projection_With_CalWeek
-- Logic: Add current year week parameter as calculated column
-- Calculation: CAL_WEEK = @ip_week_cy
-- =====================================================================
Projection_With_CalWeek AS (
  SELECT
    *,
    @ip_week_cy AS cal_week
  FROM
    Join_GL_Hier
),

-- =====================================================================
-- CTE 7: STORE_ATTR_ACTUAL
-- Logic: Inline STP_WSS_SRP_ATTRIBUTES procedure's INSERT SELECT for TBL_WSS_SRP_ATTR_ACT
-- Source: CV_BASE_MD_SRPACT_S4 → ZTSRP_ATTR_ACT
-- Filters: MANDT IN ('120', '200')
-- Aggregation: SUM(RX_HRS_OPER, FS_HRS_OPER, RETAIL_SQFT_AMT, TOTAL_SQFT_AMT), GROUP BY all other columns
-- Note: Audit columns (SNAPSHOT_TIMESTAMP, SNAPSHOT_CREATED_BY) omitted as not used downstream
-- =====================================================================
STORE_ATTR_ACTUAL AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_attr_act`
  WHERE
    mandt IN ('120', '200')
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat, fs_open_dat, fs_close_dat,
    rep_mkt_code, rep_mkt_desc, address, city, state, zipcode, division_code, division_desc,
    division_mgr, area_code, area_desc, area_mgr, region_code, region_desc, region_mgr,
    district_code, district_desc, zone_code, district_mgr, zone_desc, major_region_code,
    major_region, store_type, store_type_desc, rx_store_desc, distribution_cntr,
    distribution_cntr_desc, store_24_hr, budget_open_dat, budget_closed_dat, budget_reloc_dat,
    const_open_dat, const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc, original_fs_open_dat,
    original_rx_open_dat, group_one, group_two, ard_sort_key, area_cost_center,
    regiion_cost_center, county_code, county_desc, sqft_bracket_desc, sqft_sort_key,
    acquisition_code, acquisition_desc, photo_lab_add_dat, photo_lab_close_dat, min_hrs_ind,
    createdby, created_on, rx_division_code, rx_area_code, rx_region_code, rx_district_code
),

-- =====================================================================
-- CTE 8: Join_Projection_StoreAttr
-- Logic: Left join with store attributes
-- Join: LEFT JOIN on _B631_S_PROFTCTR = STORE_ATTR_ACTUAL.PRCTR and MANDT
-- =====================================================================
Join_Projection_StoreAttr AS (
  SELECT
    p.*,
    sa.rep_mkt_code,
    sa.rep_mkt_desc,
    sa.emerg_mkt_ind,
    sa.division_code,
    sa.division_desc,
    sa.area_code,
    sa.area_desc,
    sa.region_code,
    sa.region_desc,
    sa.district_code,
    sa.district_desc,
    sa.city,
    sa.state,
    sa.fs_open_dat,
    sa.rx_open_dat,
    sa.strnum,
    sa.rx_division_code,
    sa.rx_area_code,
    sa.rx_region_code,
    sa.rx_district_code
  FROM
    Projection_With_CalWeek p
    LEFT JOIN STORE_ATTR_ACTUAL sa
      ON p._B631_S_PROFTCTR = sa.prctr
      AND p.mandt = sa.mandt
),

-- =====================================================================
-- CTE 9: RETAIL_CAL
-- Logic: Filter retail calendar week data
-- Source: CV_BASE_MD_RCALWEEK_S4 → ZTFIGL_RCALWEEK
-- Filters: RCLNT IN ('120', '200')
-- =====================================================================
RETAIL_CAL AS (
  SELECT
    zrwstrtdate,
    zrwenddate,
    zzweek,
    rclnt
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek`
  WHERE
    rclnt IN ('120', '200')
),

-- =====================================================================
-- CTE 10: Join_RetailCal
-- Logic: Left join with retail calendar to get week start and end dates
-- Join: LEFT JOIN on CAL_WEEK = RETAIL_CAL.ZZWEEK and MANDT = RETAIL_CAL.RCLNT
-- =====================================================================
Join_RetailCal AS (
  SELECT
    j.*,
    rc.zrwstrtdate,
    rc.zrwenddate
  FROM
    Join_Projection_StoreAttr j
    LEFT JOIN RETAIL_CAL rc
      ON j.cal_week = rc.zzweek
      AND j.mandt = rc.rclnt
),

-- =====================================================================
-- CTE 11: COMP_FLAG_LY
-- Logic: Inline STP_WSS_SRP_ATTRIBUTES procedure's INSERT SELECT for TBL_WSS_SRP_COMPFLAG
-- Source: CV_BASE_MD_COMPFL_S4 → ZTSRP_COMPFL
-- Filters: MANDT IN ('110', '200'), ZWEEK = @ip_week_cy, COMP_VER = @ip_version_comp_flag
-- Note: Procedure filter ZWEEK = @v_week preserved; final consumer uses @ip_week_cy
-- Note: Audit columns (SNAPSHOT_TIMESTAMP, CREATED_BY) omitted as not used downstream
-- =====================================================================
COMP_FLAG_LY AS (
  SELECT
    mandt,
    prctr,
    strnum,
    fs_comp_wk,
    rx_comp_wk,
    zweek,
    zmonth,
    zyear,
    comp_ver
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztsrp_compfl`
  WHERE
    mandt IN ('110', '200')
    AND zweek = @ip_week_cy
    AND comp_ver = @ip_version_comp_flag
),

-- =====================================================================
-- CTE 12: Join_CompFlag
-- Logic: Left join with comparison flag data
-- Join: LEFT JOIN on _B631_S_PROFTCTR = COMP_FLAG_LY.PRCTR and MANDT
-- =====================================================================
Join_CompFlag AS (
  SELECT
    wn.*,
    cfl.fs_comp_wk,
    cfl.rx_comp_wk
  FROM
    Join_RetailCal wn
    LEFT JOIN COMP_FLAG_LY cfl
      ON wn._B631_S_PROFTCTR = cfl.prctr
      AND wn.mandt = cfl.mandt
),

-- =====================================================================
-- CTE 13: PROFIT_CENTER_TEXT
-- Logic: Filter profit center text descriptions
-- Source: CV_BASE_MD_CEPCT_S4 → CEPCT
-- Filters: MANDT IN ('120', '200')
-- =====================================================================
PROFIT_CENTER_TEXT AS (
  SELECT
    mandt,
    prctr,
    ltext AS profit_center_text
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE
    mandt IN ('120', '200')
),

-- =====================================================================
-- CTE 14: Join_ProfitCenterText
-- Logic: Left join with profit center text
-- Join: LEFT JOIN on _B631_S_PROFTCTR = PROFIT_CENTER_TEXT.PRCTR and MANDT
-- =====================================================================
Join_ProfitCenterText AS (
  SELECT
    jcf.*,
    pct.profit_center_text
  FROM
    Join_CompFlag jcf
    LEFT JOIN PROFIT_CENTER_TEXT pct
      ON jcf._B631_S_PROFTCTR = pct.prctr
      AND jcf.mandt = pct.mandt
),

-- =====================================================================
-- CTE 15: FLAGS
-- Logic: Calculate FS/RX flag, comparison flag, and week number
-- Calculation: CAL_FS_RX_FLAG = CASE WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'FS' THEN 'FS' WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'RX' THEN 'RX' ELSE '' END
-- Calculation: CAL_COMP_FLAG = CASE WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'FS' THEN FS_COMP_WK WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'RX' THEN RX_COMP_WK ELSE FS_COMP_WK END
-- Calculation: CAL_WEEK_NUMBER = RIGHT(@ip_week_cy, 2)
-- =====================================================================
FLAGS AS (
  SELECT
    *,
    CASE
      WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'FS' THEN 'FS'
      WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS cal_fs_rx_flag,
    CASE
      WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'FS' THEN fs_comp_wk
      WHEN LEFT(_BIC_ZWWPC_PA1, 2) = 'RX' THEN rx_comp_wk
      ELSE fs_comp_wk
    END AS cal_comp_flag,
    RIGHT(@ip_week_cy, 2) AS cal_week_number
  FROM
    Join_ProfitCenterText
)

-- =====================================================================
-- FINAL OUTPUT
-- Logic: Aggregate measures and group by all attributes
-- Aggregation: SUM(_B631_S_AMOUNT) * -1, SUM(_BIC_ZIO_D1AMT through _BIC_ZIO_D7AMT)
-- =====================================================================
SELECT
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  _BIC_ZIO_SWEEK AS bic_zio_sweek,
  _B631_S_CHRTACCT AS b631_s_chrtacct,
  _B631_S_CO_AREA AS b631_s_co_area,
  _BIC_ZIO_CMPCD AS bic_zio_cmpcd,
  _B631_S_PROFTCTR AS b631_s_proftctr,
  cal_fs_rx_flag AS fs_rx_flag,
  _B631_S_COSTCNTR AS b631_s_costcntr,
  _B631_S_FUNCAREA AS b631_s_funcarea,
  _BIC_ZIO_VER AS bic_zio_ver,
  _BIC_ZIO_SAUDT AS bic_zio_saudt,
  _B631_S_GL_ACCT AS b631_s_gl_acct,
  _BIC_ZWWPC_PA1 AS bic_zwwpc_pa1,
  _BIC_ZWWSC_PA1 AS bic_zwwsc_pa1,
  SUM(_B631_S_AMOUNT) * -1 AS b631_s_amount,
  SUM(_BIC_ZIO_D1AMT) AS bic_zio_d1amt,
  SUM(_BIC_ZIO_D2AMT) AS bic_zio_d2amt,
  SUM(_BIC_ZIO_D3AMT) AS bic_zio_d3amt,
  SUM(_BIC_ZIO_D4AMT) AS bic_zio_d4amt,
  SUM(_BIC_ZIO_D5AMT) AS bic_zio_d5amt,
  SUM(_BIC_ZIO_D6AMT) AS bic_zio_d6amt,
  SUM(_BIC_ZIO_D7AMT) AS bic_zio_d7amt,
  currency,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  fs_comp_wk,
  rx_comp_wk,
  cal_comp_flag AS comp_flag,
  division_code,
  division_desc,
  area_code,
  area_desc,
  region_code,
  region_desc,
  district_code,
  district_desc,
  city,
  state,
  zrwstrtdate,
  zrwenddate,
  cal_week_number,
  profit_center_text,
  fs_open_dat,
  rx_open_dat,
  cal_week,
  strnum,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code
FROM
  FLAGS
GROUP BY
  mandt, fiscper, fiscvarnt, fiscyear, fiscper3, _BIC_ZIO_SWEEK, _B631_S_CHRTACCT,
  _B631_S_CO_AREA, _BIC_ZIO_CMPCD, _B631_S_PROFTCTR, cal_fs_rx_flag, _B631_S_COSTCNTR,
  _B631_S_FUNCAREA, _BIC_ZIO_VER, _BIC_ZIO_SAUDT, _B631_S_GL_ACCT, _BIC_ZWWPC_PA1,
  _BIC_ZWWSC_PA1, currency, rep_mkt_code, rep_mkt_desc, emerg_mkt_ind, fs_comp_wk,
  rx_comp_wk, cal_comp_flag, division_code, division_desc, area_code, area_desc,
  region_code, region_desc, district_code, district_desc, city, state, zrwstrtdate,
  zrwenddate, cal_week_number, profit_center_text, fs_open_dat, rx_open_dat, cal_week,
  strnum, rx_division_code, rx_area_code, rx_region_code, rx_district_code
