DECLARE V_WEEK STRING DEFAULT @IP_WEEK_LY;
DECLARE IP_VERSION STRING DEFAULT @IP_VERSION;
DECLARE IP_VERSION_COMP_FLAG STRING DEFAULT @IP_VERSION_COMP_FLAG;
DECLARE IP_WEEK_ENDING_FROM STRING DEFAULT @IP_WEEK_ENDING_FROM;
DECLARE IP_WEEK_ENDING_TO STRING DEFAULT @IP_WEEK_ENDING_TO;

DECLARE IP_WEEK_CY STRING DEFAULT @IP_WEEK_CY;

-- =====================================================================
-- CTE 1: WEEKLY_SNAPSHOT_DS05
-- Logic: Filters weekly actuals from base SKF data for specific fiscal variant, week, version, audit trail, and account
-- Source: CV_BASE_SKF_WEEKLY_ACTUAL_S4 → bic_azsrp_ds112
-- Filters: FISCVARNT='K4', _BIC_ZIO_SWEEK=@IP_WEEK_LY, _BIC_ZIO_VER=@IP_VERSION, _BIC_ZIO_SAUDT IN ('1','10'), _BIC_ZIO_PSKF IN ('804013','801000'), MANDT IN ('110','200')
-- =====================================================================
WITH WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    SUM(bic_zio_skfq) AS bic_zio_skfq,
    SUM(bic_zio_d1skq) AS bic_zio_d1skq,
    SUM(bic_zio_d2skq) AS bic_zio_d2skq,
    SUM(bic_zio_d3skq) AS bic_zio_d3skq,
    SUM(bic_zio_d4skq) AS bic_zio_d4skq,
    SUM(bic_zio_d5skq) AS bic_zio_d5skq,
    SUM(bic_zio_d6skq) AS bic_zio_d6skq,
    SUM(bic_zio_d7skq) AS bic_zio_d7skq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds112`
  WHERE mandt IN ('110', '200')
    AND fiscvarnt = 'K4'
    AND bic_zio_sweek = IP_WEEK_LY
    AND bic_zio_ver = IP_VERSION
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
  GROUP BY
    fiscper, fiscvarnt, fiscyear, fiscper3, mandt, recordmode,
    bic_zio_sweek, b631_s_co_area, bic_zio_cmpcd, b631_s_proftctr,
    b631_s_costcntr, b631_s_funcarea, bic_zio_ver, bic_zio_saudt,
    bic_zio_psty, bic_zio_pskf, unit
),

-- =====================================================================
-- CTE 2: CC_HIER
-- Logic: Filters cost center hierarchy nodes with specific parent node pattern and validity
-- Source: CV_BASE_MD_HRRP_NODE_S4 → hrrp_node
-- Filters: PARNODE LIKE '%CORERET%', HRYVALTO='99991231', MANDT IN ('120','200')
-- =====================================================================
CC_HIER AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE mandt IN ('120', '200')
    AND parnode LIKE '%CORERET%'
    AND hryvalto = '99991231'
),

-- =====================================================================
-- CTE 3: Join_3
-- Logic: Inner join of weekly snapshot with cost center hierarchy
-- Join: INNER JOIN on _B631_S_COSTCNTR=NODEVALUE and MANDT
-- =====================================================================
Join_3 AS (
  SELECT
    ws.fiscper,
    ws.fiscvarnt,
    ws.fiscyear,
    ws.fiscper3,
    ws.mandt,
    ws.recordmode,
    ws.bic_zio_sweek,
    ws.b631_s_co_area,
    ws.bic_zio_cmpcd,
    ws.b631_s_proftctr,
    ws.b631_s_costcntr,
    ws.b631_s_funcarea,
    ws.bic_zio_ver,
    ws.bic_zio_saudt,
    ws.bic_zio_psty,
    ws.bic_zio_pskf,
    ws.unit,
    ws.bic_zio_skfq,
    ws.bic_zio_d1skq,
    ws.bic_zio_d2skq,
    ws.bic_zio_d3skq,
    ws.bic_zio_d4skq,
    ws.bic_zio_d5skq,
    ws.bic_zio_d6skq,
    ws.bic_zio_d7skq
  FROM WEEKLY_SNAPSHOT_DS05 ws
  INNER JOIN CC_HIER ch
    ON ws.b631_s_costcntr = ch.nodevalue
    AND ws.mandt = ch.mandt
),

-- =====================================================================
-- CTE 4: Projection_1
-- Logic: Adds current week constant to the joined data
-- Calculation: CAL_WEEK = @IP_WEEK_CY
-- =====================================================================
Projection_1 AS (
  SELECT
    j3.*,
    IP_WEEK_CY AS cal_week
  FROM Join_3 j3
),

-- =====================================================================

    p1.bic_zio_d5skq,
    p1.bic_zio_d6skq,

    b631_s_funcarea,

  zrwstrtdate,

  fs_open_dat,

  rx_division_code,
