-- =====================================================================
-- CTE 1: WEEKLY_SNAPSHOT_DS05
-- Logic: Base weekly actual data filtered by fiscal variant, week, version, audit trail, and account type
-- Source: CV_BASE_SKF_WEEKLY_ACTUAL_S4 → AZSRP_DS112_VT_S4
-- Filters: FISCVARNT = 'K4', _BIC_ZIO_SWEEK = @IP_WEEK_LY, _BIC_ZIO_VER = @IP_VERSION, _BIC_ZIO_SAUDT IN ('1','10'), _BIC_ZIO_PSKF IN ('804013','801000')
-- =====================================================================
WITH WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    recordmode AS RECORDMODE,
    bic_zio_sweek AS _BIC_ZIO_SWEEK,
    b631_s_co_area AS _B631_S_CO_AREA,
    bic_zio_cmpcd AS _BIC_ZIO_CMPCD,
    b631_s_proftctr AS _B631_S_PROFTCTR,
    b631_s_costcntr AS _B631_S_COSTCNTR,
    b631_s_funcarea AS _B631_S_FUNCAREA,
    bic_zio_ver AS _BIC_ZIO_VER,
    bic_zio_saudt AS _BIC_ZIO_SAUDT,
    bic_zio_psty AS _BIC_ZIO_PSTY,
    bic_zio_pskf AS _BIC_ZIO_PSKF,
    unit AS UNIT,
    SUM(bic_zio_skfq) AS _BIC_ZIO_SKFQ,
    SUM(bic_zio_d1skq) AS _BIC_ZIO_D1SKQ,
    SUM(bic_zio_d2skq) AS _BIC_ZIO_D2SKQ,
    SUM(bic_zio_d3skq) AS _BIC_ZIO_D3SKQ,
    SUM(bic_zio_d4skq) AS _BIC_ZIO_D4SKQ,
    SUM(bic_zio_d5skq) AS _BIC_ZIO_D5SKQ,
    SUM(bic_zio_d6skq) AS _BIC_ZIO_D6SKQ,
    SUM(bic_zio_d7skq) AS _BIC_ZIO_D7SKQ
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds112`
  WHERE
    CAST(mandt AS STRING) IN ('110', '200')
    AND FISCVARNT = 'K4'
    AND bic_zio_sweek = @IP_WEEK_LY
    AND bic_zio_ver = @IP_VERSION
    AND bic_zio_saudt IN ('1','10')
    AND bic_zio_pskf IN ('804013','801000')
  GROUP BY
    FISCPER, FISCVARNT, FISCYEAR, FISCPER3, MANDT, RECORDMODE,
    _BIC_ZIO_SWEEK, _B631_S_CO_AREA, _BIC_ZIO_CMPCD, _B631_S_PROFTCTR,
    _B631_S_COSTCNTR, _B631_S_FUNCAREA, _BIC_ZIO_VER, _BIC_ZIO_SAUDT,
    _BIC_ZIO_PSTY, _BIC_ZIO_PSKF, UNIT
),

-- =====================================================================
-- CTE 2: CC_HIER
-- Logic: Cost center hierarchy nodes filtered by parent node and validity
-- Source: CV_BASE_MD_HRRP_NODE_S4 → HRRP_NODE
-- Filters: PARNODE LIKE '%CORERET%', HRYVALTO = '99991231', MANDT IN (120, 200)
-- =====================================================================
CC_HIER AS (
  SELECT
    mandt AS MANDT,
    parnode AS PARNODE,
    hryvalto AS HRYVALTO,
    nodevalue AS NODEVALUE
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE
    CAST(mandt AS STRING) IN ('120', '200')
    AND parnode LIKE '%CORERET%'
    AND hryvalto = DATE '9999-12-31'
),

-- =====================================================================
-- CTE 3: Join_3
-- Logic: Inner join of weekly snapshot with cost center hierarchy
-- Join: INNER JOIN on _B631_S_COSTCNTR = NODEVALUE and MANDT
-- =====================================================================
Join_3 AS (
  SELECT
    ws.FISCPER,
    ws.FISCVARNT,
    ws.FISCYEAR,
    ws.FISCPER3,
    ws.MANDT,
    ws.RECORDMODE,
    ws._BIC_ZIO_SWEEK,
    ws._B631_S_CO_AREA,
    ws._BIC_ZIO_CMPCD,
    ws._B631_S_PROFTCTR,
    ws._B631_S_COSTCNTR,
    ws._B631_S_FUNCAREA,
    ws._BIC_ZIO_VER,
    ws._BIC_ZIO_SAUDT,
    ws._BIC_ZIO_PSTY,
    ws._BIC_ZIO_PSKF,
    ws.UNIT,
    ws._BIC_ZIO_SKFQ,
    ws._BIC_ZIO_D1SKQ,
    ws._BIC_ZIO_D2SKQ,
    ws._BIC_ZIO_D3SKQ,
    ws._BIC_ZIO_D4SKQ,
    ws._BIC_ZIO_D5SKQ,
    ws._BIC_ZIO_D6SKQ,
    ws._BIC_ZIO_D7SKQ
  FROM WEEKLY_SNAPSHOT_DS05 ws
  INNER JOIN CC_HIER ch
    ON ws._B631_S_COSTCNTR = ch.NODEVALUE
    AND ws.MANDT = ch.MANDT
),

-- =====================================================================
-- CTE 4: Projection_1
-- Logic: Add calculated week column (current week parameter)
-- Calculation: CAL_WEEK = @IP_WEEK_CY (constant parameter)
-- =====================================================================
Projection_1 AS (
  SELECT
    j3.*,
    @IP_WEEK_CY AS CAL_WEEK
  FROM Join_3 j3
),

-- =====================================================================
-- CTE 5: STORE_ATTR_ACTUAL
-- Logic: Store attributes aggregated from base master data
-- Source: CV_BASE_MD_SRPACT_S4 → ZTSRP_ATTR_ACT
-- Filters: MANDT IN (120, 200)
-- Aggregation: SUM on measure columns, GROUP BY all attributes
-- =====================================================================
STORE_ATTR_ACTUAL AS (
  SELECT
    CAST(mandt AS STRING) AS MANDT,
    strnum AS STRNUM,
    prctr AS PRCTR,
    rep_mkt_code AS REP_MKT_CODE,
    rep_mkt_desc AS REP_MKT_DESC,
    emerg_mkt_ind AS EMERG_MKT_IND,
    division_code AS DIVISION_CODE,
    division_desc AS DIVISION_DESC,
    area_code AS AREA_CODE,
    area_desc AS AREA_DESC,
    region_code AS REGION_CODE,
    region_desc AS REGION_DESC,
    district_code AS DISTRICT_CODE,
    district_desc AS DISTRICT_DESC,
    city AS CITY,
    state AS STATE,
    fs_open_dat AS FS_OPEN_DAT,
    rx_open_dat AS RX_OPEN_DAT,
    rx_division_code AS RX_DIVISION_CODE,
    rx_area_code AS RX_AREA_CODE,
    rx_region_code AS RX_REGION_CODE,
    rx_district_code AS RX_DISTRICT_CODE
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE
    CAST(mandt AS STRING) IN ('120', '200')
  GROUP BY
    MANDT, STRNUM, PRCTR, REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND,
    DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC, REGION_CODE,
    REGION_DESC, DISTRICT_CODE, DISTRICT_DESC, CITY, STATE,
    FS_OPEN_DAT, RX_OPEN_DAT, RX_DIVISION_CODE, RX_AREA_CODE,
    RX_REGION_CODE, RX_DISTRICT_CODE
),

-- =====================================================================
-- CTE 6: Join_2
-- Logic: Left join with store attributes
-- Join: LEFT OUTER JOIN on _B631_S_PROFTCTR = PRCTR and MANDT
-- =====================================================================
Join_2 AS (
  SELECT
    p1.FISCPER,
    p1.FISCVARNT,
    p1.FISCYEAR,
    p1.FISCPER3,
    p1.MANDT,
    p1.RECORDMODE,
    p1._BIC_ZIO_SWEEK,
    p1._B631_S_CO_AREA,
    p1._BIC_ZIO_CMPCD,
    p1._B631_S_PROFTCTR,
    p1._B631_S_COSTCNTR,
    p1._B631_S_FUNCAREA,
    p1._BIC_ZIO_VER,
    p1._BIC_ZIO_SAUDT,
    p1._BIC_ZIO_PSTY,
    p1._BIC_ZIO_PSKF,
    p1.UNIT,
    p1._BIC_ZIO_SKFQ,
    p1._BIC_ZIO_D1SKQ,
    p1._BIC_ZIO_D2SKQ,
    p1._BIC_ZIO_D3SKQ,
    p1._BIC_ZIO_D4SKQ,
    p1._BIC_ZIO_D5SKQ,
    p1._BIC_ZIO_D6SKQ,
    p1._BIC_ZIO_D7SKQ,
    p1.CAL_WEEK,
    sa.REP_MKT_CODE,
    sa.REP_MKT_DESC,
    sa.EMERG_MKT_IND,
    sa.DIVISION_CODE,
    sa.DIVISION_DESC,
    sa.AREA_CODE,
    sa.AREA_DESC,
    sa.REGION_CODE,
    sa.REGION_DESC,
    sa.DISTRICT_CODE,
    sa.DISTRICT_DESC,
    sa.STRNUM,
    sa.CITY,
    sa.STATE,
    sa.FS_OPEN_DAT,
    sa.RX_OPEN_DAT,
    sa.RX_DIVISION_CODE,
    sa.RX_AREA_CODE,
    sa.RX_REGION_CODE,
    sa.RX_DISTRICT_CODE
  FROM Projection_1 p1
  LEFT JOIN STORE_ATTR_ACTUAL sa
    ON p1._B631_S_PROFTCTR = sa.PRCTR
    AND p1.MANDT = sa.MANDT
),

-- =====================================================================
-- CTE 7: RETAIL_CAL
-- Logic: Retail calendar week master data
-- Source: CV_BASE_MD_RCALWEEK_S4 → ZTFIGL_RCALWEEK
-- Filters: RCLNT IN (120, 200)
-- =====================================================================
RETAIL_CAL AS (
  SELECT
    rclnt AS RCLNT,
    zzweek AS ZZWEEK,
    zrwstrtdate AS ZRWSTRTDATE,
    zrwenddate AS ZRWENDDATE
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE
    CAST(rclnt AS STRING) IN ('120', '200')
),

-- =====================================================================
-- CTE 8: Join_1
-- Logic: Left join with retail calendar
-- Join: LEFT OUTER JOIN on CAL_WEEK = ZZWEEK
-- =====================================================================
Join_1 AS (
  SELECT
    j2.*,
    rc.ZRWSTRTDATE,
    rc.ZRWENDDATE
  FROM Join_2 j2
  LEFT JOIN RETAIL_CAL rc
    ON j2.CAL_WEEK = rc.ZZWEEK
),

-- =====================================================================
-- CTE 9: COMP_FLAG_ACT01
-- Logic: Comp flag data filtered by week range and version
-- Source: CV_BASE_MD_COMPFL_S4 → ZTSRP_COMPFL
-- Filters: MANDT IN (110, 200), ZWEEK >= @IP_WEEK_ENDING_FROM, ZWEEK <= @IP_WEEK_ENDING_TO, COMP_VER = @IP_VERSION_COMP_FLAG
-- =====================================================================
COMP_FLAG_ACT01 AS (
  SELECT
    CAST(mandt AS STRING) AS MANDT,
    prctr AS PRCTR,
    zweek AS ZWEEK,
    zyear AS ZYEAR,
    comp_ver AS COMP_VER,
    rx_comp_wk AS RX_COMP_WK
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE
    CAST(mandt AS STRING) IN ('110', '200')
    AND zweek >= @IP_WEEK_ENDING_FROM
    AND zweek <= @IP_WEEK_ENDING_TO
    AND comp_ver = @IP_VERSION_COMP_FLAG
),

-- =====================================================================
-- CTE 10: Join_4
-- Logic: Left join with comp flag data
-- Join: LEFT OUTER JOIN on _B631_S_PROFTCTR = PRCTR and MANDT
-- =====================================================================
Join_4 AS (
  SELECT
    j1.*,
    cfa.RX_COMP_WK
  FROM Join_1 j1
  LEFT JOIN COMP_FLAG_ACT01 cfa
    ON j1._B631_S_PROFTCTR = cfa.PRCTR
    AND j1.MANDT = cfa.MANDT
),

-- =====================================================================
-- CTE 11: PROFIT_CENTER_TEXT
-- Logic: Profit center text descriptions
-- Source: CV_BASE_MD_CEPCT_S4 → CEPCT
-- Filters: MANDT IN (120, 200)
-- =====================================================================
PROFIT_CENTER_TEXT AS (
  SELECT
    CAST(mandt AS STRING) AS MANDT,
    prctr AS PRCTR,
    ltext AS PROFIT_CENTER_TEXT
  FROM
    `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE
    CAST(mandt AS STRING) IN ('120', '200')
),

-- =====================================================================
-- CTE 12: Join_5
-- Logic: Left join with profit center text
-- Join: LEFT OUTER JOIN on _B631_S_PROFTCTR = PRCTR and MANDT
-- =====================================================================
Join_5 AS (
  SELECT
    j4.*,
    pct.PROFIT_CENTER_TEXT
  FROM Join_4 j4
  LEFT JOIN PROFIT_CENTER_TEXT pct
    ON j4._B631_S_PROFTCTR = pct.PRCTR
    AND j4.MANDT = pct.MANDT
),

-- =====================================================================
-- CTE 13: FLAGS
-- Logic: Final projection with calculated columns
-- Calculation: CAL_WEEK_NUMBER = RIGHT(CAL_WEEK, 2), CAL_ZIO_SKFQ_QTY = CASE WHEN _BIC_ZIO_PSKF = '804013' THEN _BIC_ZIO_SKFQ * 2 ELSE _BIC_ZIO_SKFQ END
-- =====================================================================
FLAGS AS (
  SELECT
    MANDT,
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    RECORDMODE,
    _BIC_ZIO_SWEEK,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    UNIT,
    _BIC_ZIO_SKFQ,
    _BIC_ZIO_D1SKQ,
    _BIC_ZIO_D2SKQ,
    _BIC_ZIO_D3SKQ,
    _BIC_ZIO_D4SKQ,
    _BIC_ZIO_D5SKQ,
    _BIC_ZIO_D6SKQ,
    _BIC_ZIO_D7SKQ,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,
    DIVISION_DESC,
    AREA_CODE,
    AREA_DESC,
    REGION_CODE,
    REGION_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    STRNUM,
    CITY,
    STATE,
    CAL_WEEK,
    ZRWSTRTDATE,
    ZRWENDDATE,
    RX_COMP_WK,
    PROFIT_CENTER_TEXT,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    RIGHT(CAL_WEEK, 2) AS CAL_WEEK_NUMBER,
    CASE
      WHEN _BIC_ZIO_PSKF = '804013' THEN CAST(_BIC_ZIO_SKFQ AS NUMERIC) * 2
      ELSE CAST(_BIC_ZIO_SKFQ AS NUMERIC)
    END AS CAL_ZIO_SKFQ_QTY
  FROM Join_5
)

-- =====================================================================
-- Final Output: Aggregated result with all measures and attributes
-- Aggregation: SUM on all measure columns, GROUP BY all attributes
-- =====================================================================
SELECT
  MANDT,
  FISCPER,
  FISCVARNT,
  FISCYEAR,
  FISCPER3,
  RECORDMODE,
  _BIC_ZIO_SWEEK,
  _B631_S_CO_AREA,
  _BIC_ZIO_CMPCD,
  _B631_S_PROFTCTR,
  _B631_S_COSTCNTR,
  _B631_S_FUNCAREA,
  _BIC_ZIO_VER,
  _BIC_ZIO_SAUDT,
  REP_MKT_CODE,
  REP_MKT_DESC,
  EMERG_MKT_IND,
  _BIC_ZIO_PSTY,
  _BIC_ZIO_PSKF,
  UNIT,
  RX_COMP_WK,
  DIVISION_CODE,
  DIVISION_DESC,
  AREA_CODE,
  AREA_DESC,
  REGION_CODE,
  REGION_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  STRNUM,
  CITY,
  STATE,
  CAL_WEEK_NUMBER,
  ZRWSTRTDATE,
  ZRWENDDATE,
  PROFIT_CENTER_TEXT,
  FS_OPEN_DAT,
  RX_OPEN_DAT,
  CAL_WEEK,
  RX_DIVISION_CODE,
  RX_AREA_CODE,
  RX_REGION_CODE,
  RX_DISTRICT_CODE,
  SUM(_BIC_ZIO_SKFQ) AS _BIC_ZIO_SKFQ_90as1,
  SUM(_BIC_ZIO_D1SKQ) AS _BIC_ZIO_D1SKQ,
  SUM(_BIC_ZIO_D2SKQ) AS _BIC_ZIO_D2SKQ,
  SUM(_BIC_ZIO_D3SKQ) AS _BIC_ZIO_D3SKQ,
  SUM(_BIC_ZIO_D4SKQ) AS _BIC_ZIO_D4SKQ,
  SUM(_BIC_ZIO_D5SKQ) AS _BIC_ZIO_D5SKQ,
  SUM(_BIC_ZIO_D6SKQ) AS _BIC_ZIO_D6SKQ,
  SUM(_BIC_ZIO_D7SKQ) AS _BIC_ZIO_D7SKQ,
  SUM(CAL_ZIO_SKFQ_QTY) AS _BIC_ZIO_SKFQ
FROM FLAGS
GROUP BY
  MANDT, FISCPER, FISCVARNT, FISCYEAR, FISCPER3, RECORDMODE,
  _BIC_ZIO_SWEEK, _B631_S_CO_AREA, _BIC_ZIO_CMPCD, _B631_S_PROFTCTR,
  _B631_S_COSTCNTR, _B631_S_FUNCAREA, _BIC_ZIO_VER, _BIC_ZIO_SAUDT,
  REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND, _BIC_ZIO_PSTY, _BIC_ZIO_PSKF,
  UNIT, RX_COMP_WK, DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC,
  REGION_CODE, REGION_DESC, DISTRICT_CODE, DISTRICT_DESC, STRNUM, CITY,
  STATE, CAL_WEEK_NUMBER, ZRWSTRTDATE, ZRWENDDATE, PROFIT_CENTER_TEXT,
  FS_OPEN_DAT, RX_OPEN_DAT, CAL_WEEK, RX_DIVISION_CODE, RX_AREA_CODE,
  RX_REGION_CODE, RX_DISTRICT_CODE
