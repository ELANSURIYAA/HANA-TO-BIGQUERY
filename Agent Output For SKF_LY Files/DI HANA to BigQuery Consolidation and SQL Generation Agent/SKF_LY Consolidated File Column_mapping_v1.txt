/*==============================================================================
 * CTE: Final_Projection_With_Calculated_Columns
 * Logic: Add calculated columns CAL_WEEK_NUMBER and CAL_ZIO_SKFQ_QTY
 * Calculated Columns:
 *   - CAL_WEEK_NUMBER: RIGHT(CAL_WEEK, 2)
 *   - CAL_ZIO_SKFQ_QTY: CASE WHEN _BIC_ZIO_PSKF = '804013' 
 *                            THEN CAST(_BIC_ZIO_SKFQ AS NUMERIC) * 2
 *                            ELSE CAST(_BIC_ZIO_SKFQ AS NUMERIC) END
 *==============================================================================*/
Final_Projection_With_Calculated_Columns AS (
  SELECT
    MANDT,
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    RECORDMODE,

    _BIC_ZIO_D3SKQ,
    _BIC_ZIO_D4SKQ,
    _BIC_ZIO_D5SKQ,
    _BIC_ZIO_D6SKQ,
    _BIC_ZIO_D7SKQ,
    CASE
      WHEN _BIC_ZIO_PSKF = '804013' THEN CAST(_BIC_ZIO_SKFQ AS NUMERIC) * 2
      ELSE CAST(_BIC_ZIO_SKFQ AS NUMERIC)
    END AS CAL_ZIO_SKFQ_QTY
  FROM
    Joined_With_Profit_Center_Text
)

/*==============================================================================

 *   - _BIC_ZIO_SKFQ_90as1: SUM(_BIC_ZIO_SKFQ)
