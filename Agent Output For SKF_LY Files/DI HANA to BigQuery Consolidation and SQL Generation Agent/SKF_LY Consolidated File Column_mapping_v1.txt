    bic_zio_ver AS _BIC_ZIO_VER,                 -- Column mapping applied
    bic_zio_saudt AS _BIC_ZIO_SAUDT,             -- Column mapping applied
    bic_zio_psty AS _BIC_ZIO_PSTY,               -- Column mapping applied
    bic_zio_pskf AS _BIC_ZIO_PSKF,               -- Column mapping applied

    bic_zio_skfq AS _BIC_ZIO_SKFQ,               -- Column mapping applied

  WHERE MANDT IN ('110', '200')                   -- MANDT filter from CV definition
    AND FISCVARNT = 'K4'                          -- Fiscal variant filter
    AND bic_zio_sweek = @IP_WEEK_LY               -- Week filter (parameter)
    AND bic_zio_ver = @IP_VERSION                 -- Version filter (parameter)
    AND bic_zio_saudt IN ('1','10')               -- Audit trail filter
    AND bic_zio_pskf IN ('804013','801000')       -- Account filter

-- =============================================================================
-- CTE 2: CC_HIER
-- Source: CV_BASE_MD_HRRP_NODE_S4 (Calculation View)
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node
-- Logic: Hierarchy nodes filtered by parent node pattern and validity date
-- Traceability: CV_BASE_MD_HRRP_NODE_S4_DI HANA to BigQuery SQL Conversion Agent.txt
-- =============================================================================
CC_HIER AS (
  SELECT
    mandt AS MANDT,                               -- Column mapping applied
    parnode AS PARNODE,                           -- Column mapping applied
    hryvalto AS HRYVALTO,                         -- Column mapping applied
    nodevalue AS NODEVALUE                        -- Column mapping applied
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE mandt IN ('120', '200')                   -- MANDT filter from CV definition
    AND parnode LIKE '%CORERET%'                  -- Parent node filter
    AND hryvalto = '99991231'                     -- Validity date filter
),

-- =============================================================================
-- CTE 3: Join_3
-- Logic: INNER JOIN of weekly actuals with hierarchy nodes on cost center
-- Traceability: CV_COMP_SKF_ACTUAL_STATIC_DI HANA to BigQuery SQL Conversion Agent.txt
-- =============================================================================
Join_3 AS (
  SELECT
    ws.*,
    ch.NODEVALUE AS _B631_S_COSTCNTR
  FROM WEEKLY_SNAPSHOT_DS05 ws
  INNER JOIN CC_HIER ch
    ON ws._B631_S_COSTCNTR = ch.NODEVALUE
    AND ws.MANDT = ch.MANDT
),

-- =============================================================================
-- CTE 4: Projection_1
-- Logic: Add calculated week column (parameter value)
-- Traceability: CV_COMP_SKF_ACTUAL_STATIC_DI HANA to BigQuery SQL Conversion Agent.txt
-- =============================================================================
Projection_1 AS (
  SELECT
    j3.*,
    @IP_WEEK_CY AS CAL_WEEK
  FROM Join_3 j3
),

-- =============================================================================

    AND zweek <= @IP_WEEK_ENDING_TO               -- Week range filter (parameter)

-- =============================================================================

  SELECT
    j1.*,
    cfa.RX_COMP_WK
  FROM Join_1 j1
  LEFT JOIN COMP_FLAG_ACT01 cfa
    ON j1._B631_S_PROFTCTR = cfa.PRCTR
    AND j1.MANDT = cfa.MANDT
),
