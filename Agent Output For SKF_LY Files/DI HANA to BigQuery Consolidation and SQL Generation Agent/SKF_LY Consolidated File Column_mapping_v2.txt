-- ============================================================================
-- CTE 1: WEEKLY_SNAPSHOT_DS05
-- Source: CV_BASE_SKF_WEEKLY_ACTUAL_S4
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds112
-- Purpose: Weekly SKF actuals filtered by fiscal variant, week, version, and account
-- ============================================================================
WITH WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    bic_zio_skfq,
    bic_zio_d1skq,
    bic_zio_d2skq,
    bic_zio_d3skq,
    bic_zio_d4skq,
    bic_zio_d5skq,
    bic_zio_d6skq,
    bic_zio_d7skq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.bic_azsrp_ds112`
  WHERE fiscvarnt = 'K4'
    AND bic_zio_sweek = @IP_WEEK_LY
    AND bic_zio_ver = @IP_VERSION
    AND bic_zio_saudt IN ('1','10')
    AND bic_zio_pskf IN ('804013','801000')
    AND mandt IN (110, 200)
),

-- ============================================================================
-- CTE 2: CC_HIER
-- Source: CV_BASE_MD_HRRP_NODE_S4
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node
-- Purpose: Cost center hierarchy nodes filtered by parent pattern and validity
-- ============================================================================
CC_HIER AS (
  SELECT
    mandt,
    parnode,
    hryvalto,
    nodevalue
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.hrrp_node`
  WHERE parnode LIKE '%CORERET%'
    AND hryvalto = '99991231'
    AND mandt IN (120, 200)
),

-- ============================================================================
-- CTE 3: Join_3
-- Purpose: Join weekly snapshot with cost center hierarchy
-- ============================================================================
Join_3 AS (
  SELECT
    ws.fiscper,
    ws.fiscvarnt,
    ws.fiscyear,
    ws.fiscper3,
    ws.mandt,
    ws.recordmode,
    ws.bic_zio_sweek,
    ws.b631_s_co_area,
    ws.bic_zio_cmpcd,
    ws.b631_s_proftctr,
    ws.b631_s_costcntr,
    ws.b631_s_funcarea,
    ws.bic_zio_ver,
    ws.bic_zio_saudt,
    ws.bic_zio_psty,
    ws.bic_zio_pskf,
    ws.unit,
    ws.bic_zio_skfq,
    ws.bic_zio_d1skq,
    ws.bic_zio_d2skq,
    ws.bic_zio_d3skq,
    ws.bic_zio_d4skq,
    ws.bic_zio_d5skq,
    ws.bic_zio_d6skq,
    ws.bic_zio_d7skq
  FROM WEEKLY_SNAPSHOT_DS05 ws
  INNER JOIN CC_HIER ch
    ON ws.b631_s_costcntr = ch.nodevalue
    AND ws.mandt = ch.mandt
),

-- ============================================================================
-- CTE 4: Projection_1
-- Purpose: Add current year week parameter as constant column
-- ============================================================================
Projection_1 AS (
  SELECT
    j3.*,
    @IP_WEEK_CY AS CAL_WEEK
  FROM Join_3 j3
),

-- ============================================================================
-- CTE 5: STORE_ATTR_ACTUAL
-- Source: CV_COMP_MD_SRPACT_STATIC
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act
-- Purpose: Store attribute master data (aggregated)
-- Note: This represents the output of STP_WSS_SRP_ATTRIBUTES procedure
-- ============================================================================
STORE_ATTR_ACTUAL AS (
  SELECT
    mandt,
    strnum,
    prctr,
    kokrs,
    kostl,
    rx_open_dat,
    rx_close_dat,
    fs_open_dat,
    fs_close_dat,
    rep_mkt_code,
    rep_mkt_desc,
    address,
    city,
    state,
    zipcode,
    division_code,
    division_desc,
    division_mgr,
    area_code,
    area_desc,
    area_mgr,
    region_code,
    region_desc,
    region_mgr,
    district_code,
    district_desc,
    zone_code,
    district_mgr,
    zone_desc,
    major_region_code,
    major_region,
    store_type,
    store_type_desc,
    rx_store_desc,
    distribution_cntr,
    distribution_cntr_desc,
    store_24_hr,
    budget_open_dat,
    budget_closed_dat,
    budget_reloc_dat,
    const_open_dat,
    const_close_dat,
    const_relo_dat,
    relocation_dat,
    emerg_mkt_ind,
    budget_emerg_mkt_ind,
    company_code,
    company_code_desc,
    original_fs_open_dat,
    original_rx_open_dat,
    group_one,
    group_two,
    ard_sort_key,
    area_cost_center,
    regiion_cost_center,
    county_code,
    county_desc,
    sqft_bracket_desc,
    sqft_sort_key,
    acquisition_code,
    acquisition_desc,
    photo_lab_add_dat,
    photo_lab_close_dat,
    min_hrs_ind,
    createdby,
    created_on,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    SUM(rx_hrs_oper) AS rx_hrs_oper,
    SUM(fs_hrs_oper) AS fs_hrs_oper,
    SUM(rx_store) AS rx_store,
    SUM(retail_sqft_amt) AS retail_sqft_amt,
    SUM(total_sqft_amt) AS total_sqft_amt
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_attr_act`
  WHERE mandt IN (120, 200)
  GROUP BY
    mandt, strnum, prctr, kokrs, kostl, rx_open_dat, rx_close_dat,
    fs_open_dat, fs_close_dat, rep_mkt_code, rep_mkt_desc, address,
    city, state, zipcode, division_code, division_desc, division_mgr,
    area_code, area_desc, area_mgr, region_code, region_desc, region_mgr,
    district_code, district_desc, zone_code, district_mgr, zone_desc,
    major_region_code, major_region, store_type, store_type_desc,
    rx_store_desc, distribution_cntr, distribution_cntr_desc, store_24_hr,
    budget_open_dat, budget_closed_dat, budget_reloc_dat, const_open_dat,
    const_close_dat, const_relo_dat, relocation_dat, emerg_mkt_ind,
    budget_emerg_mkt_ind, company_code, company_code_desc,
    original_fs_open_dat, original_rx_open_dat, group_one, group_two,
    ard_sort_key, area_cost_center, regiion_cost_center, county_code,
    county_desc, sqft_bracket_desc, sqft_sort_key, acquisition_code,
    acquisition_desc, photo_lab_add_dat, photo_lab_close_dat, min_hrs_ind,
    createdby, created_on, rx_division_code, rx_area_code, rx_region_code,
    rx_district_code
),

-- ============================================================================
-- CTE 6: Join_2
-- Purpose: Enrich with store attributes
-- ============================================================================
Join_2 AS (
  SELECT
    p1.*,
    sa.rep_mkt_code,
    sa.rep_mkt_desc,
    sa.emerg_mkt_ind,
    sa.division_code,
    sa.division_desc,
    sa.area_code,
    sa.area_desc,
    sa.region_code,
    sa.region_desc,
    sa.district_code,
    sa.district_desc,
    sa.strnum,
    sa.city,
    sa.state,
    sa.fs_open_dat,
    sa.rx_open_dat,
    sa.rx_division_code,
    sa.rx_area_code,
    sa.rx_region_code,
    sa.rx_district_code
  FROM Projection_1 p1
  LEFT JOIN STORE_ATTR_ACTUAL sa
    ON p1.b631_s_proftctr = sa.prctr
    AND p1.mandt = sa.mandt
),

-- ============================================================================
-- CTE 7: RETAIL_CAL
-- Source: CV_BASE_MD_RCALWEEK_S4
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek
-- Purpose: Retail calendar week master data
-- ============================================================================
RETAIL_CAL AS (
  SELECT
    rclnt,
    zzweek,
    zrwstrtdate,
    zrwenddate
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztfigl_rcalweek`
  WHERE rclnt IN (120, 200)
),

-- ============================================================================
-- CTE 8: Join_1
-- Purpose: Enrich with retail calendar dates
-- ============================================================================
Join_1 AS (
  SELECT
    j2.*,
    rc.zrwstrtdate,
    rc.zrwenddate
  FROM Join_2 j2
  LEFT JOIN RETAIL_CAL rc
    ON j2.CAL_WEEK = rc.zzweek
),

-- ============================================================================
-- CTE 9: COMP_FLAG_ACT01
-- Source: CV_COMP_MD_COMPFL_STATIC
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl
-- Purpose: Comparison flag data filtered by week range and version
-- ============================================================================
COMP_FLAG_ACT01 AS (
  SELECT
    mandt,
    prctr,
    zweek,
    zyear,
    comp_ver,
    rx_comp_wk
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.ztsrp_compfl`
  WHERE zweek >= @IP_WEEK_ENDING_FROM
    AND zweek <= @IP_WEEK_ENDING_TO
    AND comp_ver = @IP_VERSION_COMP_FLAG
),

-- ============================================================================
-- CTE 10: Join_4
-- Purpose: Enrich with comparison flags
-- ============================================================================
Join_4 AS (
  SELECT
    j1.*,
    cfa.rx_comp_wk
  FROM Join_1 j1
  LEFT JOIN COMP_FLAG_ACT01 cfa
    ON j1.b631_s_proftctr = cfa.prctr
    AND j1.mandt = cfa.mandt
),

-- ============================================================================
-- CTE 11: PROFIT_CENTER_TEXT
-- Source: CV_BASE_MD_CEPCT_S4
-- Base Table: edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct
-- Purpose: Profit center text descriptions
-- ============================================================================
PROFIT_CENTER_TEXT AS (
  SELECT
    mandt,
    prctr,
    ltext AS PROFIT_CENTER_TEXT
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_cepct`
  WHERE mandt IN (120, 200)
),

-- ============================================================================
-- CTE 12: Join_5
-- Purpose: Enrich with profit center text
-- ============================================================================
Join_5 AS (
  SELECT
    j4.*,
    pct.PROFIT_CENTER_TEXT
  FROM Join_4 j4
  LEFT JOIN PROFIT_CENTER_TEXT pct
    ON j4.b631_s_proftctr = pct.prctr
    AND j4.mandt = pct.mandt
),

-- ============================================================================
-- CTE 13: FLAGS
-- Purpose: Final projection with calculated columns
-- Calculated Columns:
--   - CAL_WEEK_NUMBER: Extract last 2 characters of week
--   - CAL_ZIO_SKFQ_QTY: Adjusted SKF quantity (doubled for account 804013)
-- ============================================================================
FLAGS AS (
  SELECT
    bic_zio_d7skq,
    bic_zio_d6skq,
    bic_zio_d5skq,
    bic_zio_d4skq,
    bic_zio_d3skq,
    bic_zio_d2skq,
    bic_zio_d1skq,
    bic_zio_skfq,
    unit,
    bic_zio_pskf,
    bic_zio_psty,
    rx_comp_wk,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    b631_s_proftctr,
    b631_s_costcntr,
    b631_s_funcarea,
    bic_zio_ver,
    bic_zio_saudt,
    rep_mkt_code,
    rep_mkt_desc,
    emerg_mkt_ind,
    division_code,
    division_desc,
    area_code,
    area_desc,
    region_code,
    region_desc,
    district_code,
    district_desc,
    strnum,
    city,
    state,
    zrwstrtdate,
    zrwenddate,
    PROFIT_CENTER_TEXT,
    fs_open_dat,
    rx_open_dat,
    CAL_WEEK,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    -- Calculated Column 1: Week number (last 2 digits)
    RIGHT(CAL_WEEK, 2) AS CAL_WEEK_NUMBER,
    -- Calculated Column 2: Adjusted SKF quantity
    CASE
      WHEN bic_zio_pskf = '804013' THEN CAST(bic_zio_skfq AS NUMERIC) * 2
      ELSE CAST(bic_zio_skfq AS NUMERIC)
    END AS CAL_ZIO_SKFQ_QTY
  FROM Join_5
)

-- ============================================================================
-- FINAL OUTPUT: CV_COMP_SKF_ACTUAL_STATIC
-- Purpose: Aggregated weekly SKF actuals with full dimensional context
-- Aggregation: SUM on all measure columns, GROUP BY on all dimension columns
-- Output Columns: 43 dimensions + 9 measures = 52 total columns
-- ============================================================================
SELECT
  -- Dimension Columns (GROUP BY)
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  b631_s_funcarea,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  rx_comp_wk,
  division_code,
  division_desc,
  area_code,
  area_desc,
  region_code,
  region_desc,
  district_code,
  district_desc,
  strnum,
  city,
  state,
  CAL_WEEK_NUMBER,
  zrwstrtdate,
  zrwenddate,
  PROFIT_CENTER_TEXT,
  fs_open_dat,
  rx_open_dat,
  CAL_WEEK,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  
  -- Measure Columns (SUM aggregation)
  SUM(bic_zio_skfq) AS bic_zio_skfq_90as1,
  SUM(bic_zio_d1skq) AS bic_zio_d1skq,
  SUM(bic_zio_d2skq) AS bic_zio_d2skq,
  SUM(bic_zio_d3skq) AS bic_zio_d3skq,
  SUM(bic_zio_d4skq) AS bic_zio_d4skq,
  SUM(bic_zio_d5skq) AS bic_zio_d5skq,
  SUM(bic_zio_d6skq) AS bic_zio_d6skq,
  SUM(bic_zio_d7skq) AS bic_zio_d7skq,
  SUM(CAL_ZIO_SKFQ_QTY) AS bic_zio_skfq

FROM FLAGS

GROUP BY
  mandt,
  fiscper,
  fiscvarnt,
  fiscyear,
  fiscper3,
  recordmode,
  bic_zio_sweek,
  b631_s_co_area,
  bic_zio_cmpcd,
  b631_s_proftctr,
  b631_s_costcntr,
  b631_s_funcarea,
  bic_zio_ver,
  bic_zio_saudt,
  rep_mkt_code,
  rep_mkt_desc,
  emerg_mkt_ind,
  bic_zio_psty,
  bic_zio_pskf,
  unit,
  rx_comp_wk,
  division_code,
  division_desc,
  area_code,
  area_desc,
  region_code,
  region_desc,
  district_code,
  district_desc,
  strnum,
  city,
  state,
  CAL_WEEK_NUMBER,
  zrwstrtdate,
  zrwenddate,
  PROFIT_CENTER_TEXT,
  fs_open_dat,
  rx_open_dat,
  CAL_WEEK,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code;
