--        runtime parameters, bic_zio_ver = @ip_version, bic_zio_saudt
--        IN ('1','10'), and bic_zio_pskf IN ('804013','801000').
--        Renames bic_zio_pctr to b631_s_proftctr and bic_zio_cctr to
--        b631_s_costcntr as defined in CV_BASE_SKF_WEEKLY_FORECAST_S4.

-- Calculation: bic_zio_pctr aliased as b631_s_proftctr (meaningful rename),
--              bic_zio_cctr aliased as b631_s_costcntr (meaningful rename)
-- =====================================================================
WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,
    bic_zio_pctr        AS b631_s_proftctr,
    bic_zio_cctr        AS b631_s_costcntr,
    bic_zio_ver,
    bic_zio_saudt,
    bic_zio_psty,
    bic_zio_pskf,
    unit,
    bic_zio_skfq
  FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_bic_azsrp_ds031`
  WHERE
    mandt IN ('120', '200')
    AND fiscvarnt = 'K4'
    AND bic_zio_sweek BETWEEN @ip_week_ending_from AND @ip_week_ending_to
    AND bic_zio_ver = @ip_version
    AND bic_zio_saudt IN ('1', '10')
    AND bic_zio_pskf IN ('804013', '801000')
),

-- =====================================================================
-- CTE 10: FLAGS

--        cal_week_number = RIGHT(bic_zio_sweek, 2) (bic_zio_sweek is

--        bic_zio_pskf = '804013' THEN bic_zio_skfq * 2 ELSE

-- Source: Joined_With_ProfitCenterText
-- Calculation: cal_week_number = RIGHT(bic_zio_sweek, 2)
--              cal_bic_zio_skfq = CASE WHEN bic_zio_pskf = '804013'
--                                 THEN bic_zio_skfq * 2
--                                 ELSE bic_zio_skfq END

    unit,
    bic_zio_pskf,
    bic_zio_psty,
    rx_comp_wk,
    fiscper,
    fiscvarnt,
    fiscyear,
    fiscper3,
    mandt,
    recordmode,
    bic_zio_sweek,
    b631_s_co_area,
    bic_zio_cmpcd,

    region_code,
    region_desc,
    district_code,
    district_desc,
    zrwstrtdate,
    zrwenddate,

    state,
    profit_center_text,
    fs_open_dat,
    rx_open_dat,
    strnum,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    CASE
      WHEN bic_zio_pskf = '804013' THEN bic_zio_skfq * 2
      ELSE bic_zio_skfq
    END                                                        AS cal_bic_zio_skfq
  FROM Joined_With_ProfitCenterText
)

-- Source: FLAGS
-- Aggregation: SUM(bic_zio_skfq) AS bic_zio_skfq_1,
--              SUM(cal_bic_zio_skfq) AS bic_zio_skfq
-- =====================================================================
