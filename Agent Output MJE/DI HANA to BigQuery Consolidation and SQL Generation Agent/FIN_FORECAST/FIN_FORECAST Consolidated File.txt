-- -----------------------------------------------------------------------------
-- WEEK_NUMBER: Add calculated store week number
-- -----------------------------------------------------------------------------

  SELECT
    *,
    RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_STORE_WEEK_NUMBER
  FROM Joined_Weekly_Store
),

  SELECT
    MANDT,
    PRCTR,
    STRNUM,

  SELECT
    MANDT,
    PRCTR,
    STRNUM,
    FS_COMP_WK,
    RX_COMP_WK,
    ZWEEK,
    ZMONTH,
    ZYEAR,
    COMP_VER
  FROM Base_CompFlag_Filtered
  WHERE COMP_VER = @IP_VERSION
),

-- -----------------------------------------------------------------------------
-- Join_3: Join with component flags
-- -----------------------------------------------------------------------------
Joined_Weekly_CompFlag AS (
  SELECT
    w.*,
    c.FS_COMP_WK,
    c.RX_COMP_WK
  FROM WEEK_NUMBER w
  LEFT JOIN COMP_FLAG_FORECAST c
    ON w._B631_S_PROFTCTR = c.PRCTR
    AND w._BIC_ZIO_SWEEK = c.ZWEEK
),

-- -----------------------------------------------------------------------------
-- CAL_WEEK: Expanded from CV_BASE_MD_RCALWEEK_S4
-- Source: SAP_S4.ZTFIGL_RCALWEEK
-- Logic: Calendar week master data filtered by RCLNT
-- -----------------------------------------------------------------------------
CAL_WEEK AS (
  SELECT
    RCLNT,
    ZZWEEK,
    ZCALYRP,
    ZRYEAR,
    ZRPERIOD,
    ZRYRP,
    ZRYRQTR,
    ZRWSTRTDATE,
    ZRWENDDATE
  FROM `PROJECT.DATASET.ZTFIGL_RCALWEEK`
  WHERE RCLNT IN (120, 200)
),

-- -----------------------------------------------------------------------------
-- Join_4: Join with calendar week
-- -----------------------------------------------------------------------------
Joined_Weekly_CalWeek AS (
  SELECT
    j.*,
    cal.ZRWSTRTDATE,
    cal.ZRWENDDATE
  FROM Joined_Weekly_CompFlag j
  LEFT JOIN CAL_WEEK cal
    ON j._BIC_ZIO_SWEEK = cal.ZZWEEK
),

-- -----------------------------------------------------------------------------
-- PROFIT_CENTER_TEXT: Expanded from CV_BASE_MD_CEPCT_S4
-- Source: SAP_S4.CEPCT
-- Logic: Profit center text filtered by MANDT
-- -----------------------------------------------------------------------------
PROFIT_CENTER_TEXT AS (
  SELECT
    MANDT,
    SPRAS,
    PRCTR,
    DATBI,
    KOKRS,
    KTEXT,
    LTEXT,
    MCTXT
  FROM `PROJECT.DATASET.CEPCT`
  WHERE MANDT IN (120, 200)
),

-- -----------------------------------------------------------------------------
-- Join_1: Join with profit center text
-- -----------------------------------------------------------------------------
Joined_Weekly_ProfitCenter AS (
  SELECT
    j.*,
    p.LTEXT AS PROFIT_CENTER_TEXT
  FROM Joined_Weekly_CalWeek j
  LEFT JOIN PROFIT_CENTER_TEXT p
    ON j._B631_S_PROFTCTR = p.PRCTR
),

-- -----------------------------------------------------------------------------
-- FLAGS: Final projection with calculated columns
-- -----------------------------------------------------------------------------
FLAGS AS (
  SELECT
    RX_COMP_WK,
    FS_COMP_WK,
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    _BIC_ZIO_SWEEK,

    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _B631_S_GL_ACCT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    _B631_S_AMOUNT_NEGATIVE,
    CURRENCY,
    _BIC_ZIO_AMT,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,
    DIVISION_DESC,
    REGION_CODE,
    REGION_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    AREA_CODE,

    STRNUM,
    ZRWSTRTDATE,
    ZRWENDDATE,
    CITY,
    STATE,
    PROFIT_CENTER_TEXT,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    -- Calculated columns
    CASE
      WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
      WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS CAL_FS_RX_FLAG,
    CASE
      WHEN (CASE
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'FS' THEN FS_COMP_WK
      WHEN (CASE
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'RX' THEN RX_COMP_WK
      ELSE FS_COMP_WK
    END AS CAL_COMP_FLAG,
    RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    (_B631_S_AMOUNT_NEGATIVE * -1) AS _B631_S_AMOUNT
  FROM Joined_Weekly_ProfitCenter
