-- ============================================================================
-- CONSOLIDATED BIGQUERY SQL: CV_COMP_FIN_FORECAST
-- ============================================================================
-- This consolidated SQL expands all upstream dependencies inline
-- Parameters: @IP_VERSION, @IP_WEEK_ENDING_FROM, @IP_WEEK_ENDING_TO
-- ============================================================================

WITH WEEKLY_SNAPSHOT_DS05 AS (
  -- Expanded from CV_BASE_FIN_WEEKLY_FORECAST_S4
  -- Source: CVS_FRIP.AZSRP_DS041_VT_S4
  -- Applies filters: MANDT IN (120, 200), FISCVARNT = 'K4', 
  --   _BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO,
  --   _BIC_ZIO_VER = @IP_VERSION, _BIC_ZIO_SAUDT IN ('1','10')
  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    _BIC_ZIO_SWEEK,
    _B631_S_CHRTACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _BIC_ZIO_PCTR AS _B631_S_PROFTCTR,
    _BIC_ZIO_CCTR AS _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _B631_S_GL_ACCT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    _B631_S_AMOUNT AS _B631_S_AMOUNT_NEGATIVE,
    CURRENCY,
    _BIC_ZIO_AMT
  FROM PROJECT.DATASET.AZSRP_DS041_VT_S4
  WHERE 
    MANDT IN (120, 200)
    AND FISCVARNT = 'K4'
    AND _BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
    AND _BIC_ZIO_VER = @IP_VERSION
    AND _BIC_ZIO_SAUDT IN ('1', '10')
),

STORE_ATTR_ACTUAL AS (
  -- Expanded from CV_BASE_MD_SRPACT_S4
  -- Source: SAP_S4.ZTSRP_ATTR_ACT
  -- Applies filter: MANDT IN (120, 200)
  -- Aggregates measures: RX_HRS_OPER, FS_HRS_OPER, RETAIL_SQFT_AMT, TOTAL_SQFT_AMT
  SELECT
    MANDT,
    STRNUM,
    PRCTR,
    KOKRS,
    KOSTL,
    RX_OPEN_DAT,
    RX_CLOSE_DAT,
    FS_OPEN_DAT,
    FS_CLOSE_DAT,
    REP_MKT_CODE,
    REP_MKT_DESC,
    ADDRESS,
    CITY,
    STATE,
    ZIPCODE,
    DIVISION_CODE,
    DIVISION_DESC,
    DIVISION_MGR,
    AREA_CODE,
    AREA_DESC,
    AREA_MGR,
    REGION_CODE,
    REGION_DESC,
    REGION_MGR,
    DISTRICT_CODE,
    DISTRICT_DESC,
    ZONE_CODE,
    DISTRICT_MGR,
    ZONE_DESC,
    MAJOR_REGION_CODE,
    MAJOR_REGION,
    STORE_TYPE,
    STORE_TYPE_DESC,
    RX_STORE,
    RX_STORE_DESC,
    DISTRIBUTION_CNTR,
    DISTRIBUTION_CNTR_DESC,
    STORE_24_HR,
    BUDGET_OPEN_DAT,
    BUDGET_CLOSED_DAT,
    BUDGET_RELOC_DAT,
    CONST_OPEN_DAT,
    CONST_CLOSE_DAT,
    CONST_RELO_DAT,
    RELOCATION_DAT,
    EMERG_MKT_IND,
    BUDGET_EMERG_MKT_IND,
    COMPANY_CODE,
    COMPANY_CODE_DESC,
    ORIGINAL_FS_OPEN_DAT,
    ORIGINAL_RX_OPEN_DAT,
    GROUP_ONE,
    GROUP_TWO,
    ARD_SORT_KEY,
    AREA_COST_CENTER,
    REGIION_COST_CENTER,
    COUNTY_CODE,
    COUNTY_DESC,
    SQFT_BRACKET_DESC,
    SQFT_SORT_KEY,
    ACQUISITION_CODE,
    ACQUISITION_DESC,
    PHOTO_LAB_ADD_DAT,
    PHOTO_LAB_CLOSE_DAT,
    MIN_HRS_IND,
    CREATEDBY,
    CREATED_ON,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    SUM(RX_HRS_OPER) AS RX_HRS_OPER,
    SUM(FS_HRS_OPER) AS FS_HRS_OPER,
    SUM(RETAIL_SQFT_AMT) AS RETAIL_SQFT_AMT,
    SUM(TOTAL_SQFT_AMT) AS TOTAL_SQFT_AMT
  FROM PROJECT.DATASET.ZTSRP_ATTR_ACT
  WHERE MANDT IN (120, 200)
  GROUP BY
    MANDT, STRNUM, PRCTR, KOKRS, KOSTL, RX_OPEN_DAT, RX_CLOSE_DAT, 
    FS_OPEN_DAT, FS_CLOSE_DAT, REP_MKT_CODE, REP_MKT_DESC, ADDRESS, 
    CITY, STATE, ZIPCODE, DIVISION_CODE, DIVISION_DESC, DIVISION_MGR, 
    AREA_CODE, AREA_DESC, AREA_MGR, REGION_CODE, REGION_DESC, REGION_MGR, 
    DISTRICT_CODE, DISTRICT_DESC, ZONE_CODE, DISTRICT_MGR, ZONE_DESC, 
    MAJOR_REGION_CODE, MAJOR_REGION, STORE_TYPE, STORE_TYPE_DESC, RX_STORE, 
    RX_STORE_DESC, DISTRIBUTION_CNTR, DISTRIBUTION_CNTR_DESC, STORE_24_HR, 
    BUDGET_OPEN_DAT, BUDGET_CLOSED_DAT, BUDGET_RELOC_DAT, CONST_OPEN_DAT, 
    CONST_CLOSE_DAT, CONST_RELO_DAT, RELOCATION_DAT, EMERG_MKT_IND, 
    BUDGET_EMERG_MKT_IND, COMPANY_CODE, COMPANY_CODE_DESC, ORIGINAL_FS_OPEN_DAT, 
    ORIGINAL_RX_OPEN_DAT, GROUP_ONE, GROUP_TWO, ARD_SORT_KEY, AREA_COST_CENTER, 
    REGIION_COST_CENTER, COUNTY_CODE, COUNTY_DESC, SQFT_BRACKET_DESC, 
    SQFT_SORT_KEY, ACQUISITION_CODE, ACQUISITION_DESC, PHOTO_LAB_ADD_DAT, 
    PHOTO_LAB_CLOSE_DAT, MIN_HRS_IND, CREATEDBY, CREATED_ON, RX_DIVISION_CODE, 
    RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
),

Joined_Weekly_Store AS (
  -- Left join WEEKLY_SNAPSHOT_DS05 with STORE_ATTR_ACTUAL on _B631_S_PROFTCTR = PRCTR
  SELECT
    f.FISCPER,
    f.FISCVARNT,
    f.FISCYEAR,
    f.FISCPER3,
    f.MANDT,
    f._BIC_ZIO_SWEEK,
    f._B631_S_CHRTACCT,
    f._B631_S_CO_AREA,
    f._BIC_ZIO_CMPCD,
    f._B631_S_PROFTCTR,
    f._B631_S_COSTCNTR,
    f._B631_S_FUNCAREA,
    f._BIC_ZIO_VER,
    f._BIC_ZIO_SAUDT,
    f._B631_S_GL_ACCT,
    f._BIC_ZWWPC_PA1,
    f._BIC_ZWWSC_PA1,
    f._B631_S_AMOUNT_NEGATIVE,
    f.CURRENCY,
    f._BIC_ZIO_AMT,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,
    s.EMERG_MKT_IND,
    s.DIVISION_CODE,
    s.DIVISION_DESC,
    s.REGION_CODE,
    s.REGION_DESC,
    s.DISTRICT_CODE,
    s.DISTRICT_DESC,
    s.AREA_CODE,
    s.AREA_DESC,
    s.STRNUM,
    s.CITY,
    s.STATE,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM WEEKLY_SNAPSHOT_DS05 f
  LEFT JOIN STORE_ATTR_ACTUAL s
    ON f._B631_S_PROFTCTR = s.PRCTR
),

COMP_FLAG_FORECAST AS (
  -- Expanded from CV_BASE_MD_COMPFL_S4
  -- Source: SAP_S4.ZTSRP_COMPFL
  -- Applies filters: MANDT IN (110, 200), COMP_VER = @IP_VERSION
  SELECT
    MANDT,
    PRCTR,
    STRNUM,
    COMP_VER,
    ZWEEK,
    ZMONTH,
    ZYEAR,
    FS_COMP_PRE,
    RX_COMP_PRE,
    FS_COMP_MON,
    RX_COMP_MON,
    FS_COMP_WK,
    RX_COMP_WK
  FROM PROJECT.DATASET.ZTSRP_COMPFL
  WHERE 
    MANDT IN (110, 200)
    AND COMP_VER = @IP_VERSION
),

Joined_Weekly_CompFlag AS (
  -- Left join Joined_Weekly_Store with COMP_FLAG_FORECAST 
  -- on _B631_S_PROFTCTR = PRCTR and _BIC_ZIO_SWEEK = ZWEEK
  SELECT
    w.*,
    c.FS_COMP_WK,
    c.RX_COMP_WK
  FROM Joined_Weekly_Store w
  LEFT JOIN COMP_FLAG_FORECAST c
    ON w._B631_S_PROFTCTR = c.PRCTR
    AND w._BIC_ZIO_SWEEK = c.ZWEEK
),

CAL_WEEK AS (
  -- Expanded from CV_BASE_MD_RCALWEEK_S4
  -- Source: SAP_S4.ZTFIGL_RCALWEEK
  -- Applies filter: RCLNT IN (120, 200)
  SELECT
    RCLNT,
    ZZWEEK,
    ZCALYRP,
    ZRYEAR,
    ZRPERIOD,
    ZRYRP,
    ZRYRQTR,
    ZRWSTRTDATE,
    ZRWENDDATE
  FROM PROJECT.DATASET.ZTFIGL_RCALWEEK
  WHERE RCLNT IN (120, 200)
),

Joined_Weekly_CalWeek AS (
  -- Left join Joined_Weekly_CompFlag with CAL_WEEK on _BIC_ZIO_SWEEK = ZZWEEK
  SELECT
    j.*,
    cal.ZRWSTRTDATE,
    cal.ZRWENDDATE
  FROM Joined_Weekly_CompFlag j
  LEFT JOIN CAL_WEEK cal
    ON j._BIC_ZIO_SWEEK = cal.ZZWEEK
),

PROFIT_CENTER_TEXT AS (
  -- Expanded from CV_BASE_MD_CEPCT_S4
  -- Source: SAP_S4.CEPCT
  -- Applies filter: MANDT IN (120, 200)
  SELECT
    MANDT,
    SPRAS,
    PRCTR,
    DATBI,
    KOKRS,
    KTEXT,
    LTEXT,
    MCTXT
  FROM PROJECT.DATASET.CEPCT
  WHERE MANDT IN (120, 200)
),

Joined_Weekly_ProfitCenter AS (
  -- Left join Joined_Weekly_CalWeek with PROFIT_CENTER_TEXT on _B631_S_PROFTCTR = PRCTR
  SELECT
    j.*,
    p.LTEXT AS PROFIT_CENTER_TEXT
  FROM Joined_Weekly_CalWeek j
  LEFT JOIN PROFIT_CENTER_TEXT p
    ON j._B631_S_PROFTCTR = p.PRCTR
),

FLAGS AS (
  -- Calculate derived columns: CAL_FS_RX_FLAG, CAL_COMP_FLAG, CAL_WEEK_NUMBER, _B631_S_AMOUNT
  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    _BIC_ZIO_SWEEK,
    _B631_S_CHRTACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _B631_S_GL_ACCT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    _B631_S_AMOUNT_NEGATIVE,
    CURRENCY,
    _BIC_ZIO_AMT,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    FS_COMP_WK,
    RX_COMP_WK,
    DIVISION_CODE,
    DIVISION_DESC,
    REGION_CODE,
    REGION_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    AREA_CODE,
    AREA_DESC,
    STRNUM,
    ZRWSTRTDATE,
    ZRWENDDATE,
    CITY,
    STATE,
    PROFIT_CENTER_TEXT,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    -- Calculated columns
    CASE
      WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
      WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS CAL_FS_RX_FLAG,
    CASE
      WHEN (CASE
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'FS' THEN FS_COMP_WK
      WHEN (CASE
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'RX' THEN RX_COMP_WK
      ELSE FS_COMP_WK
    END AS CAL_COMP_FLAG,
    RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    (_B631_S_AMOUNT_NEGATIVE * -1) AS _B631_S_AMOUNT
  FROM Joined_Weekly_ProfitCenter
)

-- Final aggregated output as per CUBE semantics
SELECT
  MANDT,
  FISCPER,
  FISCVARNT,
  FISCYEAR,
  FISCPER3,
  _BIC_ZIO_SWEEK,
  _B631_S_CHRTACCT,
  _B631_S_CO_AREA,
  _BIC_ZIO_CMPCD,
  _B631_S_PROFTCTR,
  CAL_FS_RX_FLAG AS FS_RX_FLAG,
  _B631_S_COSTCNTR,
  _B631_S_FUNCAREA,
  _BIC_ZIO_VER,
  _BIC_ZIO_SAUDT,
  _B631_S_GL_ACCT,
  _BIC_ZWWPC_PA1,
  _BIC_ZWWSC_PA1,
  CURRENCY,
  REP_MKT_CODE,
  REP_MKT_DESC,
  EMERG_MKT_IND,
  FS_COMP_WK,
  RX_COMP_WK,
  CAL_COMP_FLAG AS COMP_FLAG,
  DIVISION_CODE,
  DIVISION_DESC,
  REGION_CODE,
  REGION_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  AREA_CODE,
  AREA_DESC,
  STRNUM,
  ZRWSTRTDATE,
  ZRWENDDATE,
  CAL_WEEK_NUMBER,
  CITY,
  STATE,
  PROFIT_CENTER_TEXT,
  FS_OPEN_DAT,
  RX_OPEN_DAT,
  RX_DIVISION_CODE,
  RX_AREA_CODE,
  RX_REGION_CODE,
  RX_DISTRICT_CODE,
  -- Aggregated measures
  SUM(_B631_S_AMOUNT_NEGATIVE) AS _B631_S_AMOUNT_NEGATIVE,
  SUM(_BIC_ZIO_AMT) AS _BIC_ZIO_AMT,
  SUM(_B631_S_AMOUNT) AS _B631_S_AMOUNT
FROM FLAGS
GROUP BY
  MANDT,
  FISCPER,
  FISCVARNT,
  FISCYEAR,
  FISCPER3,
  _BIC_ZIO_SWEEK,
  _B631_S_CHRTACCT,

  _BIC_ZIO_CMPCD,
  _B631_S_PROFTCTR,
  CAL_FS_RX_FLAG,

  _BIC_ZIO_SAUDT,
  _B631_S_GL_ACCT,
  _BIC_ZWWPC_PA1,
  _BIC_ZWWSC_PA1,
  CURRENCY,
  REP_MKT_CODE,
  REP_MKT_DESC,
  EMERG_MKT_IND,
  FS_COMP_WK,
  RX_COMP_WK,
  CAL_COMP_FLAG,
  DIVISION_CODE,
  DIVISION_DESC,
  REGION_CODE,
  REGION_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  AREA_CODE,
  AREA_DESC,
  STRNUM,
  ZRWSTRTDATE,
  ZRWENDDATE,
  CAL_WEEK_NUMBER,
  CITY,
  STATE,
  PROFIT_CENTER_TEXT,
  FS_OPEN_DAT,
  RX_OPEN_DAT,

  RX_DISTRICT_CODE
