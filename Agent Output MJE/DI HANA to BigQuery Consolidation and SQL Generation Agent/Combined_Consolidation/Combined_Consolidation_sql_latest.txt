-- =====================================================================
-- CTE 6: CV_COMP_FIN_FORECAST
-- Logic: Financial forecast composite view with joins and calculations
-- Source: CV_BASE_FIN_WEEKLY_FORECAST_S4 joined with master data
-- Filters: FISCVARNT = 'K4', _BIC_ZIO_SWEEK BETWEEN parameters, 
--          _BIC_ZIO_VER = @IP_FIN_VERSION, _BIC_ZIO_SAUDT IN ('1','10'),
--          FS_RX_FLAG IN ('FS', 'RX'), COMP_VER = @IP_FIN_VERSION
-- Join: Multiple left joins on profit center and week
-- Calculation: CAL_FS_RX_FLAG, CAL_COMP_FLAG, CAL_WEEK_NUMBER, _B631_S_AMOUNT
-- Aggregation: SUM on measures grouped by all attributes
-- =====================================================================
CV_COMP_FIN_FORECAST AS (
  WITH FIN_FORECAST_BASE AS (
    SELECT
      f.*,
      s.REP_MKT_CODE, s.REP_MKT_DESC, s.EMERG_MKT_IND,
      s.DIVISION_CODE, s.DIVISION_DESC, s.REGION_CODE, s.REGION_DESC,
      s.DISTRICT_CODE, s.DISTRICT_DESC, s.AREA_CODE, s.AREA_DESC,
      s.STRNUM, s.CITY, s.STATE, s.FS_OPEN_DAT, s.RX_OPEN_DAT,
      s.RX_DIVISION_CODE, s.RX_AREA_CODE, s.RX_REGION_CODE, s.RX_DISTRICT_CODE,
      RIGHT(CAST(f._BIC_ZIO_SWEEK AS STRING), 2) AS CAL_STORE_WEEK_NUMBER
    FROM CV_BASE_FIN_WEEKLY_FORECAST_S4 f
    LEFT JOIN CV_BASE_MD_SRPACT_S4 s
      ON f._B631_S_PROFTCTR = s.PRCTR
    WHERE f.FISCVARNT = 'K4'
      AND f._BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
      AND f._BIC_ZIO_VER = @IP_FIN_VERSION
      AND f._BIC_ZIO_SAUDT IN ('1', '10')
  ),
  FIN_FORECAST_WITH_COMP AS (
    SELECT
      fb.*,
      c.FS_COMP_WK, c.RX_COMP_WK
    FROM FIN_FORECAST_BASE fb
    LEFT JOIN CV_BASE_MD_COMPFL_S4 c
      ON fb._B631_S_PROFTCTR = c.PRCTR
      AND fb._BIC_ZIO_SWEEK = c.ZWEEK
      AND c.COMP_VER = @IP_FIN_VERSION
  ),
  FIN_FORECAST_WITH_WEEK AS (
    SELECT
      fc.*,
      w.ZRWSTRTDATE, w.ZRWENDDATE
    FROM FIN_FORECAST_WITH_COMP fc
    LEFT JOIN CV_BASE_MD_RCALWEEK_S4 w
      ON fc._BIC_ZIO_SWEEK = w.ZZWEEK
  ),
  FIN_FORECAST_WITH_PC_TEXT AS (
    SELECT
      fw.*,
      p.LTEXT AS PROFIT_CENTER_TEXT
    FROM FIN_FORECAST_WITH_WEEK fw
    LEFT JOIN CV_BASE_MD_CEPCT_S4 p
      ON fw._B631_S_PROFTCTR = p.PRCTR
  ),
  FIN_FORECAST_CALCULATED AS (
    SELECT
      *,
      CASE
        WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
        WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
        ELSE ''
      END AS CAL_FS_RX_FLAG,
      CASE
        WHEN (CASE
                WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
                WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
                ELSE ''
              END) = 'FS' THEN FS_COMP_WK
        WHEN (CASE
                WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
                WHEN LEFT(CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
                ELSE ''
              END) = 'RX' THEN RX_COMP_WK
        ELSE FS_COMP_WK
      END AS CAL_COMP_FLAG,
      RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
      (_B631_S_AMOUNT * -1) AS _B631_S_AMOUNT_CALC
    FROM FIN_FORECAST_WITH_PC_TEXT
  )
  SELECT
    MANDT, _BIC_ZIO_SWEEK, _B631_S_PROFTCTR,
    CAL_FS_RX_FLAG AS FS_RX_FLAG,
    REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND,
    CAL_COMP_FLAG AS COMP_FLAG,
    DIVISION_CODE, DIVISION_DESC, REGION_CODE, REGION_DESC,
    DISTRICT_CODE, DISTRICT_DESC, AREA_CODE, AREA_DESC,
    STRNUM, ZRWSTRTDATE, ZRWENDDATE, CAL_WEEK_NUMBER,
    CITY, STATE, CURRENCY AS FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(_B631_S_AMOUNT_CALC) AS _B631_S_AMOUNT,
    SUM(_BIC_ZIO_AMT) AS _BIC_ZIO_AMT
  FROM FIN_FORECAST_CALCULATED
  WHERE CAL_FS_RX_FLAG IN ('FS', 'RX')
  GROUP BY
    MANDT, _BIC_ZIO_SWEEK, _B631_S_PROFTCTR, FS_RX_FLAG,
    REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND, COMP_FLAG,
    DIVISION_CODE, DIVISION_DESC, REGION_CODE, REGION_DESC,
    DISTRICT_CODE, DISTRICT_DESC, AREA_CODE, AREA_DESC,
    STRNUM, ZRWSTRTDATE, ZRWENDDATE, CAL_WEEK_NUMBER,
    CITY, STATE, FORECAST_CURRENCY, PROFIT_CENTER_TEXT,
    FS_OPEN_DAT, RX_OPEN_DAT, RX_DIVISION_CODE, RX_AREA_CODE,
    RX_REGION_CODE, RX_DISTRICT_CODE
),

-- =====================================================================
-- CTE 8: CV_COMP_SKF_FORECAST
-- Logic: SKF forecast composite view with joins and calculations
-- Source: CV_BASE_SKF_WEEKLY_FORECAST_S4 joined with master data
-- Filters: FISCVARNT = 'K4', _BIC_ZIO_SWEEK BETWEEN parameters,
--          _BIC_ZIO_VER = @IP_SKF_VERSION, _BIC_ZIO_SAUDT IN ('1','10'),
--          _BIC_ZIO_PSKF IN ('804013', '801000'), COMP_VER = @IP_SKF_VERSION
-- Join: Multiple left joins on profit center and week
-- Calculation: CAL_WEEK_NUMBER, CAL_BIC_ZIO_SKFQ
-- Aggregation: SUM on measures grouped by all attributes
-- =====================================================================
CV_COMP_SKF_FORECAST AS (
  WITH SKF_FORECAST_BASE AS (
    SELECT
      f.*,
      s.REP_MKT_CODE, s.REP_MKT_DESC, s.EMERG_MKT_IND,
      s.DIVISION_CODE, s.DIVISION_DESC, s.AREA_CODE, s.AREA_DESC,
      s.REGION_CODE, s.REGION_DESC, s.DISTRICT_CODE, s.DISTRICT_DESC,
      s.CITY, s.STATE, s.FS_OPEN_DAT, s.RX_OPEN_DAT, s.STRNUM,
      s.RX_DIVISION_CODE, s.RX_AREA_CODE, s.RX_REGION_CODE, s.RX_DISTRICT_CODE
    FROM CV_BASE_SKF_WEEKLY_FORECAST_S4 f
    LEFT JOIN CV_BASE_MD_SRPACT_S4 s
      ON f._B631_S_PROFTCTR = s.PRCTR
    WHERE f.FISCVARNT = 'K4'
      AND f._BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
      AND f._BIC_ZIO_VER = @IP_SKF_VERSION
      AND f._BIC_ZIO_SAUDT IN ('1', '10')
      AND f._BIC_ZIO_PSKF IN ('804013', '801000')
  ),
  SKF_FORECAST_WITH_COMP AS (
    SELECT
      sb.*,
      c.RX_COMP_WK
    FROM SKF_FORECAST_BASE sb
    LEFT JOIN CV_BASE_MD_COMPFL_S4 c
      ON sb._B631_S_PROFTCTR = c.PRCTR
      AND sb._BIC_ZIO_SWEEK = c.ZWEEK
      AND c.COMP_VER = @IP_SKF_VERSION
  ),
  SKF_FORECAST_WITH_WEEK AS (
    SELECT
      sc.*,
      w.ZRWSTRTDATE, w.ZRWENDDATE
    FROM SKF_FORECAST_WITH_COMP sc
    LEFT JOIN CV_BASE_MD_RCALWEEK_S4 w
      ON sc._BIC_ZIO_SWEEK = w.ZZWEEK
  ),
  SKF_FORECAST_WITH_PC_TEXT AS (
    SELECT
      sw.*,
      p.LTEXT AS PROFIT_CENTER_TEXT
    FROM SKF_FORECAST_WITH_WEEK sw
    LEFT JOIN CV_BASE_MD_CEPCT_S4 p
      ON sw._B631_S_PROFTCTR = p.PRCTR
  ),
  SKF_FORECAST_CALCULATED AS (
    SELECT
      *,
      RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
      CASE
        WHEN _BIC_ZIO_PSKF = '804013' THEN _BIC_ZIO_SKFQ * 2
        ELSE _BIC_ZIO_SKFQ
      END AS CAL_BIC_ZIO_SKFQ
    FROM SKF_FORECAST_WITH_PC_TEXT
  )
  SELECT
    MANDT, STRNUM, _BIC_ZIO_SWEEK, _B631_S_PROFTCTR,
    REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND,
    RX_COMP_WK, DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC,
    REGION_CODE, REGION_DESC, DISTRICT_CODE, DISTRICT_DESC,
    ZRWSTRTDATE, ZRWENDDATE, CAL_WEEK_NUMBER,
    CITY, STATE, PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(_BIC_ZIO_SKFQ) AS _BIC_ZIO_SKFQ_1,
    SUM(CAL_BIC_ZIO_SKFQ) AS _BIC_ZIO_SKFQ
  FROM SKF_FORECAST_CALCULATED
  GROUP BY
    MANDT, STRNUM, _BIC_ZIO_SWEEK, _B631_S_PROFTCTR,
    REP_MKT_CODE, REP_MKT_DESC, EMERG_MKT_IND, RX_COMP_WK,
    DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC,
    REGION_CODE, REGION_DESC, DISTRICT_CODE, DISTRICT_DESC,
    ZRWSTRTDATE, ZRWENDDATE, CAL_WEEK_NUMBER,
    CITY, STATE, PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
),

-- =====================================================================
-- CTE 11: CV_COMP_FIN_SBT
-- Logic: Financial SBT (Same Branch Transfer) composite view
-- Source: CV_BASE_FIN_BKPF_S4 inner joined with CV_BASE_FIN_ACDOCA_S4
-- Filters: XREF2_HD = 'A1', BLART = 'ZT', PRCTR IN ('0000562057','0000562058'),
--          RACCT = '0004403000', ZZWEEK BETWEEN parameters
-- Join: Multiple joins with store attributes, calendar, profit center text
-- Calculation: CAL_COMP_FLAG, CAL_WEEK_NUMBER, WSL calculation
-- Aggregation: SUM on WSL measures
-- =====================================================================
CV_COMP_FIN_SBT AS (
  WITH SBT_BASE AS (
    SELECT
      b.MANDT, b.BUKRS, b.BELNR, b.GJAHR, b.XREF2_HD, b.BUDAT,
      a.WSL, a.PRCTR, a.RACCT, a.ZZWEEK, a.RWCUR
    FROM CV_BASE_FIN_BKPF_S4 b
    INNER JOIN CV_BASE_FIN_ACDOCA_S4 a
      ON b.BUKRS = a.RBUKRS
      AND b.BELNR = a.BELNR
      AND b.GJAHR = a.GJAHR
    WHERE b.XREF2_HD = 'A1'
      AND b.BLART = 'ZT'
      AND a.PRCTR IN ('0000562057', '0000562058')
      AND a.RACCT = '0004403000'
      AND a.ZZWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
  ),
  SBT_WITH_STORE AS (
    SELECT
      sb.*,
      s.REP_MKT_CODE, s.REP_MKT_DESC, s.CITY, s.STATE,
      s.DIVISION_CODE, s.DIVISION_DESC, s.AREA_CODE, s.AREA_DESC,
      s.REGION_CODE, s.REGION_DESC, s.DISTRICT_CODE, s.DISTRICT_DESC,
      s.STRNUM, s.FS_OPEN_DAT, s.RX_OPEN_DAT, s.EMERG_MKT_IND,
      s.RX_DIVISION_CODE, s.RX_AREA_CODE, s.RX_REGION_CODE, s.RX_DISTRICT_CODE
    FROM SBT_BASE sb
    LEFT JOIN CV_BASE_MD_SRPACT_S4 s
      ON sb.PRCTR = s.PRCTR
  ),
  SBT_WITH_CALENDAR AS (
    SELECT
      sw.*,
      c.ZRWENDDATE, c.ZRWSTRTDATE
    FROM SBT_WITH_STORE sw
    LEFT JOIN CV_BASE_MD_RCALWEEK_S4 c
      ON sw.ZZWEEK = c.ZZWEEK
  ),
  SBT_WITH_CALC AS (
    SELECT
      *,
      CASE WHEN PRCTR = '0000562057' THEN '1' ELSE '0' END AS CAL_COMP_FLAG,
      RIGHT(CAST(ZZWEEK AS STRING), 2) AS CAL_WEEK_NUMBER
    FROM SBT_WITH_CALENDAR
  ),
  SBT_WITH_PC_TEXT AS (
    SELECT
      sc.*,
      p.LTEXT AS PROFIT_CENTER_TEXT
    FROM SBT_WITH_CALC sc
    LEFT JOIN CV_BASE_MD_CEPCT_S4 p
      ON sc.PRCTR = p.PRCTR
  )
  SELECT
    PRCTR, BUDAT, XREF2_HD, GJAHR, BELNR, BUKRS, MANDT,
    REP_MKT_CODE, REP_MKT_DESC, CITY, STATE,
    DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC,
    REGION_CODE, REGION_DESC, DISTRICT_CODE, DISTRICT_DESC,
    ZZWEEK, STRNUM, CAL_COMP_FLAG, CAL_WEEK_NUMBER,
    ZRWENDDATE, ZRWSTRTDATE, RWCUR, PROFIT_CENTER_TEXT,
    FS_OPEN_DAT, RX_OPEN_DAT, EMERG_MKT_IND,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(WSL) * -1 AS WSL
  FROM SBT_WITH_PC_TEXT
  GROUP BY
    PRCTR, BUDAT, XREF2_HD, GJAHR, BELNR, BUKRS, MANDT,
    REP_MKT_CODE, REP_MKT_DESC, CITY, STATE,
    DIVISION_CODE, DIVISION_DESC, AREA_CODE, AREA_DESC,
    REGION_CODE, REGION_DESC, DISTRICT_CODE, DISTRICT_DESC,
    ZZWEEK, STRNUM, CAL_COMP_FLAG, CAL_WEEK_NUMBER,
    ZRWENDDATE, ZRWSTRTDATE, RWCUR, PROFIT_CENTER_TEXT,
    FS_OPEN_DAT, RX_OPEN_DAT, EMERG_MKT_IND,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
),

  WITH FIN_MJE_ACDOCA AS (
    SELECT
      RCLNT, PRCTR, ZZWEEK, KSL, BLART, RBUKRS, GJAHR, BELNR, WWPC_PA, RACCT

    WHERE ZZWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
  ),
  FIN_MJE_WITH_BKPF AS (
    SELECT
      a.RCLNT, a.PRCTR, a.ZZWEEK, a.KSL, b.MANDT, b.XREF2_HD, a.WWPC_PA, a.RACCT
    FROM FIN_MJE_ACDOCA a
    INNER JOIN CV_BASE_FIN_BKPF_S4 b
      ON a.RBUKRS = b.BUKRS
      AND a.BELNR = b.BELNR
      AND a.GJAHR = b.GJAHR
    WHERE b.XREF2_HD IN ('MJE','TS','GENCO','MAR','SAPARINVOICES','TP')

    FROM FIN_MJE_FILTERED f
    INNER JOIN CV_BASE_MD_HRRP_NODE_S4 p
      ON f.PRCTR = p.NODEVALUE
      AND f.MANDT = p.MANDT
    WHERE p.PARNODE LIKE '%CORE_RET%' AND p.HRYVALTO = '99991231'

  FIN_MJE_WITH_GL_HIER AS (
    SELECT j.*
    FROM FIN_MJE_WITH_PC_HIER j
    INNER JOIN CV_BASE_MD_HRRP_NODE_S4 g
      ON j.RACCT = g.NODEVALUE

      AND g.MANDT = p.MANDT

  ),
  FIN_MJE_WITH_STORE AS (
    SELECT
      j.MANDT, s.REP_MKT_CODE, s.REP_MKT_DESC, s.DIVISION_CODE, s.AREA_CODE,
      s.DIVISION_DESC, s.AREA_DESC, s.REGION_CODE, s.REGION_DESC,

    FROM FIN_MJE_WITH_PROD_HIER j
    LEFT JOIN CV_BASE_MD_SRPACT_S4 s
      ON j.PRCTR = s.PRCTR
      AND j.MANDT = s.MANDT

      s.RX_OPEN_DAT, s.PARNODE, s.CITY, s.STATE, s.WWPC_PA, s.STRNUM,
      s.MANDT, s.SOURCE_CODE, s.RX_DIVISION_CODE, s.RX_AREA_CODE,
      s.RX_REGION_CODE, s.RX_DISTRICT_CODE
    FROM FIN_MJE_WITH_STORE s
    LEFT JOIN CV_BASE_MD_COMPFL_S4 c
      ON s.PRCTR = c.PRCTR
      AND s.ZZWEEK = c.ZWEEK
      AND s.MANDT = c.MANDT
      AND c.COMP_VER = @IP_VERSION_COMP_FLAG
  ),
  FIN_MJE_WITH_PC_TEXT AS (
    SELECT
      c.MANDT, c.RX_COMP_WK, c.FS_COMP_WK, c.EMERG_MKT_IND, c.DISTRICT_CODE,

    LEFT JOIN CV_BASE_MD_CEPCT_S4 p
      ON c.PRCTR = p.PRCTR
      AND c.MANDT = p.MANDT
  ),
  FIN_MJE_WITH_CALENDAR AS (
    SELECT

      CASE
        WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'FS' THEN 'FS'
        WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'RX' THEN 'RX'
        ELSE ''
      END AS CAL_FS_RX_FLAG
    FROM FIN_MJE_WITH_PC_TEXT j
    LEFT JOIN CV_BASE_MD_RCALWEEK_S4 c
      ON j.ZZWEEK = c.ZZWEEK
      AND j.MANDT = c.RCLNT
  )
  SELECT
    MANDT, RX_COMP_WK, FS_COMP_WK, EMERG_MKT_IND, DISTRICT_CODE,
    REGION_DESC, REGION_CODE, RCLNT, PRCTR, ZZWEEK, DISTRICT_DESC,
    AREA_DESC, DIVISION_DESC, AREA_CODE, DIVISION_CODE, REP_MKT_DESC,
    REP_MKT_CODE, PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT, PARNODE,
    ZRWSTRTDATE, ZRWENDDATE, CITY, STATE, CAL_WEEK_NUMBER,
    CASE
      WHEN CAL_FS_RX_FLAG = 'FS' THEN MAX(FS_COMP_WK)
      WHEN CAL_FS_RX_FLAG = 'RX' THEN MAX(RX_COMP_WK)
      ELSE MAX(FS_COMP_WK)
    END AS COMP_FLAG,
    CAL_FS_RX_FLAG, STRNUM, SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(KSL) * -1 AS KSL,
    SUM(CASE WHEN RIGHT(PARNODE, 2) = 'FS' THEN KSL * -1 ELSE 0 END) AS CAL_FS_SALES,
    SUM(CASE WHEN RIGHT(PARNODE, 2) = 'RX' THEN KSL * -1 ELSE 0 END) AS CAL_RX_SALES
  FROM FIN_MJE_WITH_CALENDAR
  GROUP BY
    MANDT, RX_COMP_WK, FS_COMP_WK, EMERG_MKT_IND, DISTRICT_CODE,
    REGION_DESC, REGION_CODE, RCLNT, PRCTR, ZZWEEK, DISTRICT_DESC,
    AREA_DESC, DIVISION_DESC, AREA_CODE, DIVISION_CODE, REP_MKT_DESC,
    REP_MKT_CODE, PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT, PARNODE,
    ZRWSTRTDATE, ZRWENDDATE, CITY, STATE, CAL_WEEK_NUMBER, CAL_FS_RX_FLAG,
    STRNUM, SOURCE_CODE, RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE,
    RX_DISTRICT_CODE
),

-- =====================================================================
-- CTE 14: CV_BASE_SKF_FAGLSKF_PN_S4
-- Logic: SKF statistical key figure plan data
-- Source: FAGLSKF_PN_VT_S4 table
-- Filters: RCLNT IN (120, 200)
-- Aggregation: SUM on MSL
-- =====================================================================
CV_BASE_SKF_FAGLSKF_PN_S4 AS (
  SELECT
    RCLNT, STAGR, DATE_FROM, DOCNR, DOCLN, DATE_TO, RBUKRS, RCNTR,
    PRCTR, RFAREA, RBUSA, KOKRS, SEGMENT, ACTIV, OBART, RUNIT,
    LOGSYS, RRCTY, RVERS, REFDOCNR, CANC_DOCNR, USNAM, TIMESTAMP,
    DELTA_IND, ZZSCODE, ZZWEEK,
    SUM(MSL) AS MSL
  FROM `PROJECT.DATASET.FAGLSKF_PN_VT_S4`
  WHERE RCLNT IN (120, 200)
  GROUP BY
    RCLNT, STAGR, DATE_FROM, DOCNR, DOCLN, DATE_TO, RBUKRS, RCNTR,
    PRCTR, RFAREA, RBUSA, KOKRS, SEGMENT, ACTIV, OBART, RUNIT,
    LOGSYS, RRCTY, RVERS, REFDOCNR, CANC_DOCNR, USNAM, TIMESTAMP,
    DELTA_IND, ZZSCODE, ZZWEEK
),

-- =====================================================================
-- CTE 15: CV_COMP_SKF_MANUAL_JE
-- Logic: SKF manual journal entries composite view
-- Source: CV_BASE_SKF_FAGLSKF_PN_S4 with hierarchy and master joins
-- Filters: STAGR IN ('801000','804013'), ZZWEEK BETWEEN parameters,
--          ZZSCODE != 'SLS', PARNODE matching '*CORERET*', HRYVALTO = '99991231',
--          COMP_VER = @IP_VERSION_COMP_FLAG
-- Join: Inner join with hierarchy, left joins with master data
-- Calculation: CAL_MSL, CAL_WEEK_NUMBER, COMP_FLAG
-- Aggregation: SUM on CAL_MSL as MSL
-- =====================================================================
CV_COMP_SKF_MANUAL_JE AS (
  WITH SKF_MJE_BASE AS (
    SELECT
      f.RCLNT, f.STAGR, f.DATE_FROM, f.PRCTR, f.ZZWEEK, f.MSL,
      f.RCNTR, f.ZZSCODE AS SOURCE_CODE
    FROM CV_BASE_SKF_FAGLSKF_PN_S4 f
    INNER JOIN CV_BASE_MD_HRRP_NODE_S4 h
      ON f.RCNTR = h.NODEVALUE
    WHERE f.STAGR IN ('801000', '804013')
      AND f.ZZWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
      AND f.ZZSCODE != 'SLS'
      AND REGEXP_CONTAINS(h.PARNODE, r'.*CORERET.*')
      AND h.HRYVALTO = '99991231'
  ),
  SKF_MJE_WITH_STORE AS (
    SELECT
      s.REP_MKT_CODE, s.REP_MKT_DESC, s.CITY, s.STATE, s.DIVISION_CODE,
      s.DIVISION_DESC, s.AREA_CODE, s.AREA_DESC, s.REGION_CODE,
      s.REGION_DESC, s.DISTRICT_CODE, s.DISTRICT_DESC, s.EMERG_MKT_IND,
      f.RCLNT, f.STAGR, f.DATE_FROM, f.PRCTR, f.ZZWEEK, f.MSL, f.RCNTR,
      s.FS_OPEN_DAT, s.RX_OPEN_DAT, s.STRNUM, f.SOURCE_CODE,
      s.RX_DIVISION_CODE, s.RX_AREA_CODE, s.RX_REGION_CODE, s.RX_DISTRICT_CODE
    FROM SKF_MJE_BASE f
    LEFT JOIN CV_BASE_MD_SRPACT_S4 s
      ON f.PRCTR = s.PRCTR
  ),
  SKF_MJE_WITH_COMP AS (
    SELECT
      j.*, c.FS_COMP_WK, c.RX_COMP_WK
    FROM SKF_MJE_WITH_STORE j
    LEFT JOIN CV_BASE_MD_COMPFL_S4 c
      ON j.PRCTR = c.PRCTR
      AND j.ZZWEEK = c.ZWEEK
      AND c.COMP_VER = @IP_VERSION_COMP_FLAG
  ),
  SKF_MJE_WITH_CALENDAR AS (
    SELECT
      j.*, w.ZRWSTRTDATE, w.ZRWENDDATE
    FROM SKF_MJE_WITH_COMP j
    LEFT JOIN CV_BASE_MD_RCALWEEK_S4 w
      ON j.ZZWEEK = w.ZZWEEK
  ),
  SKF_MJE_WITH_PC_TEXT AS (
    SELECT
      j.*, p.LTEXT AS PROFIT_CENTER_TEXT
    FROM SKF_MJE_WITH_CALENDAR j
    LEFT JOIN CV_BASE_MD_CEPCT_S4 p
      ON j.PRCTR = p.PRCTR
  ),
  SKF_MJE_CALCULATED AS (
    SELECT
      *,
      CASE WHEN STAGR = '801000' THEN MSL ELSE MSL * 2 END AS CAL_MSL,
      RIGHT(CAST(ZZWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
      RX_COMP_WK AS CAL_COMP_FLAG
    FROM SKF_MJE_WITH_PC_TEXT
  )
  SELECT
    EMERG_MKT_IND, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    AREA_DESC, AREA_CODE, DIVISION_DESC, DIVISION_CODE, STATE, CITY,
    REP_MKT_DESC, REP_MKT_CODE, ZZWEEK, PRCTR, DATE_FROM, STAGR, RCLNT,
    FS_COMP_WK, RX_COMP_WK, ZRWSTRTDATE, ZRWENDDATE, RCNTR,
    PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT, CAL_WEEK_NUMBER,
    CAL_COMP_FLAG AS COMP_FLAG, STRNUM, SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(CAL_MSL) AS MSL
  FROM SKF_MJE_CALCULATED
  GROUP BY
    EMERG_MKT_IND, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    AREA_DESC, AREA_CODE, DIVISION_DESC, DIVISION_CODE, STATE, CITY,
    REP_MKT_DESC, REP_MKT_CODE, ZZWEEK, PRCTR, DATE_FROM, STAGR, RCLNT,
    FS_COMP_WK, RX_COMP_WK, ZRWSTRTDATE, ZRWENDDATE, RCNTR,
    PROFIT_CENTER_TEXT, FS_OPEN_DAT, RX_OPEN_DAT, CAL_WEEK_NUMBER,
    COMP_FLAG, STRNUM, SOURCE_CODE, RX_DIVISION_CODE, RX_AREA_CODE,
    RX_REGION_CODE, RX_DISTRICT_CODE
),

-- =====================================================================
-- CTE 16: CV_COMP_FORECAST_MJE_COMBINED
-- Logic: Union all five data sources (FIN_FORECAST, SKF_FORECAST, FIN_SBT,
--        FIN_MANUAL_JE, SKF_MANUAL_JE) with explicit column mapping and constants
-- Source: CV_COMP_FIN_FORECAST, CV_COMP_SKF_FORECAST, CV_COMP_FIN_SBT,
--         CV_COMP_FIN_MANUAL_JE, CV_COMP_SKF_MANUAL_JE
-- Calculation: Mapping columns to unified output structure with source-specific
--              constants for CAL_SOURCE, CAL_DATA_CATEGORY1, CAL_DATA_CATEGORY2
-- =====================================================================
CV_COMP_FORECAST_MJE_COMBINED AS (
  -- Branch 1: FIN_FORECAST
  SELECT
    FS_RX_FLAG,
    _B631_S_AMOUNT AS FIN_FCST_AMOUNT,

    'Transactional Sales' AS CAL_SOURCE,

    DIVISION_DESC, DIVISION_CODE, COMP_FLAG, _BIC_ZIO_SWEEK,
    REP_MKT_DESC, REP_MKT_CODE, EMERG_MKT_IND, _B631_S_PROFTCTR AS PRCTR,
    MANDT, FS_RX_FLAG AS FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, 'FIN' AS CAL_DATA_CATEGORY1,
    'FIN_FCST' AS CAL_DATA_CATEGORY2, FS_OPEN_DAT, RX_OPEN_DAT,
    NULL AS WSL, NULL AS CAL_FS_SALES, NULL AS CAL_RX_SALES, NULL AS MSL,
    AREA_CODE, AREA_DESC, NULL AS SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
  FROM CV_COMP_FIN_FORECAST

  UNION ALL

  -- Branch 2: SKF_FORECAST
  SELECT
    'RX' AS FS_RX_FLAG,
    NULL AS FIN_FCST_AMOUNT,
    _BIC_ZIO_SKFQ AS SKF_FCST_QTY,
    'Transactional Sales' AS CAL_SOURCE,
    STRNUM, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    DIVISION_DESC, DIVISION_CODE, RX_COMP_WK AS COMP_FLAG, _BIC_ZIO_SWEEK,
    REP_MKT_DESC, REP_MKT_CODE, EMERG_MKT_IND, _B631_S_PROFTCTR AS PRCTR,
    MANDT, 'RX' AS FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, NULL AS FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, 'SKF' AS CAL_DATA_CATEGORY1,
    'SKF_FCST' AS CAL_DATA_CATEGORY2, FS_OPEN_DAT, RX_OPEN_DAT,
    NULL AS WSL, NULL AS CAL_FS_SALES, NULL AS CAL_RX_SALES, NULL AS MSL,
    AREA_CODE, AREA_DESC, NULL AS SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
  FROM CV_COMP_SKF_FORECAST

  UNION ALL

  -- Branch 3: FIN_SBT
  SELECT
    'FS' AS FS_RX_FLAG,
    NULL AS FIN_FCST_AMOUNT,
    NULL AS SKF_FCST_QTY,
    'Manual Adjustment Journal Entries' AS CAL_SOURCE,
    STRNUM, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    DIVISION_DESC, DIVISION_CODE, CAL_COMP_FLAG AS COMP_FLAG, ZZWEEK AS _BIC_ZIO_SWEEK,
    REP_MKT_DESC, REP_MKT_CODE, EMERG_MKT_IND, PRCTR,
    MANDT, 'FS' AS FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, NULL AS FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, 'FIN' AS CAL_DATA_CATEGORY1,
    'FIN_SBT' AS CAL_DATA_CATEGORY2, FS_OPEN_DAT, RX_OPEN_DAT,
    WSL, NULL AS CAL_FS_SALES, NULL AS CAL_RX_SALES, NULL AS MSL,
    AREA_CODE, AREA_DESC, NULL AS SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
  FROM CV_COMP_FIN_SBT

  UNION ALL

  -- Branch 4: FIN_MANUAL_JE
  SELECT
    CAL_FS_RX_FLAG AS FS_RX_FLAG,
    NULL AS FIN_FCST_AMOUNT,
    NULL AS SKF_FCST_QTY,
    'Manual Adjustment Journal Entries' AS CAL_SOURCE,
    STRNUM, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    DIVISION_DESC, DIVISION_CODE, COMP_FLAG, ZZWEEK AS _BIC_ZIO_SWEEK,
    REP_MKT_DESC, REP_MKT_CODE, EMERG_MKT_IND, PRCTR,
    MANDT, 'RX' AS FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, NULL AS FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, 'FIN' AS CAL_DATA_CATEGORY1,
    'FIN_MJE' AS CAL_DATA_CATEGORY2, FS_OPEN_DAT, RX_OPEN_DAT,
    NULL AS WSL, CAL_FS_SALES, CAL_RX_SALES, NULL AS MSL,
    AREA_CODE, AREA_DESC, SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
  FROM CV_COMP_FIN_MANUAL_JE

  UNION ALL

  -- Branch 5: SKF_MANUAL_JE
  SELECT
    'RX' AS FS_RX_FLAG,
    NULL AS FIN_FCST_AMOUNT,
    NULL AS SKF_FCST_QTY,
    'Manual Adjustment Journal Entries' AS CAL_SOURCE,
    STRNUM, DISTRICT_DESC, DISTRICT_CODE, REGION_DESC, REGION_CODE,
    DIVISION_DESC, DIVISION_CODE, COMP_FLAG, ZZWEEK AS _BIC_ZIO_SWEEK,
    REP_MKT_DESC, REP_MKT_CODE, EMERG_MKT_IND, PRCTR,
    RCLNT AS MANDT, 'RX' AS FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, NULL AS FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, 'SKF' AS CAL_DATA_CATEGORY1,
    'SKF_MJE' AS CAL_DATA_CATEGORY2, FS_OPEN_DAT, RX_OPEN_DAT,

    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE

),

-- =====================================================================
-- CTE 17: CV_COMP_FORECAST_MJE_FINAL
-- Logic: Final aggregation with calculated attributes
-- Source: CV_COMP_FORECAST_MJE_COMBINED
-- Calculation: CAL_COMP_FLAG_DESC, CAL_EMERG_MKT_IND
-- Aggregation: SUM on all measures, GROUP BY all attributes
-- =====================================================================
CV_COMP_FORECAST_MJE_FINAL AS (
  SELECT
    MANDT, CAL_SOURCE, DIVISION_CODE, DIVISION_DESC, REGION_CODE,
    REGION_DESC, DISTRICT_CODE, DISTRICT_DESC, STRNUM, PRCTR,
    EMERG_MKT_IND, REP_MKT_CODE, REP_MKT_DESC, COMP_FLAG, FS_RX_FLAG,
    _BIC_ZIO_SWEEK, FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, FORECAST_CURRENCY AS FIN_FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, CAL_DATA_CATEGORY1, CAL_DATA_CATEGORY2,
    FS_OPEN_DAT, RX_OPEN_DAT, AREA_CODE, AREA_DESC, SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE,
    SUM(FIN_FCST_AMOUNT) AS FIN_FCST_AMOUNT,
    SUM(SKF_FCST_QTY) AS SKF_FCST_QTY,
    SUM(WSL) AS SBT,
    SUM(MSL) AS SKF_MJE_QTY,
    SUM(CAL_FS_SALES) AS CAL_FS_SALES_MJE,
    SUM(CAL_RX_SALES) AS CAL_RX_SALES_MJE,
    CASE
      WHEN COMP_FLAG = '1' THEN 'Comp'
      WHEN COMP_FLAG = '0' THEN 'Non-Comp'
      ELSE ''
    END AS CAL_COMP_FLAG_DESC,
    CASE
      WHEN EMERG_MKT_IND = 'Y' THEN 'Yes'
      WHEN EMERG_MKT_IND = 'N' THEN 'No'
      ELSE ''
    END AS CAL_EMERG_MKT_IND
  FROM CV_COMP_FORECAST_MJE_COMBINED
  GROUP BY
    MANDT, CAL_SOURCE, DIVISION_CODE, DIVISION_DESC, REGION_CODE,
    REGION_DESC, DISTRICT_CODE, DISTRICT_DESC, STRNUM, PRCTR,
    EMERG_MKT_IND, REP_MKT_CODE, REP_MKT_DESC, COMP_FLAG, FS_RX_FLAG,
    _BIC_ZIO_SWEEK, FS_RX_FLAG_EXT, ZRWSTRTDATE, ZRWENDDATE,
    CAL_WEEK_NUMBER, CITY, STATE, FIN_FORECAST_CURRENCY,
    PROFIT_CENTER_TEXT, CAL_DATA_CATEGORY1, CAL_DATA_CATEGORY2,
    FS_OPEN_DAT, RX_OPEN_DAT, AREA_CODE, AREA_DESC, SOURCE_CODE,
    RX_DIVISION_CODE, RX_AREA_CODE, RX_REGION_CODE, RX_DISTRICT_CODE
)
