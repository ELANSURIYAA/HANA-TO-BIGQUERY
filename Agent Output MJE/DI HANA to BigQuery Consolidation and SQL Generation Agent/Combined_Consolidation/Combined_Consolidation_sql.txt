-- =====================================================================
-- FINAL SELECT: CV_COMP_FORECAST_MJE_STATIC Output
-- Logic: Aggregates measures by SUM and groups by all dimension columns, with calculated attributes
-- Source: VARIANCES
-- Aggregation: SUM on FIN_FCST_AMOUNT, SKF_FCST_QTY, SBT, SKF_MJE_QTY, CAL_FS_SALES_MJE, CAL_RX_SALES_MJE
-- Calculation: CAL_COMP_FLAG_DESC (CASE on COMP_FLAG), CAL_EMERG_MKT_IND (CASE on EMERG_MKT_IND)
-- =====================================================================
SELECT
  MANDT,
  CAL_SOURCE,
  DIVISION_CODE,
  DIVISION_DESC,
  REGION_CODE,
  REGION_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  STRNUM,
  PRCTR,
  EMERG_MKT_IND,
  REP_MKT_CODE,
  REP_MKT_DESC,
  COMP_FLAG,
  FS_RX_FLAG,
  _BIC_ZIO_SWEEK,
  FS_RX_FLAG_EXT,
  ZRWSTRTDATE,
  ZRWENDDATE,
  CAL_WEEK_NUMBER,
  CITY,
  STATE,
  FIN_FORECAST_CURRENCY,
  PROFIT_CENTER_TEXT,
  CAL_DATA_CATEGORY1,
  CAL_DATA_CATEGORY2,
  FS_OPEN_DAT,
  RX_OPEN_DAT,
  AREA_CODE,
  AREA_DESC,
  SOURCE_CODE,
  RX_DIVISION_CODE,
  RX_AREA_CODE,
  RX_REGION_CODE,
  RX_DISTRICT_CODE,
  SUM(FIN_FCST_AMOUNT) AS FIN_FCST_AMOUNT,
  SUM(SKF_FCST_QTY) AS SKF_FCST_QTY,
  SUM(SBT) AS SBT,
  SUM(SKF_MJE_QTY) AS SKF_MJE_QTY,
  SUM(CAL_FS_SALES_MJE) AS CAL_FS_SALES_MJE,
  SUM(CAL_RX_SALES_MJE) AS CAL_RX_SALES_MJE,
  CASE COMP_FLAG
    WHEN '1' THEN 'Comp'
    WHEN '0' THEN 'Non-Comp'
    ELSE ''
  END AS CAL_COMP_FLAG_DESC,
  CASE
    WHEN EMERG_MKT_IND = 'Y' THEN 'Yes'
    WHEN EMERG_MKT_IND = 'N' THEN 'No'
    ELSE ''
  END AS CAL_EMERG_MKT_IND
FROM VARIANCES
GROUP BY
  MANDT,
  CAL_SOURCE,
  DIVISION_CODE,
  DIVISION_DESC,
  REGION_CODE,
  REGION_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  STRNUM,
  PRCTR,
