-- =====================================================================

    _BIC_ZIO_SWEEK,
    FISCYEAR,

    _B631_S_CHRTACCT,

-- =====================================================================
-- CTE 6: FIN_FORECAST_Joined
-- Logic: Financial forecast with all joins and calculated columns
-- Source: CV_BASE_FIN_WEEKLY_FORECAST_S4_Filtered + master data joins
-- Join: LEFT JOIN on PRCTR for store attributes, comp flags, calendar, profit center text
-- Filters: FS_RX_FLAG IN ('FS', 'RX')
-- Calculation: CAL_FS_RX_FLAG, CAL_COMP_FLAG, CAL_WEEK_NUMBER, _B631_S_AMOUNT
-- =====================================================================
FIN_FORECAST_Joined AS (
  SELECT
    f.MANDT,
    f._BIC_ZIO_SWEEK,
    f._B631_S_PROFTCTR,
    CASE
      WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
      WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS FS_RX_FLAG,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,
    s.EMERG_MKT_IND,
    CASE
      WHEN (CASE
              WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'FS' THEN c.FS_COMP_WK
      WHEN (CASE
              WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'RX' THEN c.RX_COMP_WK
      ELSE c.FS_COMP_WK
    END AS COMP_FLAG,
    (f._B631_S_AMOUNT_NEGATIVE * -1) AS _B631_S_AMOUNT,
    s.DIVISION_CODE,
    s.DIVISION_DESC,
    s.REGION_CODE,
    s.REGION_DESC,
    s.DISTRICT_CODE,
    s.DISTRICT_DESC,
    s.AREA_CODE,
    s.AREA_DESC,
    s.STRNUM,
    w.ZRWSTRTDATE,
    w.ZRWENDDATE,
    RIGHT(CAST(f._BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    s.CITY,
    s.STATE,
    f.CURRENCY AS FORECAST_CURRENCY,
    p.PROFIT_CENTER_TEXT,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM CV_BASE_FIN_WEEKLY_FORECAST_S4_Filtered f
  LEFT JOIN CV_BASE_MD_SRPACT_S4_Filtered s
    ON f._B631_S_PROFTCTR = s.PRCTR
  LEFT JOIN CV_BASE_MD_COMPFL_S4_Filtered c
    ON f._B631_S_PROFTCTR = c.PRCTR
    AND f._BIC_ZIO_SWEEK = c.ZWEEK
  LEFT JOIN CV_BASE_MD_RCALWEEK_S4_Filtered w
    ON f._BIC_ZIO_SWEEK = w.ZZWEEK
  LEFT JOIN CV_BASE_MD_CEPCT_S4_Filtered p
    ON f._B631_S_PROFTCTR = p.PRCTR
  WHERE CASE
          WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
          WHEN LEFT(CAST(f._BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
          ELSE ''
        END IN ('FS', 'RX')
),

-- CTE 11: FIN_SBT_Joined
-- Logic: Financial SBT (Store-Based Transactions) with all joins and calculated columns
-- Source: INNER JOIN of BKPF and ACDOCA + LEFT JOINs to master data
-- Join: INNER JOIN on BUKRS, BELNR, GJAHR; LEFT JOIN on PRCTR for store attributes, calendar, profit center text
-- Calculation: CAL_COMP_FLAG (based on PRCTR), CAL_WEEK_NUMBER, WSL (negated)
-- =====================================================================

    b.BUKRS,

    b.XREF2_HD,
    b.BUDAT,
    a.PRCTR,
    (a.WSL * -1) AS WSL,
    a.ZZWEEK,
    a.RWCUR,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,

    s.DIVISION_DESC,
    s.AREA_CODE,

    s.RX_REGION_CODE,

    ON a.PRCTR = p.PRCTR

-- =====================================================================
-- CTE 15: FIN_MANUAL_JE_Joined
-- Logic: Financial manual journal entries with hierarchy filters and all joins
-- Source: INNER JOIN of ACDOCA and BKPF + hierarchy filters + LEFT JOINs to master data
-- Join: Multiple INNER JOINs for hierarchies (PC, GL, PROD_CLASS), LEFT JOINs for store attributes, comp flags, calendar, profit center text
-- Filters: XREF2_HD NOT IN ('RT', 'RV', 'SK'), PC_HIER PARNODE LIKE '%CORE_RET%', 
--          GL_HIER PARNODE LIKE '%181', PROD_CLASS_HIER PARNODE LIKE '%FS' OR '%RX'
-- Calculation: CAL_WEEK_NUMBER, CAL_FS_RX_FLAG, KSL (negated), CAL_FS_SALES, CAL_RX_SALES
-- =====================================================================
FIN_MANUAL_JE_Joined AS (
  SELECT
    j.MANDT,
    c.RX_COMP_WK,
    c.FS_COMP_WK,
    s.EMERG_MKT_IND,
    s.DISTRICT_CODE,
    s.REGION_DESC,
    s.REGION_CODE,
    j.RCLNT,
    j.PRCTR,
    j.ZZWEEK,
    s.DISTRICT_DESC,
    s.AREA_DESC,
    s.DIVISION_DESC,
    s.AREA_CODE,
    s.DIVISION_CODE,
    s.REP_MKT_DESC,
    s.REP_MKT_CODE,
    (j.KSL * -1) AS KSL,
    p.PROFIT_CENTER_TEXT,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    h3.PARNODE,
    w.ZRWSTRTDATE,
    w.ZRWENDDATE,
    s.CITY,
    s.STATE,
    RIGHT(CAST(j.ZZWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    CASE
      WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'FS' THEN 'FS'
      WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS CAL_FS_RX_FLAG,
    CASE
      WHEN (CASE
              WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'FS' THEN c.FS_COMP_WK
      WHEN (CASE
              WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'FS' THEN 'FS'
              WHEN LEFT(CAST(j.WWPC_PA AS STRING), 2) = 'RX' THEN 'RX'
              ELSE ''
            END) = 'RX' THEN c.RX_COMP_WK
      ELSE c.FS_COMP_WK
    END AS COMP_FLAG,
    s.STRNUM,
    b.XREF2_HD AS SOURCE_CODE,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM CV_BASE_FIN_ACDOCA_S4_For_Manual_JE j
  INNER JOIN CV_BASE_FIN_BKPF_S4_For_Manual_JE b
    ON j.RBUKRS = b.BUKRS
    AND j.BELNR = b.BELNR
    AND j.GJAHR = b.GJAHR
  INNER JOIN CV_BASE_MD_HRRP_NODE_S4_Filtered h1
    ON j.PRCTR = h1.NODEVALUE
    AND j.RCLNT = h1.MANDT
    AND h1.PARNODE LIKE '%CORE_RET%'
    AND h1.HRYVALTO = '99991231'
  INNER JOIN CV_BASE_MD_HRRP_NODE_S4_Filtered h2
    ON j.RACCT = h2.NODEVALUE
    AND j.RCLNT = h2.MANDT
    AND h2.PARNODE LIKE '%181'
    AND h2.HRYVALTO = '99991231'
    AND h2.HRYID = 'CVS2'
  INNER JOIN CV_BASE_MD_HRRP_NODE_S4_Filtered h3
    ON j.WWPC_PA = h3.NODEVALUE
    AND j.RCLNT = h3.MANDT
    AND (h3.PARNODE LIKE '%FS' OR h3.PARNODE LIKE '%RX')
    AND h3.HRYVALTO = '99991231'
    AND h3.HRYID LIKE '%ZPRODUCTCLASS'
  LEFT JOIN CV_BASE_MD_SRPACT_S4_Filtered s
    ON j.PRCTR = s.PRCTR
    AND j.RCLNT = s.MANDT
  LEFT JOIN CV_BASE_MD_COMPFL_S4_Filtered c
    ON j.PRCTR = c.PRCTR
    AND j.ZZWEEK = c.ZWEEK
    AND j.RCLNT = c.MANDT
  LEFT JOIN CV_BASE_MD_CEPCT_S4_Filtered p
    ON j.PRCTR = p.PRCTR
    AND j.RCLNT = p.MANDT
  LEFT JOIN CV_BASE_MD_RCALWEEK_S4_Filtered w
    ON j.ZZWEEK = w.ZZWEEK
    AND j.RCLNT = w.RCLNT
  WHERE b.XREF2_HD NOT IN ('RT', 'RV', 'SK')
),

-- CTE 17: SKF_MANUAL_JE_Joined
-- Logic: SKF manual journal entries with hierarchy filters and all joins
-- Source: INNER JOIN of FAGLSKF and hierarchy + LEFT JOINs to master data
-- Join: INNER JOIN on hierarchy (CORERET), LEFT JOINs for store attributes, comp flags, calendar, profit center text
-- Filters: HIER PARNODE LIKE '%CORERET%', HRYVALTO = '99991231'

  SELECT
    s.EMERG_MKT_IND,
    s.DISTRICT_DESC,
    s.DISTRICT_CODE,
    s.REGION_DESC,
    s.REGION_CODE,

    s.STATE,

    s.REP_MKT_DESC,

    f.ZZWEEK,
    f.PRCTR,
    f.DATE_FROM,
    f.STAGR,
    f.RCLNT,
    c.FS_COMP_WK,
    c.RX_COMP_WK,
    w.ZRWSTRTDATE,
    w.ZRWENDDATE,
    CASE WHEN f.STAGR = '801000' THEN f.MSL ELSE f.MSL * 2 END AS MSL,
    f.RCNTR,
    p.PROFIT_CENTER_TEXT,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    RIGHT(CAST(f.ZZWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,

    s.STRNUM,
    f.ZZSCODE AS SOURCE_CODE,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM CV_BASE_SKF_FAGLSKF_PN_S4_Filtered f
  INNER JOIN CV_BASE_MD_HRRP_NODE_S4_Filtered h
    ON f.RCNTR = h.NODEVALUE
    AND h.PARNODE LIKE '%CORERET%'
    AND h.HRYVALTO = '99991231'
  LEFT JOIN CV_BASE_MD_SRPACT_S4_Filtered s
    ON f.PRCTR = s.PRCTR
  LEFT JOIN CV_BASE_MD_COMPFL_S4_Filtered c
    ON f.PRCTR = c.PRCTR
    AND f.ZZWEEK = c.ZWEEK
  LEFT JOIN CV_BASE_MD_RCALWEEK_S4_Filtered w
    ON f.ZZWEEK = w.ZZWEEK
  LEFT JOIN CV_BASE_MD_CEPCT_S4_Filtered p
    ON f.PRCTR = p.PRCTR
),

    ELSE ''
  END AS CAL_COMP_FLAG_DESC,

  CAL_SOURCE,
