SOURCE_LOGIC:
- The Calculation View is a CUBE with outputViewType="Aggregation", so the final output requires GROUP BY and aggregation.
- Data flow:
    1. FAGLSKF_PN_S4: Projects and filters rows from CV_BASE_SKF_FAGLSKF_PN_S4:
        - Filters: STAGR IN ('801000', '804013'), ZZWEEK BETWEEN $$IP_WEEK_ENDING_FROM$$ AND $$IP_WEEK_ENDING_TO$$, ZZSCODE != 'SLS'
    2. HIER: Projects from CV_BASE_MD_HRRP_NODE_S4, filtered by PARNODE matching '*CORERET*' and HRYVALTO = '99991231'
    3. Join_7: INNER JOIN FAGLSKF_PN_S4 and HIER ON RCNTR = NODEVALUE
    4. STORE_ATTR_SKF_JOURNAL_ENT: Projects from CV_BASE_MD_SRPACT_S4 (attributes as above)
    5. Join_2: LEFT OUTER JOIN Join_7 and STORE_ATTR_SKF_JOURNAL_ENT ON PRCTR
    6. COMP_FLAG_SKF_JOURNAL_ENT: Projects from CV_BASE_MD_COMPFL_S4, filtered by COMP_VER = $$IP_VERSION_COMP_FLAG$$
    7. Join_3: LEFT OUTER JOIN Join_2 and COMP_FLAG_SKF_JOURNAL_ENT ON PRCTR, ZZWEEK=ZWEEK
    8. CAL_WEEK: Projects from CV_BASE_MD_RCALWEEK_S4 (attributes as above)
    9. Join_6: LEFT OUTER JOIN Join_3 and CAL_WEEK ON ZZWEEK
    10. PROFIT_CENTER_TEXT: Projects from CV_BASE_MD_CEPCT_S4, mapping LTEXT as PROFIT_CENTER_TEXT
    11. Join_8: LEFT OUTER JOIN Join_6 and PROFIT_CENTER_TEXT ON PRCTR
    12. SKF_QTY: Final projection with calculated columns CAL_MSL, CAL_WEEK_NUMBER, CAL_COMP_FLAG.
- Output columns and measure: All columns from SKF_QTY node, with MSL as SUM(CAL_MSL).
- Final output: GROUP BY all non-measure output columns, aggregate CAL_MSL as SUM(MSL).

WITH FAGLSKF_PN_S4 AS (
  SELECT
    RCLNT,
    STAGR,
    DATE_FROM,
    MSL,
    PRCTR,
    ZZSCODE,
    ZZWEEK,
    RCNTR
  FROM PROJECT.DATASET.CV_BASE_SKF_FAGLSKF_PN_S4
  WHERE STAGR IN ('801000', '804013')
    AND ZZWEEK >= @IP_WEEK_ENDING_FROM
    AND ZZWEEK <= @IP_WEEK_ENDING_TO
    AND ZZSCODE != 'SLS'
),
HIER AS (
  SELECT
    MANDT,
    PARNODE,
    HRYVALTO,
    NODEVALUE
  FROM PROJECT.DATASET.CV_BASE_MD_HRRP_NODE_S4
  WHERE REGEXP_CONTAINS(PARNODE, r'.*CORERET.*')
    AND HRYVALTO = '99991231'
),
Joined_FAGLSKF_with_HIER AS (
  SELECT
    F.RCLNT,
    F.STAGR,
    F.DATE_FROM,
    F.PRCTR,
    F.ZZWEEK,
    F.MSL,
    H.NODEVALUE AS RCNTR,
    F.ZZSCODE AS SOURCE_CODE
  FROM FAGLSKF_PN_S4 F
  INNER JOIN HIER H
    ON F.RCNTR = H.NODEVALUE
),
STORE_ATTR_SKF_JOURNAL_ENT AS (
  SELECT
    MANDT,
    STRNUM,
    PRCTR,
    REP_MKT_CODE,

    CITY,

    DIVISION_CODE,
    DIVISION_DESC,
    AREA_CODE,

    REGION_CODE,

    DISTRICT_DESC,
    EMERG_MKT_IND,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE
  FROM PROJECT.DATASET.CV_BASE_MD_SRPACT_S4
),

  SELECT
    S.REP_MKT_CODE,

    S.CITY,
    S.STATE,
    S.DIVISION_CODE,
    S.DIVISION_DESC,
    S.AREA_CODE,
    S.AREA_DESC,
    S.REGION_CODE,
    S.REGION_DESC,

    S.DISTRICT_DESC,
    S.EMERG_MKT_IND,
    F.RCLNT,
    F.STAGR,
    F.DATE_FROM,
    F.PRCTR,
    F.ZZWEEK,
    F.MSL,
    F.RCNTR,
    S.FS_OPEN_DAT,
    S.RX_OPEN_DAT,
    S.STRNUM,
    F.SOURCE_CODE,
    S.RX_DIVISION_CODE,
    S.RX_AREA_CODE,
    S.RX_REGION_CODE,
    S.RX_DISTRICT_CODE
  FROM Joined_FAGLSKF_with_HIER F
  LEFT JOIN STORE_ATTR_SKF_JOURNAL_ENT S
    ON F.PRCTR = S.PRCTR
),
COMP_FLAG_SKF_JOURNAL_ENT AS (
  SELECT
    FS_COMP_WK,
    RX_COMP_WK,
    MANDT,
    PRCTR,
    STRNUM,
    COMP_VER,
    ZWEEK,
    ZYEAR,
    ZMONTH
  FROM PROJECT.DATASET.CV_BASE_MD_COMPFL_S4
  WHERE COMP_VER = @IP_VERSION_COMP_FLAG
),
Joined_StoreAttr_with_CompFlag AS (
  SELECT
    J.*,
    C.FS_COMP_WK,
    C.RX_COMP_WK
  FROM Joined_FAGLSKF_HIER_with_STORE_ATTR J
  LEFT JOIN COMP_FLAG_SKF_JOURNAL_ENT C
    ON J.PRCTR = C.PRCTR
    AND J.ZZWEEK = C.ZWEEK

CAL_WEEK AS (

    ZZWEEK,

  LEFT JOIN CAL_WEEK C

  LEFT JOIN PROFIT_CENTER_TEXT P
