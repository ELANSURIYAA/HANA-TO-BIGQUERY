WITH BKPF AS (
  SELECT
    MANDT,
    BUKRS,
    BELNR,
    GJAHR,
    XREF2_HD,
    BUDAT,
    BLART
  FROM PROJECT.DATASET.CV_BASE_FIN_BKPF_S4
  WHERE XREF2_HD = 'A1'
    AND BLART = 'ZT'
),
ACDOCA AS (
  SELECT
    RCLNT,
    RBUKRS,
    BELNR,
    GJAHR,
    WSL,
    PRCTR,
    RACCT,
    ZZWEEK,
    RWCUR
  FROM PROJECT.DATASET.CV_BASE_FIN_ACDOCA_S4
  WHERE PRCTR IN ('0000562057', '0000562058')
    AND RACCT = '0004403000'
    AND ZZWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
),
SBT AS (
  SELECT
    BKPF.MANDT,
    BKPF.BUKRS,
    BKPF.BELNR,
    BKPF.GJAHR,
    BKPF.XREF2_HD,
    BKPF.BUDAT,
    ACDOCA.WSL,
    ACDOCA.PRCTR,
    ACDOCA.RACCT,
    ACDOCA.ZZWEEK,
    ACDOCA.RWCUR
  FROM BKPF
  INNER JOIN ACDOCA
    ON BKPF.BUKRS = ACDOCA.RBUKRS
    AND BKPF.BELNR = ACDOCA.BELNR
    AND BKPF.GJAHR = ACDOCA.GJAHR
),
STORE_ATTRIBUTES AS (
  SELECT
    MANDT,
    STRNUM,
    PRCTR,
    REP_MKT_CODE,
    REP_MKT_DESC,
    CITY,
    STATE,

    RX_DISTRICT_CODE
  FROM PROJECT.DATASET.CV_BASE_MD_SRPACT_S4
),
Join_1 AS (
  SELECT
    SBT.MANDT,
    SBT.BUKRS,
    SBT.BELNR,
    SBT.GJAHR,
    SBT.XREF2_HD,
    SBT.BUDAT,
    SBT.PRCTR,
    SBT.WSL,
    SBT.ZZWEEK,
    SBT.RWCUR,
    SA.REP_MKT_CODE,
    SA.REP_MKT_DESC,
    SA.CITY,
    SA.STATE,
    SA.DIVISION_CODE,
    SA.DIVISION_DESC,
    SA.AREA_CODE,
    SA.AREA_DESC,
    SA.REGION_CODE,

    SA.DISTRICT_CODE,

    SA.STRNUM,

    SA.EMERG_MKT_IND,
    SA.RX_DIVISION_CODE,
    SA.RX_AREA_CODE,
    SA.RX_REGION_CODE,
    SA.RX_DISTRICT_CODE

  LEFT OUTER JOIN STORE_ATTRIBUTES AS SA
    ON SBT.PRCTR = SA.PRCTR
),
RETAIL_CALENAR AS (
  SELECT
    ZZWEEK,
    ZRWENDDATE,
    ZRWSTRTDATE,
    RCLNT
  FROM PROJECT.DATASET.CV_BASE_MD_RCALWEEK_S4
),
Join_2 AS (
  SELECT
    J1.*,
    RC.ZRWENDDATE,
    RC.ZRWSTRTDATE
  FROM Join_1 AS J1
  LEFT OUTER JOIN RETAIL_CALENAR AS RC
    ON J1.ZZWEEK = RC.ZZWEEK

COMP_FLAG AS (
  SELECT
    *,
    CASE WHEN PRCTR = '0000562057' THEN '1' ELSE '0' END AS CAL_COMP_FLAG,
    RIGHT(CAST(ZZWEEK AS STRING), 2) AS CAL_WEEK_NUMBER

  FROM PROJECT.DATASET.CV_BASE_MD_CEPCT_S4
