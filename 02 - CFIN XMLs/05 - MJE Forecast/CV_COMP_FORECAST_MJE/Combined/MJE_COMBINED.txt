    district_desc,

    region_desc,
    region_code,
    division_desc,
    division_code,
    rx_comp_wk                   AS comp_flag,
    bic_zio_sweek,
    rep_mkt_desc,
    rep_mkt_code,
    emerg_mkt_ind,
    b631_s_proftctr              AS prctr,
    mandt,
    'RX'                         AS fs_rx_flag_ext,
    zrwstrtdate,
    zrwenddate,
    cal_week_number,
    city,
    state,
    CAST(NULL AS STRING)         AS forecast_currency,
    profit_center_text,
    'SKF'                        AS cal_data_category1,
    'SKF_FCST'                   AS cal_data_category2,
    fs_open_dat,
    rx_open_dat,
    CAST(NULL AS NUMERIC)        AS wsl,
    CAST(NULL AS NUMERIC)        AS cal_fs_sales,
    CAST(NULL AS NUMERIC)        AS cal_rx_sales,
    CAST(NULL AS NUMERIC)        AS msl,
    area_code,

     'FS'                         AS fs_rx_flag,

-- Logic: Projects all columns from COMBINED, renaming forecast_currency
--        to fin_forecast_currency, wsl to sbt, msl to skf_mje_qty,
--        cal_fs_sales to cal_fs_sales_mje, and cal_rx_sales to
--        cal_rx_sales_mje, as required by the final output contract.
-- Source: COMBINED
-- =====================================================================

    region_code,
    region_desc,

    prctr,

    rep_mkt_desc,
    comp_flag,
    fs_rx_flag,
    bic_zio_sweek,
    fs_rx_flag_ext,
    zrwstrtdate,
    zrwenddate,
    cal_week_number,
    city,
    state,
    forecast_currency            AS fin_forecast_currency,
    fin_fcst_amount,
    skf_fcst_qty,
    profit_center_text,
    cal_data_category1,
    cal_data_category2,

    rx_open_dat,

    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code
  FROM COMBINED

-- =====================================================================
-- Final SELECT
-- Logic: Aggregates all measure columns using SUM, groups by all
--        dimension columns, and computes two calculated attributes
--        (cal_comp_flag_desc, cal_emerg_mkt_ind) as CASE expressions
--        applied after aggregation, per HANA CUBE/Aggregation semantics.
-- Source: VARIANCES
-- Aggregation: SUM(fin_fcst_amount), SUM(skf_fcst_qty), SUM(sbt),
--              SUM(skf_mje_qty), SUM(cal_fs_sales_mje), SUM(cal_rx_sales_mje)
-- Calculation: cal_comp_flag_desc via CASE on comp_flag;
--              cal_emerg_mkt_ind via CASE on emerg_mkt_ind
-- =====================================================================
SELECT
  mandt,
  cal_source,
  division_code,
  division_desc,
  region_code,
  region_desc,
  district_code,
  district_desc,
  strnum,
  prctr,
  emerg_mkt_ind,
  rep_mkt_code,
  rep_mkt_desc,
  comp_flag,
  fs_rx_flag,
  bic_zio_sweek,
  fs_rx_flag_ext,
  zrwstrtdate,
  zrwenddate,
  cal_week_number,
  city,
  state,
  fin_forecast_currency,
  profit_center_text,
  cal_data_category1,
  cal_data_category2,
  fs_open_dat,
  rx_open_dat,
  area_code,
  area_desc,
  source_code,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code,
  SUM(fin_fcst_amount)           AS fin_fcst_amount,
  SUM(skf_fcst_qty)              AS skf_fcst_qty,
  SUM(sbt)                       AS sbt,
  SUM(skf_mje_qty)               AS skf_mje_qty,
  SUM(cal_fs_sales_mje)          AS cal_fs_sales_mje,
  SUM(cal_rx_sales_mje)          AS cal_rx_sales_mje,
  CASE comp_flag
    WHEN '1' THEN 'Comp'
    WHEN '0' THEN 'Non-Comp'
    ELSE ''
  END                            AS cal_comp_flag_desc,
  CASE
    WHEN emerg_mkt_ind = 'Y' THEN 'Yes'
    WHEN emerg_mkt_ind = 'N' THEN 'No'
    ELSE ''
  END                            AS cal_emerg_mkt_ind
FROM VARIANCES
GROUP BY
  mandt,
  cal_source,
  division_code,
  division_desc,
  region_code,
  region_desc,
  district_code,
  district_desc,
  strnum,
  prctr,
  emerg_mkt_ind,
  rep_mkt_code,
  rep_mkt_desc,
  comp_flag,
  fs_rx_flag,
  bic_zio_sweek,
  fs_rx_flag_ext,
  zrwstrtdate,
  zrwenddate,
  cal_week_number,
  city,
  state,
  fin_forecast_currency,
  profit_center_text,
  cal_data_category1,
  cal_data_category2,
  fs_open_dat,
  rx_open_dat,
  area_code,
  area_desc,
  source_code,
  rx_division_code,
  rx_area_code,
  rx_region_code,
  rx_district_code
