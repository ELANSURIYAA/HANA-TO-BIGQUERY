WITH WEEKDATES AS (
    SELECT
        CAST(CURR.RCLNT AS STRING) AS MANDT,
        CAST(PREV.ZZWEEK AS STRING) AS PREV_WEEK_NBR
    FROM `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek` AS CURR
    LEFT JOIN
        `edp-qa-finstorage.edp_saps4_shared_cnsv.v_ztfigl_rcalweek` AS PREV
        ON CAST(PREV.RCLNT AS STRING) = CAST(CURR.RCLNT AS STRING)
       AND CAST(PREV.ZRWENDDATE AS DATE) =
           DATE_SUB(
               CAST(CURR.ZRWSTRTDATE AS DATE),
               INTERVAL 1 DAY
           )
    WHERE
        CURRENT_DATE('America/New_York')
        BETWEEN CAST(CURR.ZRWSTRTDATE AS DATE)
            AND CAST(CURR.ZRWENDDATE AS DATE)
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY CAST(CURR.RCLNT AS STRING)
        ORDER BY
            CAST(CURR.ZRWSTRTDATE AS DATE) DESC,
            CAST(CURR.ZZWEEK AS STRING) DESC
    ) = 1
),
