  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,

    MANDT,
    RECORDMODE,

    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    SUM(_BIC_ZIO_SKFQ) AS _BIC_ZIO_SKFQ,
    SUM(_BIC_ZIO_D1SKQ) AS _BIC_ZIO_D1SKQ,
    SUM(_BIC_ZIO_D2SKQ) AS _BIC_ZIO_D2SKQ,

    SUM(_BIC_ZIO_D6SKQ) AS _BIC_ZIO_D6SKQ,
    SUM(_BIC_ZIO_D7SKQ) AS _BIC_ZIO_D7SKQ
  FROM
    `PROJECT.DATASET.AZSRP_DS112_VT_S4`
  WHERE
    MANDT IN (110, 200)
    AND FISCVARNT = 'K4'
    AND _BIC_ZIO_SWEEK = IP_WEEK_LY
    AND _BIC_ZIO_VER = IP_VERSION
    AND _BIC_ZIO_SAUDT IN ('1','10')
    AND _BIC_ZIO_PSKF IN ('804013','801000')

    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,

    UNIT,
    _BIC_ZIO_SWEEK,
