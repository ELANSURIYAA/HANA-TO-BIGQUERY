-- =====================================================================
-- BIGQUERY SQL CONSOLIDATION: CV_CONS_WEEKLY_FLASH_REPORT_STATIC
-- =====================================================================
-- 
-- CONSOLIDATION SUMMARY:
-- This consolidated SQL represents the complete end-to-end logic for the
-- weekly flash sales reporting system, tracing from base tables through
-- all intermediate calculation views to the final consumer-facing report.
--
-- LINEAGE FLOW:
-- Base Tables → Base Calculation Views → Composite Views →

-- DECLARE PARAMETERS (for procedure execution context)
DECLARE IP_WEEK_ENDING_FROM STRING DEFAULT '20240101';
DECLARE IP_WEEK_ENDING_TO STRING DEFAULT '20240107';
DECLARE IP_UPD_TIMESTAMP_FROM STRING DEFAULT '20240101000000';
DECLARE IP_UPD_TIMESTAMP_TO STRING DEFAULT '20240107235959';
DECLARE IP_VERSION STRING DEFAULT '1';

WITH

-- =====================================================================
-- BASE LAYER: NAVIX (Store Master Data)
-- =====================================================================
BASE_NAVIX AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CURRENCY
  FROM
    `PROJECT.DATASET.POSDW_NAVIX`
),

-- =====================================================================
-- BASE LAYER: TLOGF (Transaction Log - Front Store)
-- =====================================================================
BASE_TLOGF AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    SUM(SALESAMOUNT) AS SALESAMOUNT,
    SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY
  FROM
    `PROJECT.DATASET.POSDW_TLOGF`
  WHERE
    RECORDQUALIFIER IN (5, 6)
    AND TRANSTYPECODE NOT IN (1107, 1197, 1020)
    AND WORKSTATIONID != '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP != 0
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY
),

-- =====================================================================
-- BASE LAYER: TLOGF_X (Transaction Log Extended - Scripts)
-- =====================================================================
BASE_TLOGF_X AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    WORKSTATIONID,
    ARCHIVED,
    ZZ_UPD_TIMESTAMP,
    ZZ_CUSTTYPE,
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE
  FROM
    `PROJECT.DATASET.POSDW_TLOGF_X`
  WHERE
    RECORDQUALIFIER = 25
    AND WORKSTATIONID != '0000000000'
    AND ZZ_UPD_TIMESTAMP != '0'
),

-- =====================================================================
-- BASE LAYER: TLOGF_COVID (COVID Sales)
-- =====================================================================
BASE_TLOGF_COVID AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    SUM(SALESAMOUNT) AS SALESAMOUNT,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY,
    ITEMID
  FROM
    `PROJECT.DATASET.POSDW_TLOGF`
  WHERE
    RECORDQUALIFIER IN (5, 6)
    AND TRANSTYPECODE NOT IN (1107, 1197, 1020)
    AND WORKSTATIONID != '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP != '0'
    AND ITEMID = 'A01-433556'
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY,
    ITEMID
),

-- =====================================================================
-- BASE LAYER: PARAMETERS (Flash Parameters)
-- =====================================================================
BASE_PARAMETERS AS (
  SELECT
    MANDT,
    PARAM_NAME,
    PARAM_VALUE_LOW AS LOW_CHAR,
    PARAM_VALUE_HIGH AS HIGH_CHAR
  FROM
    `PROJECT.DATASET.ZTFIRP_FLASH_PRM`
),

-- =====================================================================
-- BASE LAYER: RETAIL CALENDAR (S4 Calendar Week Master)
-- =====================================================================
BASE_MD_RCALWEEK_S4 AS (
  SELECT
    RCLNT,
    ZZWEEK,
    ZCALYRP,
    ZRYEAR,
    ZRPERIOD,
    ZRYRP,
    ZRYRQTR,
    ZRWSTRTDATE,
    ZRWENDDATE
  FROM
    `PROJECT.DATASET.ZTFIGL_RCALWEEK`
  WHERE
    RCLNT IN (120, 200)
),

-- =====================================================================
-- PARAMETER FILTERS
-- =====================================================================
FS_RETAIL_TYPES AS (
  SELECT
    PARAM_NAME,
    LOW_CHAR
  FROM BASE_PARAMETERS
  WHERE PARAM_NAME = 'FS_RETAIL_TYPE_CODE'
),

FS_DISC_TYPES AS (
  SELECT
    PARAM_NAME,
    LOW_CHAR
  FROM BASE_PARAMETERS
  WHERE PARAM_NAME = 'FS_DISC_TYPE_CODE'
),

  SELECT
    PARAM_NAME,

  WHERE PARAM_NAME = 'EMP_DISC_TYPE_CODE'
),

    PARAM_NAME,
    LOW_CHAR

  WHERE PARAM_NAME = 'RX_RETAIL_TYPE_CODE'
),

RX_RETAIL_TYPES_COVID AS (

),

-- =====================================================================
-- COMPOSITE LAYER: FLASH SALES VT CAR
-- =====================================================================
FS_SALES AS (

    t.MANDT,
    t.RETAILSTOREID,

  INNER JOIN FS_RETAIL_TYPES p
    ON t.RETAILTYPECODE = p.LOW_CHAR
  WHERE t.RECORDQUALIFIER = 5
),

FS_DISCOUNT AS (
  SELECT
    t.MANDT,
    t.RETAILSTOREID,
    t.BUSINESSDAYDATE,
    t.RECORDQUALIFIER,

  INNER JOIN FS_DISC_TYPES p
    ON t.DISCTYPECODE = p.LOW_CHAR
  WHERE t.RECORDQUALIFIER = 6
),

EMP_DISCOUNT AS (
  SELECT
    t.MANDT,
    t.RETAILSTOREID,
    t.BUSINESSDAYDATE,
    t.RECORDQUALIFIER,

    ON t.DISCTYPECODE = p.LOW_CHAR
  WHERE t.RECORDQUALIFIER = 6
),

RX_SALES AS (
  SELECT
    t.MANDT,
    t.RETAILSTOREID,
    t.BUSINESSDAYDATE,
    t.RECORDQUALIFIER,

  INNER JOIN RX_RETAIL_TYPES p
    ON t.RETAILTYPECODE = p.LOW_CHAR
  WHERE t.RECORDQUALIFIER = 5

  SELECT
    t.MANDT,
    t.RETAILSTOREID,
    t.BUSINESSDAYDATE,

SCRIPTS AS (
  SELECT
    s.MANDT,
    s.RETAILSTOREID,
    s.BUSINESSDAYDATE,
    s.ZZ_RX_CNT_NS,
    s.ZZ_RX_CNT_REFILL,
    s.ZZ_RX_CNT_GE84_NS,
    s.ZZ_RX_CNT_GE84_RE,
    s.ZZ_RX_MCRX_GE84_NS,
    s.ZZ_RX_MCRX_GE84_RE,
    s.ZZ_CUSTTYPE,
    n.CURRENCY,
    s.ZZ_UPD_TIMESTAMP AS SCRIPTS_UPD_TIMESTAMP,
    s.WORKSTATIONID,
    CASE 
      WHEN s.ZZ_CUSTTYPE IN ('F','M','R') THEN 1 
      WHEN s.ZZ_CUSTTYPE IN ('X','Y','Z') THEN -1 
      ELSE 0 
    END AS CAL_FS_UNITS,
    CASE 
      WHEN s.ZZ_RX_CNT_GE84_NS = '' THEN 0 
      ELSE CAST(s.ZZ_RX_CNT_GE84_NS AS INT64) 
    END AS CAL_RX_CNT_GE84_NS,
    CASE 
      WHEN s.ZZ_RX_CNT_GE84_RE = '' THEN 0 
      ELSE CAST(s.ZZ_RX_CNT_GE84_RE AS INT64) 
    END AS CAL_RX_CNT_GE84_RE,
    CASE 
      WHEN s.ZZ_RX_CNT_NS = '' THEN 0 
      ELSE CAST(s.ZZ_RX_CNT_NS AS INT64) 
    END AS CAL_RX_CNT_NS,
    CASE 
      WHEN s.ZZ_RX_CNT_REFILL = '' THEN 0 
      ELSE CAST(s.ZZ_RX_CNT_REFILL AS INT64) 
    END AS CAL_RX_CNT_RE,
    CASE 
      WHEN s.WORKSTATIONID = '0000000555' THEN 
        CASE 
          WHEN s.ZZ_CUSTTYPE = 'C' THEN 1 
          WHEN s.ZZ_CUSTTYPE = 'W' THEN -1 
          ELSE 0 
        END 
      ELSE 0 
    END AS CAL_CVD_UNITS
  FROM BASE_TLOGF_X s
  INNER JOIN BASE_NAVIX n
    ON s.MANDT = n.MANDT
    AND s.RETAILSTOREID = n.RETAILSTOREID
    AND s.BUSINESSDAYDATE = n.BUSINESSDAYDATE
),

    NULL AS CAL_RX_CNT_GE84_NS_CALC,
    NULL AS CAL_RX_CNT_GE84_RE_CALC,
    FS_SALESAMOUNT,
    NULL AS EMP_REDUCTIONAMOUNT,
    NULL AS RX_SALESAMOUNT,
    NULL AS REDUCTIONAMOUNT,
    NULL AS CAL_RX_CNT_NS_CALC,
    NULL AS CAL_RX_CNT_RE_CALC,
    TRANSCURRENCY,
    FS_UPD_TIMESTAMP,
    NULL AS RX_UPD_TIMESTAMP,

    FS_UPD_TIMESTAMP AS ZZ_UPD_TIMESTAMP,

    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    NULL AS ZZ_RX_CNT_NS,
    NULL AS ZZ_RX_CNT_REFILL,
    NULL AS ZZ_RX_CNT_GE84_NS,
    NULL AS ZZ_RX_CNT_GE84_RE,
    NULL AS ZZ_RX_MCRX_GE84_NS,
    NULL AS ZZ_RX_MCRX_GE84_RE,
    NULL AS CAL_FS_UNITS,
    NULL AS CAL_RX_CNT_GE84_NS_CALC,
    NULL AS CAL_RX_CNT_GE84_RE_CALC,
    NULL AS FS_SALESAMOUNT,
    EMP_REDUCTIONAMOUNT,
    NULL AS RX_SALESAMOUNT,
    NULL AS REDUCTIONAMOUNT,
    NULL AS CAL_RX_CNT_NS_CALC,
    NULL AS CAL_RX_CNT_RE_CALC,
    TRANSCURRENCY,
    NULL AS FS_UPD_TIMESTAMP,
    NULL AS RX_UPD_TIMESTAMP,
    NULL AS SCRIPTS_UPD_TIMESTAMP,
    EMP_DISC_UPD_TIMESTAMP AS ZZ_UPD_TIMESTAMP,
    NULL AS CVD_AMOUNT,
    NULL AS CAL_CVD_UNITS
  FROM EMP_DISCOUNT

  UNION ALL

  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    NULL AS ZZ_RX_CNT_NS,
    NULL AS ZZ_RX_CNT_REFILL,
    NULL AS ZZ_RX_CNT_GE84_NS,
    NULL AS ZZ_RX_CNT_GE84_RE,
    NULL AS ZZ_RX_MCRX_GE84_NS,
    NULL AS ZZ_RX_MCRX_GE84_RE,

    NULL AS CAL_RX_CNT_GE84_RE_CALC,

    RX_SALESAMOUNT,

    NULL AS ZZ_RX_CNT_GE84_RE,
    NULL AS ZZ_RX_MCRX_GE84_NS,
    NULL AS ZZ_RX_MCRX_GE84_RE,
    NULL AS CAL_FS_UNITS,
    NULL AS CAL_RX_CNT_GE84_NS_CALC,
    NULL AS CAL_RX_CNT_GE84_RE_CALC,
    NULL AS FS_SALESAMOUNT,
    NULL AS EMP_REDUCTIONAMOUNT,
    NULL AS RX_SALESAMOUNT,
    REDUCTIONAMOUNT,
    NULL AS CAL_RX_CNT_NS_CALC,
    NULL AS CAL_RX_CNT_RE_CALC,
    TRANSCURRENCY,
    NULL AS FS_UPD_TIMESTAMP,
    NULL AS RX_UPD_TIMESTAMP,
    NULL AS SCRIPTS_UPD_TIMESTAMP,
    FS_DISC_UPD_TIMESTAMP AS ZZ_UPD_TIMESTAMP,

  SELECT
    MANDT,
    RETAILSTOREID,

-- =====================================================================
-- AGGREGATE COMBINED DATA
-- =====================================================================
COMBINE_DATA AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    SUM(FS_SALESAMOUNT) AS FS_SALESAMOUNT,
    SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT,
    SUM(RX_SALESAMOUNT) AS RX_SALESAMOUNT,
    SUM(CAL_FS_UNITS) AS CAL_FS_UNITS,
    SUM(EMP_REDUCTIONAMOUNT) AS EMP_REDUCTIONAMOUNT,
    SUM(CAL_RX_CNT_GE84_NS_CALC) AS CAL_RX_CNT_GE84_NS,
    SUM(CAL_RX_CNT_GE84_RE_CALC) AS CAL_RX_CNT_GE84_RE,
    SUM(CAL_RX_CNT_NS_CALC) AS CAL_RX_CNT_NS,
    SUM(CAL_RX_CNT_RE_CALC) AS CAL_RX_CNT_RE,
    TRANSCURRENCY,
    MAX(FS_UPD_TIMESTAMP) AS FS_UPD_TIMESTAMP,
    MAX(RX_UPD_TIMESTAMP) AS RX_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP) AS SCRIPTS_UPD_TIMESTAMP,
    ZZ_UPD_TIMESTAMP,
    SUM(CVD_AMOUNT) AS CVD_AMOUNT,
    SUM(CAL_CVD_UNITS) AS CAL_CVD_UNITS
  FROM UNION_SALES
  GROUP BY 
    MANDT, 
    RETAILSTOREID, 
    BUSINESSDAYDATE, 
    TRANSCURRENCY, 
    ZZ_UPD_TIMESTAMP
),

-- =====================================================================
-- FLASH SALES VT CAR (Final aggregated with parameters)
-- =====================================================================
FLASH_SALES_VT_CAR AS (
  SELECT
    n.MANDT,
    n.RETAILSTOREID,
    n.BUSINESSDAYDATE,
    c.FS_SALESAMOUNT,
    c.REDUCTIONAMOUNT,
    c.RX_SALESAMOUNT,
    c.CAL_FS_UNITS,
    c.EMP_REDUCTIONAMOUNT,
    c.CAL_RX_CNT_GE84_NS,
    c.CAL_RX_CNT_GE84_RE,
    c.CAL_RX_CNT_NS,
    c.CAL_RX_CNT_RE,
    c.TRANSCURRENCY,
    c.RX_UPD_TIMESTAMP,
    c.FS_UPD_TIMESTAMP,
    c.SCRIPTS_UPD_TIMESTAMP,
    c.ZZ_UPD_TIMESTAMP,
    c.CVD_AMOUNT,
    c.CAL_CVD_UNITS,
    IP_UPD_TIMESTAMP_FROM AS CAL_UPD_TIMESTAMP_FROM,
    IP_UPD_TIMESTAMP_TO AS CAL_UPD_TIMESTAMP_TO
  FROM BASE_NAVIX n
  INNER JOIN COMBINE_DATA c
    ON n.MANDT = c.MANDT
    AND n.RETAILSTOREID = c.RETAILSTOREID
    AND n.BUSINESSDAYDATE = c.BUSINESSDAYDATE
  WHERE
    c.ZZ_UPD_TIMESTAMP >= IP_UPD_TIMESTAMP_FROM
    AND c.ZZ_UPD_TIMESTAMP <= IP_UPD_TIMESTAMP_TO
),

-- =====================================================================
-- COMPOSITE LAYER: CV_COMP_FIN_FLASH
-- =====================================================================
FLASH_WITH_WEEK AS (
  SELECT
    f.MANDT,
    f.BUSINESSDAYDATE,
    f.FS_SALESAMOUNT,
    f.REDUCTIONAMOUNT,
    f.RX_SALESAMOUNT,
    f.CAL_FS_UNITS,
    f.EMP_REDUCTIONAMOUNT,
    f.CAL_RX_CNT_GE84_NS,
    f.CAL_RX_CNT_GE84_RE,
    f.RETAILSTOREID AS RETAILSTOREID_CAR,
    f.CAL_RX_CNT_NS,
    f.CAL_RX_CNT_RE,
    f.RX_UPD_TIMESTAMP,
    f.FS_UPD_TIMESTAMP,
    f.SCRIPTS_UPD_TIMESTAMP,
    f.ZZ_UPD_TIMESTAMP,
    f.TRANSCURRENCY,
    f.CVD_AMOUNT,
    f.CAL_CVD_UNITS,
    REPLACE(
      FORMAT_DATE('%Y%m%d',
        DATE_ADD(
          PARSE_DATE('%Y%m%d', f.BUSINESSDAYDATE),
          INTERVAL (5 - MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', f.BUSINESSDAYDATE)) + 5, 7)) DAY
        )
      ),
      '-', ''
    ) AS CAL_WEEK_ENDING_DATE
  FROM FLASH_SALES_VT_CAR f
  WHERE
    (f.RETAILSTOREID < '0000020000' OR f.RETAILSTOREID > '0000024999')
    AND (f.BUSINESSDAYDATE >= IP_WEEK_ENDING_FROM AND f.BUSINESSDAYDATE <= IP_WEEK_ENDING_TO)
),

WEEK_JOIN AS (
  SELECT
    rc.ZZWEEK,
    rc.ZRYEAR,
    rc.ZRPERIOD,
    rc.ZRYRP,
    rc.ZRWENDDATE,
    rc.ZRWSTRTDATE,
    f.MANDT,
    f.BUSINESSDAYDATE,
    f.FS_SALESAMOUNT,
    f.REDUCTIONAMOUNT,
    f.RX_SALESAMOUNT,
    f.CAL_FS_UNITS,
    f.EMP_REDUCTIONAMOUNT,
    f.CAL_RX_CNT_GE84_NS,
    f.CAL_RX_CNT_GE84_RE,
    f.CAL_WEEK_ENDING_DATE,
    f.CAL_RX_CNT_NS,
    f.CAL_RX_CNT_RE,
    f.RETAILSTOREID_CAR,
    f.RX_UPD_TIMESTAMP,
    f.FS_UPD_TIMESTAMP,
    f.SCRIPTS_UPD_TIMESTAMP,
    f.ZZ_UPD_TIMESTAMP,
    f.TRANSCURRENCY,
    f.CVD_AMOUNT,
    f.CAL_CVD_UNITS
  FROM FLASH_WITH_WEEK f
  LEFT JOIN BASE_MD_RCALWEEK_S4 rc
    ON f.CAL_WEEK_ENDING_DATE = rc.ZRWENDDATE
),

STORE_ATTR_JOIN AS (
  SELECT
    w.*,
    s.PRCTR,
    s.EMERG_MKT_IND,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,
    s.DIVISION_CODE,
    s.DIVISION_DESC,
    s.REGION_CODE,
    s.REGION_DESC,
    s.DISTRICT_CODE,
    s.DISTRICT_DESC,
    s.CITY,
    s.STATE,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    s.AREA_CODE,
    s.AREA_DESC,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE,
    w.RETAILSTOREID_CAR AS RETAILSTOREID
  FROM WEEK_JOIN w
  LEFT JOIN `PROJECT.DATASET.CV_BASE_MD_SRPACT_S4` s
    ON w.RETAILSTOREID_CAR = s.STRNUM
),

HIERARCHY_FILTER AS (
  SELECT
