    _BIC_ZWWSC_PA1,

      WHEN SUM(CASE WHEN FLAG = 'FC' THEN _B631_S_AMOUNT_DUMMY ELSE NULL END) IS NULL

    MAX(FLAG) AS FLAG

    FISCPER,

    FISCYEAR,
    FISCPER3,
