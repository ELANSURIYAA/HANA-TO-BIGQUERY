    S.RX_DISTRICT_CODE
  FROM
    PROJECTION_1 P
    LEFT JOIN STORE_ATTR_BUDGET S
      ON P._B631_S_PROFTCTR = S.PRCTR
      AND P.MANDT = S.MANDT

    R.ZRWSTRTDATE,
    R.ZRWENDDATE
  FROM
    JOIN_2 J
    LEFT JOIN RETAIL_CAL R
      ON J.CAL_WEEK = R.ZZWEEK

    AND ZWEEK <= @IP_WEEK_ENDING_TO_BUDGET  -- REQUIRES VALIDATION: Parameter assumed
    AND COMP_VER = @IP_VERSION_COMP_FLAG_BUDGET  -- REQUIRES VALIDATION: Parameter assumed
),

-- =====================================================================

-- =====================================================================
JOIN_4 AS (
  SELECT
    J.*,
    C.RX_COMP_WK
  FROM
    JOIN_1 J
    LEFT JOIN COMP_FLAG_BUDGET C
      ON J._B631_S_PROFTCTR = C.PRCTR
      AND J.MANDT = C.MANDT
),

    J.*,
    P.PROFIT_CENTER_TEXT
  FROM

    LEFT JOIN PROFIT_CENTER_TEXT P
      ON J._B631_S_PROFTCTR = P.PRCTR
      AND J.MANDT = P.MANDT

-- =====================================================================
