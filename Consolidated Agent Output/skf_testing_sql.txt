  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,

    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    RECORDMODE,
    UNIT,
    _BIC_ZIO_SKFQ
  FROM
    PROJECT.DATASET.AZSRP_DS092_VT_S4
  WHERE
    MANDT IN ('110', '200')
    AND @IP_FC_COUNT != '0'
),

    _BIC_ZIO_SKFQ
  FROM
    PROJECT.DATASET.AZSRP_DS031_VT_S4
  WHERE
    MANDT IN ('110', '200')
    AND @IP_FC_COUNT = '0'
),

-- Source: CV_BASE_SKF_WEEKLY_BUDGET_S4 -> Combined Frozen and Live Data
Combined_Frozen_Live_Data AS (
  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    RECORDMODE,
    UNIT,
    _BIC_ZIO_SKFQ,
    'FC' AS FLAG
  FROM
    Frozen_Cube
  UNION ALL
  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    '' AS _B631_S_CHRTACCT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _BIC_ZIO_PCTR AS _B631_S_PROFTCTR,
    _BIC_ZIO_CCTR AS _B631_S_COSTCNTR,
    '' AS _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    RECORDMODE,
    UNIT,
    _BIC_ZIO_SKFQ,
    'LC' AS FLAG
  FROM
    Live_Cube
),

-- Source: CV_BASE_SKF_WEEKLY_BUDGET_S4 -> Aggregated Weekly Budget
CV_BASE_SKF_WEEKLY_BUDGET_S4_Expanded AS (
  SELECT
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    RECORDMODE,
    UNIT,
    IF(
      SUM(CASE WHEN FLAG = 'FC' THEN _BIC_ZIO_SKFQ ELSE NULL END) IS NULL,
      SUM(CASE WHEN FLAG = 'LC' THEN _BIC_ZIO_SKFQ ELSE NULL END),
      SUM(CASE WHEN FLAG = 'FC' THEN _BIC_ZIO_SKFQ ELSE NULL END)
    ) AS _BIC_ZIO_SKFQ
  FROM
    Combined_Frozen_Live_Data
  GROUP BY
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    RECORDMODE,
    UNIT
),

-- =====================================================================
-- MAIN CALCULATION VIEW: CV_COMP_SKF_BUDGET_STATIC
-- =====================================================================

-- Step 1: Filter Weekly Snapshot
WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    RECORDMODE,
    _BIC_ZIO_SWEEK,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _BIC_ZIO_PSTY,
    _BIC_ZIO_PSKF,
    UNIT,
    _BIC_ZIO_SKFQ,
    'BUDGET' AS FLAG
  FROM CV_BASE_SKF_WEEKLY_BUDGET_S4_Expanded
  WHERE
    FISCVARNT = 'K4'
    AND _BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
    AND _BIC_ZIO_VER = @IP_VERSION
    AND _BIC_ZIO_SAUDT IN ('1', '10')
    AND _BIC_ZIO_PSKF IN ('804013', '801000')
),

-- Step 2: Filter Hierarchy Nodes
Filtered_Hierarchy_Nodes AS (
  SELECT
    MANDT,
    PARNODE,
    HRYVALTO,
    NODEVALUE,
    MANDT AS MANDT_1
  FROM CV_BASE_MD_HRRP_NODE_S4_Expanded
  WHERE
    REGEXP_CONTAINS(PARNODE, r'CORERET$')
    AND HRYVALTO = '99991231'
),

-- Step 3: Join Weekly Snapshot with Hierarchy
Joined_Weekly_Snapshot_With_Hierarchy AS (
  SELECT
    wsd.FISCPER,
    wsd.FISCVARNT,
    wsd.FISCYEAR,
    wsd.FISCPER3,
    wsd.MANDT,
    wsd.RECORDMODE,
    wsd._BIC_ZIO_SWEEK,
    wsd._B631_S_CO_AREA,
    wsd._BIC_ZIO_CMPCD,
    wsd._B631_S_PROFTCTR,
    hn.NODEVALUE AS _B631_S_COSTCNTR,
    wsd._BIC_ZIO_VER,
    wsd._BIC_ZIO_SAUDT,
    wsd._BIC_ZIO_PSTY,
    wsd._BIC_ZIO_PSKF,
    wsd.UNIT,
    wsd._BIC_ZIO_SKFQ,
    wsd.FLAG
  FROM WEEKLY_SNAPSHOT_DS05 wsd
  INNER JOIN Filtered_Hierarchy_Nodes hn
    ON wsd._B631_S_COSTCNTR = hn.NODEVALUE
),

-- Step 4: Prepare Store Attributes
STORE_ATTR_ACTUAL AS (
  SELECT
    MANDT,
    STRNUM,
    PRCTR,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,
    DIVISION_DESC,
    AREA_CODE,
    AREA_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    REGION_CODE,
    REGION_DESC,
    CITY,
    STATE,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE
  FROM CV_COMP_MD_SRPACT_STATIC_Expanded
),

-- Step 5: Join with Store Attributes
Joined_Weekly_With_Store_Attrs AS (
  SELECT
    j.*,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,
    s.EMERG_MKT_IND,
    s.STRNUM,
    s.DIVISION_CODE,
    s.DIVISION_DESC,
    s.AREA_CODE,
    s.AREA_DESC,
    s.DISTRICT_CODE,
    s.DISTRICT_DESC,
    s.REGION_CODE,
    s.REGION_DESC,
    s.CITY,
    s.STATE,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM Joined_Weekly_Snapshot_With_Hierarchy j
  LEFT JOIN STORE_ATTR_ACTUAL s
    ON j._B631_S_PROFTCTR = s.PRCTR
),

-- Step 6: Prepare Comp Flags
COMP_FLAG_BUDGET AS (
  SELECT
    MANDT,
    PRCTR,
    STRNUM,
    RX_COMP_WK,
    ZWEEK,
    ZMONTH,
    ZYEAR,
    COMP_VER
  FROM CV_COMP_MD_COMPFL_STATIC_Expanded
  WHERE COMP_VER = @IP_VERSION
),

-- Step 7: Join with Comp Flags
Joined_With_Comp_Flags AS (
  SELECT
    jw.*,
    c.RX_COMP_WK,
    c.MANDT AS MANDT_1
  FROM Joined_Weekly_With_Store_Attrs jw
  LEFT JOIN COMP_FLAG_BUDGET c
    ON jw._B631_S_PROFTCTR = c.PRCTR
    AND jw._BIC_ZIO_SWEEK = c.ZWEEK
),

-- Step 8: Prepare Calendar Week
CAL_WEEK AS (
  SELECT
    ZRWSTRTDATE,
    ZRWENDDATE,
    ZZWEEK,
    RCLNT
  FROM CV_BASE_MD_RCALWEEK_S4_Expanded
),

-- Step 9: Join with Calendar Week
Joined_With_Calendar_Week AS (
  SELECT
    j.*,
    cw.ZRWENDDATE,
    cw.ZRWSTRTDATE,
    cw.RCLNT
  FROM Joined_With_Comp_Flags j
  LEFT JOIN CAL_WEEK cw
    ON j._BIC_ZIO_SWEEK = cw.ZZWEEK
),

-- Step 10: Prepare Profit Center Text
PROFIT_CENTER_TEXT AS (
  SELECT
    MANDT,
    PRCTR,
    LTEXT AS PROFIT_CENTER_TEXT
  FROM CV_BASE_MD_CEPCT_S4_Expanded
),

-- Step 11: Join with Profit Center Text
Joined_With_Profit_Center_Text AS (
  SELECT
    jw.*,
    pct.PROFIT_CENTER_TEXT
  FROM Joined_With_Calendar_Week jw
  LEFT JOIN PROFIT_CENTER_TEXT pct
    ON jw._B631_S_PROFTCTR = pct.PRCTR
),

-- Step 12: Apply Calculated Columns
FLAGS AS (
  SELECT
    _BIC_ZIO_SKFQ,
    UNIT,
    _BIC_ZIO_PSKF,
    _BIC_ZIO_PSTY,
    RX_COMP_WK,
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    RECORDMODE,
    _BIC_ZIO_SWEEK,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,
    DIVISION_DESC,
    AREA_CODE,
    AREA_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    REGION_CODE,
    REGION_DESC,
    STRNUM,
    ZRWENDDATE,
    ZRWSTRTDATE,
    CITY,
    STATE,
    PROFIT_CENTER_TEXT,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    FLAG,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    CASE WHEN _BIC_ZIO_PSKF = '804013' THEN CAST(_BIC_ZIO_SKFQ AS NUMERIC) * 2 ELSE CAST(_BIC_ZIO_SKFQ AS NUMERIC) END AS CAL_BIC_ZIO_SKFQ,
    CASE WHEN RX_COMP_WK IS NULL THEN '0' ELSE RX_COMP_WK END AS CAL_RX_COMP_WK
  FROM Joined_With_Profit_Center_Text
)

-- =====================================================================
-- FINAL OUTPUT: Aggregated Result
-- =====================================================================
SELECT
  MANDT,
  FISCPER,
  FISCVARNT,
  FISCYEAR,
  FISCPER3,
  RECORDMODE,
  _BIC_ZIO_SWEEK,
  _B631_S_CO_AREA,
  _BIC_ZIO_CMPCD,
  _B631_S_PROFTCTR,
  _B631_S_COSTCNTR,
  _BIC_ZIO_VER,
  _BIC_ZIO_SAUDT,
  REP_MKT_CODE,
  REP_MKT_DESC,
  EMERG_MKT_IND,
  _BIC_ZIO_PSTY,
  _BIC_ZIO_PSKF,
  UNIT,
  DIVISION_CODE,
  DIVISION_DESC,
  AREA_CODE,
  AREA_DESC,
  DISTRICT_CODE,
  DISTRICT_DESC,
  REGION_CODE,
  REGION_DESC,
  CAL_WEEK_NUMBER,
  STRNUM,
  ZRWENDDATE,
  ZRWSTRTDATE,
  CITY,
