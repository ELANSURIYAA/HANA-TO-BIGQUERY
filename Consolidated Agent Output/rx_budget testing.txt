Live_Cube AS (

Combined_Frozen_Live_Cube AS (
  SELECT * FROM Frozen_Cube
  UNION ALL
  SELECT * FROM Live_Cube

    _BIC_ZWWSC_PA1,

    CURRENCY,
    FLAG
),

    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,

    FISCPER3,

    _BIC_ZIO_SAUDT,

    Aggregated_Weekly_Budget
  GROUP BY
    FISCPER,
    FISCVARNT,
    _BIC_ZIO_SWEEK,
    FISCYEAR,
    FISCPER3,
    _B631_S_CHRTACCT,
    _B631_S_GL_ACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    MANDT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    RECORDMODE,
    CURRENCY

WEEKLY_SNAPSHOT_DS05 AS (
  SELECT
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    _BIC_ZIO_SWEEK,
    _B631_S_CHRTACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _B631_S_GL_ACCT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    _B631_S_AMOUNT,
    CURRENCY,
    _BIC_ZIO_AMT
  FROM
    CV_BASE_FIN_WEEKLY_BUDGET_S4_OUTPUT
  WHERE
    FISCVARNT = 'K4'
    AND _BIC_ZIO_SWEEK BETWEEN @IP_WEEK_ENDING_FROM AND @IP_WEEK_ENDING_TO
    AND _BIC_ZIO_VER = @IP_VERSION
    AND _BIC_ZIO_SAUDT IN ('1', '10')
),

HIER_NODE AS (
  SELECT
    MANDT,
    PARNODE,
    HRYVALTO,
    NODEVALUE
  FROM
    CV_BASE_MD_HRRP_NODE_S4_OUTPUT
  WHERE
    REGEXP_CONTAINS(PARNODE, r'.*CORE_RET$')
    AND HRYVALTO = '99991231'
),

Join_1 AS (
  SELECT
    w.*,
    h.PARNODE
  FROM
    WEEKLY_SNAPSHOT_DS05 w
    INNER JOIN HIER_NODE h
      ON w._B631_S_PROFTCTR = h.NODEVALUE
),

ONLY_CORE_RET_DATA AS (
  SELECT
    *
  FROM
    Join_1
),

STORE_ATTR_ACTUAL AS (
  SELECT
    MANDT,
    STRNUM,
    PRCTR,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,

    RX_REGION_CODE,
    RX_DISTRICT_CODE
  FROM
    CV_COMP_MD_SRPACT_STATIC_OUTPUT
),

Join_2 AS (
  SELECT
    f.*,
    s.REP_MKT_CODE,
    s.REP_MKT_DESC,
    s.EMERG_MKT_IND,
    s.DIVISION_CODE,
    s.DIVISION_DESC,
    s.AREA_CODE,
    s.AREA_DESC,

    s.DISTRICT_DESC,
    s.STRNUM,
    s.REGION_CODE,
    s.REGION_DESC,
    s.CITY,
    s.STATE,
    s.FS_OPEN_DAT,
    s.RX_OPEN_DAT,
    s.RX_DIVISION_CODE,
    s.RX_AREA_CODE,
    s.RX_REGION_CODE,
    s.RX_DISTRICT_CODE
  FROM
    ONLY_CORE_RET_DATA f
    LEFT JOIN STORE_ATTR_ACTUAL s
      ON f._B631_S_PROFTCTR = s.PRCTR
),

    FS_COMP_WK,

    ZWEEK,
    ZMONTH,
    ZYEAR,
    COMP_VER
  FROM
    CV_COMP_MD_COMPFL_STATIC_OUTPUT
  WHERE
    COMP_VER = @IP_VERSION
),

Join_3 AS (
  SELECT
    w.*,
    c.FS_COMP_WK,
    c.RX_COMP_WK
  FROM
    WEEK_NUMBER w
    LEFT JOIN COMP_FLAG_BUDGET c
      ON w._B631_S_PROFTCTR = c.PRCTR
      AND w._BIC_ZIO_SWEEK = c.ZWEEK
),

CAL_WEEK AS (
  SELECT
    ZZWEEK,
    ZRWSTRTDATE,
    ZRWENDDATE,
    RCLNT
  FROM
    CV_BASE_MD_RCALWEEK_S4_OUTPUT
),

Join_4 AS (
  SELECT
    j.*,
    c.ZRWSTRTDATE,
    c.ZRWENDDATE
  FROM
    Join_3 j
    LEFT JOIN CAL_WEEK c
      ON j._BIC_ZIO_SWEEK = c.ZZWEEK
),

PROFIT_CENTER_TEXT AS (
  SELECT
    MANDT,
    PRCTR,
    LTEXT AS PROFIT_CENTER_TEXT
  FROM
    CV_BASE_MD_CEPCT_S4_OUTPUT
),

Join_5 AS (
  SELECT
    j.*,
    p.PROFIT_CENTER_TEXT
  FROM
    Join_4 j
    LEFT JOIN PROFIT_CENTER_TEXT p
      ON j._B631_S_PROFTCTR = p.PRCTR
),

FLAGS AS (
  SELECT
    RX_COMP_WK,
    FS_COMP_WK,
    FISCPER,
    FISCVARNT,
    FISCYEAR,
    FISCPER3,
    MANDT,
    _BIC_ZIO_SWEEK,
    _B631_S_CHRTACCT,
    _B631_S_CO_AREA,
    _BIC_ZIO_CMPCD,
    _B631_S_PROFTCTR,
    _B631_S_COSTCNTR,
    _B631_S_FUNCAREA,
    _BIC_ZIO_VER,
    _BIC_ZIO_SAUDT,
    _B631_S_GL_ACCT,
    _BIC_ZWWPC_PA1,
    _BIC_ZWWSC_PA1,
    _B631_S_AMOUNT AS _B631_S_AMOUNT_NEGATIVE,
    CURRENCY,
    _BIC_ZIO_AMT,
    REP_MKT_CODE,
    REP_MKT_DESC,
    EMERG_MKT_IND,
    DIVISION_CODE,
    DIVISION_DESC,
    AREA_CODE,
    AREA_DESC,
    DISTRICT_CODE,
    DISTRICT_DESC,
    STRNUM,
    REGION_CODE,
    REGION_DESC,
    ZRWSTRTDATE,
    ZRWENDDATE,
    CITY,
    STATE,
    PROFIT_CENTER_TEXT,
    FS_OPEN_DAT,
    RX_OPEN_DAT,
    PARNODE AS FLAG,
    RX_DIVISION_CODE,
    RX_AREA_CODE,
    RX_REGION_CODE,
    RX_DISTRICT_CODE,
    CASE
      WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN 'FS'
      WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN 'RX'
      ELSE ''
    END AS CAL_FS_RX_FLAG,
    CASE
      WHEN CASE
             WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN FS_COMP_WK
             WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN RX_COMP_WK
             ELSE FS_COMP_WK
           END IS NULL THEN '0'
      ELSE CASE
             WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'FS' THEN FS_COMP_WK
             WHEN LEFT(SAFE_CAST(_BIC_ZWWPC_PA1 AS STRING), 2) = 'RX' THEN RX_COMP_WK
             ELSE FS_COMP_WK
           END
    END AS CAL_COMP_FLAG,
    RIGHT(CAST(_BIC_ZIO_SWEEK AS STRING), 2) AS CAL_WEEK_NUMBER,
    (_B631_S_AMOUNT * -1) AS _B631_S_AMOUNT
  FROM
    Join_5
)
