    DISCTYPECODE,

  WHERE RECORDQUALIFIER IN ('5', '6')

    AND WORKSTATIONID <> '0000000000'

    AND ZZ_UPD_TIMESTAMP <> '0'

SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,

  ARCHIVED,
  RETAILTYPECODE,
  ZZ_UPD_TIMESTAMP,
  DISCTYPECODE,
  TRANSCURRENCY,
  SUM(SALESAMOUNT) AS SALESAMOUNT,
  SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT

GROUP BY

  BUSINESSDAYDATE,
