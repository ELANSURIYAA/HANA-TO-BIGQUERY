    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY

    AND WORKSTATIONID <> '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP <> '0'

SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,
  WORKSTATIONID,
  ARCHIVED,
  RETAILTYPECODE,
  ZZ_UPD_TIMESTAMP,
  DISCTYPECODE,
  TRANSCURRENCY,
  SUM(SALESAMOUNT) AS SALESAMOUNT,
  SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT
FROM Filtered_TLOGF
GROUP BY
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,

  DISCTYPECODE,
