```sql
WITH Filtered_TLOGF AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    SALESAMOUNT,
    REDUCTIONAMOUNT,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY
  FROM `PROJECT.DATASET.TLOGF`
  WHERE RECORDQUALIFIER IN ('5', '6')
    AND TRANSTYPECODE NOT IN ('1107', '1197', '1020')
    AND WORKSTATIONID <> '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP <> '0'
)

SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,
  WORKSTATIONID,
  ARCHIVED,
  RETAILTYPECODE,
  ZZ_UPD_TIMESTAMP,
  DISCTYPECODE,
  TRANSCURRENCY,
  SUM(SALESAMOUNT) AS SALESAMOUNT,
  SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT
FROM Filtered_TLOGF
GROUP BY
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,
  WORKSTATIONID,
  ARCHIVED,
  RETAILTYPECODE,
  ZZ_UPD_TIMESTAMP,
  DISCTYPECODE,
  TRANSCURRENCY
```
