    WORKSTATIONID,
    SALESAMOUNT,
    REDUCTIONAMOUNT,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY

    AND WORKSTATIONID <> '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP <> '0'

SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
  TRANSTYPECODE,
  WORKSTATIONID,
  ARCHIVED,
  RETAILTYPECODE,
  ZZ_UPD_TIMESTAMP,
  DISCTYPECODE,
  TRANSCURRENCY,
  SUM(SALESAMOUNT) AS SALESAMOUNT,
  SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT

GROUP BY
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  RECORDQUALIFIER,
