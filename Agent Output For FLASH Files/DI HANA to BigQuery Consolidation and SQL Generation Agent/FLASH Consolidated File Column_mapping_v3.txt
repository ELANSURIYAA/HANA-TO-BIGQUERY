-- =============================================================================
-- FLASH SALES CONSOLIDATED BIGQUERY SQL
-- Final Consumer : xml_acc_cv_comp_fin_flash_static_tbl_wss_flash_sales
--                  (CV_COMP_FIN_FLASH_STATIC)
-- Description    : Fully expanded, single-query consolidation of the Flash
--                  Sales reporting pipeline.  The query inlines:
--                    1. STP_WSS_FLASH_SALES (stored procedure INSERT-SELECT)
--                    2. CV_COMP_FIN_FLASH   (composite calculation view)
--                    3. CV_BASE_MD_RCALWEEK_S4 (retail calendar dimension)
--                    4. CV_BASE_FIN_FLASH_SALES_CAR (VT_CAR consolidated view)
--                  and produces the aggregated output defined by the
--                  CV_COMP_FIN_FLASH_STATIC logical model (CUBE / Aggregation).
--
-- Runtime Parameters (all supplied by the caller; no defaults in SQL):
--   @ip_version              STRING  - Comp-flag version filter (SFN_ACTUALS_COMP_FLAG)
--   @ip_week_ending_from     STRING  - YYYYMMDD lower bound for BUSINESSDAYDATE
--                                      (derived by SFN_CURRENT_WEEK_START_DATE;
--                                       in the procedure this equals
--                                       V_WEEK_ENDING_FROM_DATE)
--   @ip_week_ending_to       STRING  - YYYYMMDD upper bound for BUSINESSDAYDATE
--                                      (derived by SFN_CURRENT_WEEK_END_DATE;
--                                       in the procedure this equals
--                                       V_WEEK_ENDING_TO_DATE)
--   @ip_upd_timestamp_from   STRING  - Lower bound for zz_upd_timestamp filter
--                                      in CV_BASE_FIN_FLASH_SALES_CAR
--                                      (in the procedure this equals
--                                       V_UPDATE_TIMESTAMP_FROM as YYYYMMDDHHMMSS)
--   @ip_upd_timestamp_to     STRING  - Upper bound for zz_upd_timestamp filter
--                                      in CV_BASE_FIN_FLASH_SALES_CAR
--                                      (in the procedure this equals
--                                       V_UPDATE_TIMESTAMP_TO as YYYYMMDDHHMMSS)
--
-- ISSUES TO FIX (non-blocking where noted; blocking where marked):
--   1. [BLOCKING] REQUIRES VALIDATION: MISSING TABLE MAPPING
--      Physical BigQuery table mappings for /POSDW/NAVIX, /POSDW/TLOGF,
--      /POSDW/TLOGF_X, and ZTFIRP_FLASH_PRM are absent from
--      Columns_Mapping_v3.csv.  Placeholder names are used throughout
--      the VT_CAR section (CTEs 1-24).
--   2. [BLOCKING] REQUIRES VALIDATION: MISSING COLUMN MAPPING
--      All columns of /POSDW/NAVIX, /POSDW/TLOGF, /POSDW/TLOGF_X, and
--      ZTFIRP_FLASH_PRM are absent from Columns_Mapping_v3.csv.
--   3. [NON-BLOCKING] zz_upd_timestamp datatype (STRING vs NUMERIC vs
--      TIMESTAMP) is unresolved for the VT_CAR filter.  The parameter
--      @ip_upd_timestamp_from / @ip_upd_timestamp_to are typed STRING to
--      match the converted SQL; validate against the physical schema.
--   4. [NON-BLOCKING] RECORDQUALIFIER = 25 in TLOGF_X: numeric vs string
--      type requires validation against the physical schema.
--   5. [NON-BLOCKING] HANA MATCH('*CORE_RET*') represented as
--      REGEXP_CONTAINS(parnode, r'.*CORE_RET.*'); exact wildcard semantics
--      should be validated.
--   6. [NON-BLOCKING] GREATEST(fs_comp_wk, rx_comp_wk) used in place of
--      HANA scalar MAX(a,b); NULL handling may differ — validate.
--   7. [NON-BLOCKING] Scalar functions SFN_ACTUALS_COMP_FLAG,
--      SFN_CURRENT_WEEK_START_DATE, SFN_CURRENT_WEEK_END_DATE are
--      unresolved and replaced by runtime parameters @ip_version,
--      @ip_week_ending_from, @ip_week_ending_to respectively.
--   8. [NON-BLOCKING] SNAPSHOT_TIMESTAMP (CURRENT_TIMESTAMP) and
--      CREATED_BY (SESSION_USER) are audit-only columns populated by the
--      procedure INSERT; they are not part of the SELECT logic of
--      CV_COMP_FIN_FLASH and are excluded from this consolidated query.
--      CV_COMP_FIN_FLASH_STATIC does include them in its logical model
--      (as pass-through attributes from TBL_WSS_FLASH_SALES), so they
--      appear in the final GROUP BY as NULL constants with correct types.
--   9. [NON-BLOCKING] hryvalto DATE type: filter converted from
--      HRYVALTO = '99991231' to hryvalto = DATE '9999-12-31'.
--  10. [NON-BLOCKING] zrwenddate is DATE; cal_week_ending_date is a
--      computed YYYYMMDD STRING.  The join uses
--      PARSE_DATE('%Y%m%d', f.cal_week_ending_date) = rc.zrwenddate.
-- =============================================================================

-- CTE 7: Joined_Scripts_With_Currency
-- Logic: Inner-joins SCRIPTS with NAVIX on MANDT, RETAILSTOREID,
--        BUSINESSDAYDATE to attach CURRENCY to script records.
--        Implements Join_1 from CV_COMP_FLASH_SALES.
-- Source: SCRIPTS, NAVIX
-- Join: INNER JOIN SCRIPTS s ON NAVIX n ON
--       s.MANDT=n.MANDT AND s.RETAILSTOREID=n.RETAILSTOREID
--       AND s.BUSINESSDAYDATE=n.BUSINESSDAYDATE
-- =====================================================================
Joined_Scripts_With_Currency AS (
  SELECT
    s.MANDT,
    s.RETAILSTOREID,
    s.BUSINESSDAYDATE,
    s.ZZ_RX_CNT_NS,
    s.ZZ_RX_CNT_REFILL,
    s.ZZ_RX_CNT_GE84_NS,
    s.ZZ_RX_CNT_GE84_RE,
    s.ZZ_RX_MCRX_GE84_NS,
    s.ZZ_RX_MCRX_GE84_RE,
    s.ZZ_CUSTTYPE,
    n.CURRENCY,
    s.ZZ_UPD_TIMESTAMP,
    s.WORKSTATIONID
  FROM SCRIPTS s
  INNER JOIN NAVIX n
    ON  s.MANDT          = n.MANDT
    AND s.RETAILSTOREID  = n.RETAILSTOREID
    AND s.BUSINESSDAYDATE = n.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 8: SCRIPT_KPIs
-- Logic: Derives calculated KPI columns from script/RX data joined with
--        currency.  Implements SCRIPT_KPIs node from CV_COMP_FLASH_SALES.
-- Source: Joined_Scripts_With_Currency
-- Calculation:
--   cal_fs_units       = 1 if ZZ_CUSTTYPE IN ('F','M','R'),
--                        -1 if IN ('X','Y','Z'), else 0
--   cal_rx_cnt_ge84_ns = 0 if ZZ_RX_CNT_GE84_NS='', else CAST to INT64
--   cal_rx_cnt_ge84_re = 0 if ZZ_RX_CNT_GE84_RE='', else CAST to INT64
--   cal_rx_cnt_ns      = 0 if ZZ_RX_CNT_NS='',      else CAST to INT64
--   cal_rx_cnt_re      = 0 if ZZ_RX_CNT_REFILL='',  else CAST to INT64
--   cal_cvd_units      = 1 if WORKSTATIONID='0000000555' AND ZZ_CUSTTYPE='C',
--                        -1 if WORKSTATIONID='0000000555' AND ZZ_CUSTTYPE='W',
--                        else 0
-- =====================================================================
SCRIPT_KPIs AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE,
    ZZ_CUSTTYPE,
    CURRENCY,
    ZZ_UPD_TIMESTAMP AS scripts_upd_timestamp,
    WORKSTATIONID,
    CASE
      WHEN ZZ_CUSTTYPE IN ('F', 'M', 'R') THEN 1
      WHEN ZZ_CUSTTYPE IN ('X', 'Y', 'Z') THEN -1
      ELSE 0
    END AS cal_fs_units,
    CASE
      WHEN ZZ_RX_CNT_GE84_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_NS AS INT64)
    END AS cal_rx_cnt_ge84_ns,
    CASE
      WHEN ZZ_RX_CNT_GE84_RE = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_RE AS INT64)
    END AS cal_rx_cnt_ge84_re,
    CASE
      WHEN ZZ_RX_CNT_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_NS AS INT64)
    END AS cal_rx_cnt_ns,
    CASE
      WHEN ZZ_RX_CNT_REFILL = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_REFILL AS INT64)
    END AS cal_rx_cnt_re,
    CASE
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'C' THEN 1
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'W' THEN -1
      ELSE 0
    END AS cal_cvd_units
  FROM Joined_Scripts_With_Currency
),

-- =====================================================================
-- CTE 21: COMBINE_DATA
-- Logic: Aggregates the six-branch union by grain keys
--        MANDT, RETAILSTOREID, BUSINESSDAYDATE, transcurrency,
--        zz_upd_timestamp.  Implements COMBINE_DATA aggregation node
--        from CV_COMP_FLASH_SALES.
-- Source: Combined_Flash_Sales_Inputs
-- Aggregation: GROUP BY MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--              transcurrency, zz_upd_timestamp;
--              SUM for sales/discount/unit/count measures;
--              COUNT for raw ZZ_RX_* string fields;
--              MAX for timestamp fields.
-- =====================================================================
COMBINE_DATA AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    transcurrency,
    zz_upd_timestamp,
    SUM(fs_salesamount)        AS fs_salesamount,
    SUM(reductionamount)       AS reductionamount,
    SUM(rx_salesamount)        AS rx_salesamount,
    COUNT(ZZ_RX_CNT_NS)        AS zz_rx_cnt_ns,
    COUNT(ZZ_RX_CNT_REFILL)    AS zz_rx_cnt_refill,
    COUNT(ZZ_RX_CNT_GE84_NS)   AS zz_rx_cnt_ge84_ns,
    COUNT(ZZ_RX_CNT_GE84_RE)   AS zz_rx_cnt_ge84_re,
    COUNT(ZZ_RX_MCRX_GE84_NS)  AS zz_rx_mcrx_ge84_ns,
    COUNT(ZZ_RX_MCRX_GE84_RE)  AS zz_rx_mcrx_ge84_re,
    SUM(cal_fs_units)           AS cal_fs_units,
    SUM(emp_reductionamount)    AS emp_reductionamount,
    SUM(cal_rx_cnt_ge84_ns)     AS cal_rx_cnt_ge84_ns,
    SUM(cal_rx_cnt_ge84_re)     AS cal_rx_cnt_ge84_re,
    SUM(cal_rx_cnt_ns)          AS cal_rx_cnt_ns,
    SUM(cal_rx_cnt_re)          AS cal_rx_cnt_re,
    MAX(fs_upd_timestamp)       AS fs_upd_timestamp,
    MAX(rx_upd_timestamp)       AS rx_upd_timestamp,
    MAX(scripts_upd_timestamp)  AS scripts_upd_timestamp,
    SUM(cvd_amount)             AS cvd_amount,
    SUM(cal_cvd_units)          AS cal_cvd_units
  FROM Combined_Flash_Sales_Inputs
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    transcurrency,
    zz_upd_timestamp
),

-- =====================================================================
-- CTE 22: MERGED
-- Logic: Inner-joins NAVIX (base retail-day grain) with COMBINE_DATA on
--        MANDT, RETAILSTOREID, BUSINESSDAYDATE to produce the pre-
--        aggregation dataset.  Implements MERGED node from
--        CV_COMP_FLASH_SALES.
-- Source: NAVIX, COMBINE_DATA
-- Join: INNER JOIN on MANDT, RETAILSTOREID, BUSINESSDAYDATE
-- =====================================================================
MERGED AS (
  SELECT
    n.MANDT,
    n.RETAILSTOREID,
    n.BUSINESSDAYDATE,
    c.fs_salesamount,
    c.reductionamount,
    c.rx_salesamount,
    c.zz_rx_cnt_ns,
    c.zz_rx_cnt_refill,
    c.zz_rx_cnt_ge84_ns,
    c.zz_rx_cnt_ge84_re,
    c.zz_rx_mcrx_ge84_ns,
    c.zz_rx_mcrx_ge84_re,
    c.cal_fs_units,
    c.emp_reductionamount,
    c.cal_rx_cnt_ge84_ns,
    c.cal_rx_cnt_ge84_re,
    c.cal_rx_cnt_ns,
    c.cal_rx_cnt_re,
    c.transcurrency,
    c.rx_upd_timestamp,
    c.fs_upd_timestamp,
    c.scripts_upd_timestamp,
    c.zz_upd_timestamp,
    c.cvd_amount,
    c.cal_cvd_units
  FROM NAVIX n
  INNER JOIN COMBINE_DATA c
    ON  n.MANDT          = c.MANDT
    AND n.RETAILSTOREID  = c.RETAILSTOREID
    AND n.BUSINESSDAYDATE = c.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 23: FLASH
-- Logic: Reads the MERGED composite result (CV_BASE_FIN_FLASH_SALES_CAR
--        output), applies the BUSINESSDAYDATE range filter from
--        CV_COMP_FIN_FLASH ($$IP_WEEK_ENDING_FROM$$ / $$IP_WEEK_ENDING_TO$$
--        passed from STP_WSS_FLASH_SALES as @ip_week_ending_from /
--        @ip_week_ending_to), and the ZZ_UPD_TIMESTAMP range filter
--        (@ip_upd_timestamp_from / @ip_upd_timestamp_to).
--        Excludes store IDs between '0000020000' and '0000024999'.
--        Computes cal_week_ending_date using HANA weekday-to-Saturday logic.
--        Implements the FLASH node of CV_COMP_FIN_FLASH.
-- Source: MERGED
-- Filters: RETAILSTOREID < '0000020000' OR RETAILSTOREID > '0000024999';
--          BUSINESSDAYDATE >= @ip_week_ending_from AND <= @ip_week_ending_to;
--          zz_upd_timestamp >= @ip_upd_timestamp_from AND <= @ip_upd_timestamp_to
-- Calculation: cal_week_ending_date = next Saturday from BUSINESSDAYDATE
--              (HANA add_days / weekday logic converted to BigQuery)
-- =====================================================================
FLASH AS (
  SELECT
    MANDT,
    BUSINESSDAYDATE,
    fs_salesamount,
    reductionamount,
    rx_salesamount,
    cal_fs_units,
    emp_reductionamount,
    cal_rx_cnt_ge84_ns,
    cal_rx_cnt_ge84_re,
    RETAILSTOREID AS retailstoreid_car,
    cal_rx_cnt_ns,
    cal_rx_cnt_re,
    rx_upd_timestamp,
    fs_upd_timestamp,
    scripts_upd_timestamp,
    zz_upd_timestamp,
    transcurrency,
    cvd_amount,
    cal_cvd_units,
    -- Compute week-ending date (Saturday) from BUSINESSDAYDATE (YYYYMMDD STRING).
    -- HANA: add_days(TO_DATE(BUSINESSDAYDATE), 5 - replace(weekday(TO_DATE(BUSINESSDAYDATE)),6,-1))
    -- BigQuery EXTRACT(DAYOFWEEK): Sun=1..Sat=7
    -- HANA weekday: Mon=0..Sun=6; weekday=6 (Sun) is replaced with -1 in HANA logic.
    FORMAT_DATE(
      '%Y%m%d',
      DATE_ADD(
        PARSE_DATE('%Y%m%d', BUSINESSDAYDATE),
        INTERVAL (
          5 - CASE
            WHEN MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7) = 6
              THEN -1
            ELSE MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7)
          END
        ) DAY
      )
    ) AS cal_week_ending_date
  FROM MERGED
  WHERE
    (RETAILSTOREID < '0000020000' OR RETAILSTOREID > '0000024999')
    AND BUSINESSDAYDATE >= @ip_week_ending_from
    AND BUSINESSDAYDATE <= @ip_week_ending_to
    AND zz_upd_timestamp >= @ip_upd_timestamp_from
    AND zz_upd_timestamp <= @ip_upd_timestamp_to
),

-- CTE 25: WEEK
-- Logic: Left-joins FLASH to RETAIL_CALENDAR on
--        PARSE_DATE('%Y%m%d', f.cal_week_ending_date) = rc.zrwenddate
--        to enrich flash sales with retail week/year/period attributes.

  SELECT
    rc.zzweek,
    rc.zryear,
    rc.zrperiod,
    rc.zryrp,
    rc.zrwenddate,
    rc.zrwstrtdate,
    f.MANDT,
    f.BUSINESSDAYDATE,
    f.fs_salesamount,
    f.reductionamount,
    f.rx_salesamount,
    f.cal_fs_units,
    f.emp_reductionamount,

    f.zz_upd_timestamp,
    f.transcurrency,
    f.cvd_amount,
    f.cal_cvd_units
  FROM FLASH f
  LEFT JOIN RETAIL_CALENDAR rc
    ON PARSE_DATE('%Y%m%d', f.cal_week_ending_date) = rc.zrwenddate
),

-- =====================================================================

-- =====================================================================
-- CTE 32: CALCULATIONS
-- Logic: Computes derived KPI columns from the joined dataset.
--        Implements CALCULATIONS node from CV_COMP_FIN_FLASH.
--        cal_scripts_90as3 = cal_rx_cnt_ns + cal_rx_cnt_re
--                            + (cal_rx_cnt_ge84_ns + cal_rx_cnt_ge84_re) * 2
--        cal_comp_flag     = IF(GREATEST(fs_comp_wk,rx_comp_wk) IS NULL,
--                               '0', GREATEST(fs_comp_wk,rx_comp_wk))
--        cal_week_number   = RIGHT(zzweek, 2)
-- Source: Joined_Comp_Flag_Data
-- Calculation: cal_scripts_90as3, cal_comp_flag, cal_week_number
-- =====================================================================
CALCULATIONS AS (
  SELECT
    cal_rx_cnt_re,
    cal_rx_cnt_ns,
    cal_rx_cnt_ge84_re,
    cal_rx_cnt_ge84_ns,
    emp_reductionamount,
    cal_fs_units,
    rx_salesamount,
    reductionamount,
    fs_salesamount,
    rep_mkt_desc,
    rep_mkt_code,
    emerg_mkt_ind,
    rx_comp_wk,
    fs_comp_wk,
    cal_week_ending_date,
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    prctr,
    retailstoreid,
    MANDT,
    division_code,
    division_desc,
    region_code,
    region_desc,
    district_code,
    district_desc,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    fs_open_dat,
    rx_open_dat,
    rx_upd_timestamp,
    fs_upd_timestamp,
    scripts_upd_timestamp,
    zz_upd_timestamp,
    transcurrency,
    area_code,
    area_desc,
    cvd_amount,
    cal_cvd_units,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    cal_rx_cnt_ns + cal_rx_cnt_re
      + (cal_rx_cnt_ge84_ns + cal_rx_cnt_ge84_re) * 2 AS cal_scripts_90as3,
    CASE
      WHEN GREATEST(fs_comp_wk, rx_comp_wk) IS NULL THEN '0'
      ELSE GREATEST(fs_comp_wk, rx_comp_wk)
    END AS cal_comp_flag,
    RIGHT(zzweek, 2) AS cal_week_number
  FROM Joined_Comp_Flag_Data
),

-- =====================================================================
-- CTE 33: EMPLOYEE_DISCOUNT
-- Logic: Aggregates employee discount amounts and max timestamps grouped
--        by dimensional attributes.  Derives synthetic PRCTR values
--        (CAL_PRCTR) based on cal_comp_flag.
--        Implements EMPLOYEE_DISCOUNT aggregation node from
--        CV_COMP_FIN_FLASH.
-- Source: CALCULATIONS
-- Aggregation: SUM(emp_reductionamount), MAX(rx_upd_timestamp),
--              MAX(fs_upd_timestamp), MAX(scripts_upd_timestamp)
--              GROUP BY all dimensional attributes
-- Calculation: cal_prctr = '0000562075' if cal_comp_flag='1',
--                          '0000562076' if cal_comp_flag='0', else NULL
-- =====================================================================
EMPLOYEE_DISCOUNT AS (
  SELECT
    SUM(emp_reductionamount) AS emp_reductionamount,
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    cal_week_ending_date,
    cal_comp_flag,
    cal_week_number,
    MANDT,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    district_code,
    district_desc,
    region_code,
    region_desc,
    rx_comp_wk,
    fs_comp_wk,
    fs_open_dat,
    rx_open_dat,
    prctr AS prctr_org,
    MAX(rx_upd_timestamp)      AS rx_upd_timestamp,
    MAX(fs_upd_timestamp)      AS fs_upd_timestamp,
    MAX(scripts_upd_timestamp) AS scripts_upd_timestamp,
    zz_upd_timestamp,
    transcurrency,
    area_code,
    area_desc,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    CASE
      WHEN cal_comp_flag = '1' THEN '0000562075'
      WHEN cal_comp_flag = '0' THEN '0000562076'
      ELSE NULL
    END AS cal_prctr
  FROM CALCULATIONS
  GROUP BY
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    cal_week_ending_date,
    cal_comp_flag,
    cal_week_number,
    MANDT,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    district_code,
    district_desc,
    region_code,
    region_desc,
    rx_comp_wk,
    fs_comp_wk,
    fs_open_dat,
    rx_open_dat,
    prctr,
    zz_upd_timestamp,
    transcurrency,
    area_code,
    area_desc,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code
),

-- =====================================================================

-- =====================================================================

-- =====================================================================
