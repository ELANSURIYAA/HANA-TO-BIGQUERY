-- =====================================================================
-- CTE 1: NAVIX
-- Logic: Projects the four grain/currency columns from the POS NAVIX

-- Source: `PROJECT.DATASET.POSDW_NAVIX`  ← PLACEHOLDER

    BUSINESSDAYDATE,

-- =====================================================================

-- Source: `PROJECT.DATASET.POSDW_NAVIX`  ← PLACEHOLDER
-- =====================================================================
NAVIX_2 AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CURRENCY
  FROM `PROJECT.DATASET.POSDW_NAVIX`
),

-- =====================================================================
-- CTE 3: Filtered_TLOGF
-- Logic: Filters the POS transaction log (CV_BASE_TLOGF, CUBE/Aggregation)
--        to rows used for FS/RX sales and discounts.
--        Filters: RECORDQUALIFIER IN ('5','6'), TRANSTYPECODE NOT IN
--        ('1107','1197','1020'), WORKSTATIONID <> '0000000000',
--        ARCHIVED = '', ZZ_UPD_TIMESTAMP <> '0'.
--        Aggregates SALESAMOUNT and REDUCTIONAMOUNT by all non-measure keys.
-- Source: `PROJECT.DATASET.POSDW_TLOGF`  ← PLACEHOLDER
-- Filters: RECORDQUALIFIER IN ('5','6'); TRANSTYPECODE NOT IN
--          ('1107','1197','1020'); WORKSTATIONID <> '0000000000';
--          ARCHIVED = ''; ZZ_UPD_TIMESTAMP <> '0'
-- Aggregation: SUM(SALESAMOUNT), SUM(REDUCTIONAMOUNT) GROUP BY all
--              non-measure attributes
-- =====================================================================
Filtered_TLOGF AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY,
    SUM(SALESAMOUNT)     AS SALESAMOUNT,
    SUM(REDUCTIONAMOUNT) AS REDUCTIONAMOUNT
  FROM `PROJECT.DATASET.POSDW_TLOGF`
  WHERE RECORDQUALIFIER IN ('5', '6')
    AND TRANSTYPECODE NOT IN ('1107', '1197', '1020')
    AND WORKSTATIONID <> '0000000000'
    AND ARCHIVED = ''
    AND ZZ_UPD_TIMESTAMP <> '0'
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    RECORDQUALIFIER,
    TRANSTYPECODE,
    WORKSTATIONID,
    ARCHIVED,
    RETAILTYPECODE,
    ZZ_UPD_TIMESTAMP,
    DISCTYPECODE,
    TRANSCURRENCY
),

-- =====================================================================
-- CTE 9: Joined_Scripts_With_Currency
-- Logic: Joins SCRIPTS to NAVIX_2 (inner) on MANDT, RETAILSTOREID,
--        BUSINESSDAYDATE to attach CURRENCY.  Represents Join_1 in
--        CV_COMP_FLASH_SALES.
-- Source: SCRIPTS, NAVIX_2
-- Join: INNER JOIN NAVIX_2 ON MANDT, RETAILSTOREID, BUSINESSDAYDATE
-- =====================================================================
Joined_Scripts_With_Currency AS (
  SELECT
    s.MANDT,
    s.RETAILSTOREID,
    s.BUSINESSDAYDATE,
    s.ZZ_RX_CNT_NS,
    s.ZZ_RX_CNT_REFILL,
    s.ZZ_RX_CNT_GE84_NS,
    s.ZZ_RX_CNT_GE84_RE,
    s.ZZ_RX_MCRX_GE84_NS,
    s.ZZ_RX_MCRX_GE84_RE,
    s.ZZ_CUSTTYPE,
    n2.CURRENCY,
    s.ZZ_UPD_TIMESTAMP,
    s.WORKSTATIONID
  FROM SCRIPTS s
  INNER JOIN NAVIX_2 n2
    ON  s.MANDT          = n2.MANDT
    AND s.RETAILSTOREID  = n2.RETAILSTOREID
    AND s.BUSINESSDAYDATE = n2.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 10: SCRIPT_KPIs
-- Logic: Derives calculated KPI columns from the joined scripts data.
--        Represents the SCRIPT_KPIs projection node in CV_COMP_FLASH_SALES.
--        Calculations:
--          CAL_FS_UNITS     : 1 if ZZ_CUSTTYPE IN ('F','M','R'),
--                             -1 if IN ('X','Y','Z'), else 0
--          CAL_RX_CNT_GE84_NS: 0 if ZZ_RX_CNT_GE84_NS='', else CAST to INT64
--          CAL_RX_CNT_GE84_RE: 0 if ZZ_RX_CNT_GE84_RE='', else CAST to INT64
--          CAL_RX_CNT_NS    : 0 if ZZ_RX_CNT_NS='',       else CAST to INT64
--          CAL_RX_CNT_RE    : 0 if ZZ_RX_CNT_REFILL='',   else CAST to INT64
--          CAL_CVD_UNITS    : 1 if WORKSTATIONID='0000000555' AND
--                               ZZ_CUSTTYPE='C', -1 if ZZ_CUSTTYPE='W', else 0
-- Source: Joined_Scripts_With_Currency
-- Calculation: CAL_FS_UNITS, CAL_RX_CNT_GE84_NS, CAL_RX_CNT_GE84_RE,
--              CAL_RX_CNT_NS, CAL_RX_CNT_RE, CAL_CVD_UNITS
-- =====================================================================
SCRIPT_KPIs AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE,
    ZZ_CUSTTYPE,
    CURRENCY,
    ZZ_UPD_TIMESTAMP AS SCRIPTS_UPD_TIMESTAMP,
    WORKSTATIONID,
    CASE
      WHEN ZZ_CUSTTYPE IN ('F', 'M', 'R') THEN 1
      WHEN ZZ_CUSTTYPE IN ('X', 'Y', 'Z') THEN -1
      ELSE 0
    END AS CAL_FS_UNITS,
    CASE
      WHEN ZZ_RX_CNT_GE84_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_NS AS INT64)
    END AS CAL_RX_CNT_GE84_NS,
    CASE
      WHEN ZZ_RX_CNT_GE84_RE = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_RE AS INT64)
    END AS CAL_RX_CNT_GE84_RE,
    CASE
      WHEN ZZ_RX_CNT_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_NS AS INT64)
    END AS CAL_RX_CNT_NS,
    CASE
      WHEN ZZ_RX_CNT_REFILL = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_REFILL AS INT64)
    END AS CAL_RX_CNT_RE,
    CASE
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'C' THEN 1
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'W' THEN -1
      ELSE 0
    END AS CAL_CVD_UNITS
  FROM Joined_Scripts_With_Currency
),

-- =====================================================================
-- CTE 23: Combined_Flash_Sales_Inputs
-- Logic: UNION ALL of six branches representing Union_1 in
--        CV_COMP_FLASH_SALES.  Each branch contributes its own metric
--        columns and NULLs for columns it does not own.
--        Branch 1 – SCRIPT_KPIs  : script metrics + CAL_* KPIs
--        Branch 2 – FS Sales     : FS_SALESAMOUNT
--        Branch 3 – Emp Discount : EMP_REDUCTIONAMOUNT
--        Branch 4 – RX Sales     : RX_SALESAMOUNT
--        Branch 5 – FS Discount  : REDUCTIONAMOUNT
--        Branch 6 – COVID Sales  : CVD_AMOUNT
-- Source: SCRIPT_KPIs, Joined_FS_Sales_With_Retail_Types,
--         Joined_Employee_Discount_With_Discount_Types,
--         Joined_RX_Sales_With_Retail_Types,
--         Joined_FS_Discount_With_Discount_Types,
--         Joined_COVID_Sales_With_RX_Retail_Types
-- =====================================================================
Combined_Flash_Sales_Inputs AS (

  -- Branch 1: Scripts / KPIs
  SELECT
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE,
    CAL_FS_UNITS,
    CAL_RX_CNT_GE84_NS,
    CAL_RX_CNT_GE84_RE,
    CAST(NULL AS NUMERIC)   AS FS_SALESAMOUNT,
    CAST(NULL AS NUMERIC)   AS EMP_REDUCTIONAMOUNT,
    CAST(NULL AS NUMERIC)   AS RX_SALESAMOUNT,
    CAST(NULL AS NUMERIC)   AS REDUCTIONAMOUNT,
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    CAL_RX_CNT_NS,
    CAL_RX_CNT_RE,
    CURRENCY                AS TRANSCURRENCY,

-- =====================================================================
-- CTE 24: COMBINE_DATA
-- Logic: Aggregates the unioned six-branch result by the grain keys
--        (MANDT, RETAILSTOREID, BUSINESSDAYDATE, TRANSCURRENCY,
--        ZZ_UPD_TIMESTAMP).  Represents the COMBINE_DATA aggregation
--        node in CV_COMP_FLASH_SALES.
--        SUM: FS_SALESAMOUNT, REDUCTIONAMOUNT, RX_SALESAMOUNT,
--             CAL_FS_UNITS, EMP_REDUCTIONAMOUNT, CAL_RX_CNT_GE84_NS,
--             CAL_RX_CNT_GE84_RE, CAL_RX_CNT_NS, CAL_RX_CNT_RE,
--             CVD_AMOUNT, CAL_CVD_UNITS
--        COUNT: ZZ_RX_CNT_NS, ZZ_RX_CNT_REFILL, ZZ_RX_CNT_GE84_NS,
--               ZZ_RX_CNT_GE84_RE, ZZ_RX_MCRX_GE84_NS,
--               ZZ_RX_MCRX_GE84_RE
--        MAX: FS_UPD_TIMESTAMP, RX_UPD_TIMESTAMP, SCRIPTS_UPD_TIMESTAMP
-- Source: Combined_Flash_Sales_Inputs
-- Aggregation: GROUP BY MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--              TRANSCURRENCY, ZZ_UPD_TIMESTAMP
-- =====================================================================
COMBINE_DATA AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY,
    ZZ_UPD_TIMESTAMP,
    SUM(FS_SALESAMOUNT)         AS FS_SALESAMOUNT,
    SUM(REDUCTIONAMOUNT)        AS REDUCTIONAMOUNT,
    SUM(RX_SALESAMOUNT)         AS RX_SALESAMOUNT,
    SUM(CAL_FS_UNITS)           AS CAL_FS_UNITS,
    SUM(EMP_REDUCTIONAMOUNT)    AS EMP_REDUCTIONAMOUNT,
    SUM(CAL_RX_CNT_GE84_NS)     AS CAL_RX_CNT_GE84_NS,
    SUM(CAL_RX_CNT_GE84_RE)     AS CAL_RX_CNT_GE84_RE,
    SUM(CAL_RX_CNT_NS)          AS CAL_RX_CNT_NS,
    SUM(CAL_RX_CNT_RE)          AS CAL_RX_CNT_RE,
    SUM(CVD_AMOUNT)             AS CVD_AMOUNT,
    SUM(CAL_CVD_UNITS)          AS CAL_CVD_UNITS,
    COUNT(ZZ_RX_CNT_NS)         AS ZZ_RX_CNT_NS,
    COUNT(ZZ_RX_CNT_REFILL)     AS ZZ_RX_CNT_REFILL,
    COUNT(ZZ_RX_CNT_GE84_NS)    AS ZZ_RX_CNT_GE84_NS_CNT,
    COUNT(ZZ_RX_CNT_GE84_RE)    AS ZZ_RX_CNT_GE84_RE_CNT,
    COUNT(ZZ_RX_MCRX_GE84_NS)   AS ZZ_RX_MCRX_GE84_NS,
    COUNT(ZZ_RX_MCRX_GE84_RE)   AS ZZ_RX_MCRX_GE84_RE,
    MAX(FS_UPD_TIMESTAMP)       AS FS_UPD_TIMESTAMP,
    MAX(RX_UPD_TIMESTAMP)       AS RX_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP)  AS SCRIPTS_UPD_TIMESTAMP
  FROM Combined_Flash_Sales_Inputs
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY,
    ZZ_UPD_TIMESTAMP
),

-- =====================================================================
-- CTE 25: MERGED
-- Logic: Inner joins NAVIX to COMBINE_DATA on MANDT, RETAILSTOREID,
--        BUSINESSDAYDATE to restrict the aggregated result to store-days
--        that exist in the NAVIX dimension.  Represents the MERGED node
--        in CV_COMP_FLASH_SALES.
-- Source: NAVIX, COMBINE_DATA
-- Join: INNER JOIN COMBINE_DATA ON MANDT, RETAILSTOREID, BUSINESSDAYDATE
-- =====================================================================
MERGED AS (
  SELECT
    n.MANDT,
    n.RETAILSTOREID,
    n.BUSINESSDAYDATE,
    c.TRANSCURRENCY,
    c.ZZ_UPD_TIMESTAMP,
    c.FS_SALESAMOUNT,
    c.REDUCTIONAMOUNT,
    c.RX_SALESAMOUNT,
    c.CAL_FS_UNITS,
    c.EMP_REDUCTIONAMOUNT,
    c.CAL_RX_CNT_GE84_NS,
    c.CAL_RX_CNT_GE84_RE,
    c.CAL_RX_CNT_NS,
    c.CAL_RX_CNT_RE,
    c.CVD_AMOUNT,
    c.CAL_CVD_UNITS,
    c.FS_UPD_TIMESTAMP,
    c.RX_UPD_TIMESTAMP,
    c.SCRIPTS_UPD_TIMESTAMP
  FROM NAVIX n
  INNER JOIN COMBINE_DATA c
    ON  n.MANDT           = c.MANDT
    AND n.RETAILSTOREID   = c.RETAILSTOREID
    AND n.BUSINESSDAYDATE = c.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 26: CV_COMP_FLASH_SALES_AGG
-- Logic: Final aggregation of the MERGED result as required by the
--        CV_COMP_FLASH_SALES CUBE/Aggregation logical model.
--        GROUP BY: MANDT, RETAILSTOREID, BUSINESSDAYDATE, TRANSCURRENCY
--        MAX: ZZ_UPD_TIMESTAMP, RX_UPD_TIMESTAMP, FS_UPD_TIMESTAMP,
--             SCRIPTS_UPD_TIMESTAMP
--        SUM: all numeric measures
--        This CTE represents the output contract of CV_COMP_FLASH_SALES
--        which feeds the FLASH_SALES_VT_CAR staging table consumed by
--        CV_BASE_FIN_FLASH_SALES_CAR.
-- Source: MERGED
-- Aggregation: GROUP BY MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--              TRANSCURRENCY
-- =====================================================================
CV_COMP_FLASH_SALES_AGG AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY,
    MAX(ZZ_UPD_TIMESTAMP)       AS ZZ_UPD_TIMESTAMP,
    SUM(FS_SALESAMOUNT)         AS FS_SALESAMOUNT,
    SUM(REDUCTIONAMOUNT)        AS REDUCTIONAMOUNT,
    SUM(RX_SALESAMOUNT)         AS RX_SALESAMOUNT,
    SUM(CAL_FS_UNITS)           AS CAL_FS_UNITS,
    SUM(EMP_REDUCTIONAMOUNT)    AS EMP_REDUCTIONAMOUNT,
    SUM(CAL_RX_CNT_GE84_NS)     AS CAL_RX_CNT_GE84_NS,
    SUM(CAL_RX_CNT_GE84_RE)     AS CAL_RX_CNT_GE84_RE,
    SUM(CAL_RX_CNT_NS)          AS CAL_RX_CNT_NS,
    SUM(CAL_RX_CNT_RE)          AS CAL_RX_CNT_RE,
    MAX(RX_UPD_TIMESTAMP)       AS RX_UPD_TIMESTAMP,
    MAX(FS_UPD_TIMESTAMP)       AS FS_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP)  AS SCRIPTS_UPD_TIMESTAMP,
    SUM(CVD_AMOUNT)             AS CVD_AMOUNT,
    SUM(CAL_CVD_UNITS)          AS CAL_CVD_UNITS
  FROM MERGED
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY
)

-- =====================================================================
-- FINAL SELECT
-- Logic: Applies the final-consumer filter from CV_BASE_FIN_FLASH_SALES_CAR
--        (ZZ_UPD_TIMESTAMP between runtime parameters) and aggregates
--        the result with the CUBE/Aggregation contract of that view.
--        Runtime parameters:
--          @ip_upd_timestamp_from  (STRING, default '19000101000000')
--          @ip_upd_timestamp_to    (STRING, default '99991231125959')
--        GROUP BY: MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--                  ZZ_UPD_TIMESTAMP, TRANSCURRENCY
--        SUM: FS_SALESAMOUNT, REDUCTIONAMOUNT, RX_SALESAMOUNT,
--             CAL_FS_UNITS, EMP_REDUCTIONAMOUNT, CAL_RX_CNT_GE84_NS,
--             CAL_RX_CNT_GE84_RE, CAL_RX_CNT_NS, CAL_RX_CNT_RE,
--             CVD_AMOUNT, CAL_CVD_UNITS
--        MAX: RX_UPD_TIMESTAMP, FS_UPD_TIMESTAMP, SCRIPTS_UPD_TIMESTAMP
-- Source: CV_COMP_FLASH_SALES_AGG
-- Filters: ZZ_UPD_TIMESTAMP >= @ip_upd_timestamp_from
--          ZZ_UPD_TIMESTAMP <= @ip_upd_timestamp_to
-- =====================================================================
SELECT
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  ZZ_UPD_TIMESTAMP,
  TRANSCURRENCY,
  SUM(FS_SALESAMOUNT)         AS FS_SALESAMOUNT,
  SUM(REDUCTIONAMOUNT)        AS REDUCTIONAMOUNT,
  SUM(RX_SALESAMOUNT)         AS RX_SALESAMOUNT,
  SUM(CAL_FS_UNITS)           AS CAL_FS_UNITS,
  SUM(EMP_REDUCTIONAMOUNT)    AS EMP_REDUCTIONAMOUNT,
  SUM(CAL_RX_CNT_GE84_NS)     AS CAL_RX_CNT_GE84_NS,
  SUM(CAL_RX_CNT_GE84_RE)     AS CAL_RX_CNT_GE84_RE,
  SUM(CAL_RX_CNT_NS)          AS CAL_RX_CNT_NS,
  SUM(CAL_RX_CNT_RE)          AS CAL_RX_CNT_RE,
  MAX(RX_UPD_TIMESTAMP)       AS RX_UPD_TIMESTAMP,
  MAX(FS_UPD_TIMESTAMP)       AS FS_UPD_TIMESTAMP,
  MAX(SCRIPTS_UPD_TIMESTAMP)  AS SCRIPTS_UPD_TIMESTAMP,
  SUM(CVD_AMOUNT)             AS CVD_AMOUNT,
  SUM(CAL_CVD_UNITS)          AS CAL_CVD_UNITS
FROM CV_COMP_FLASH_SALES_AGG
WHERE ZZ_UPD_TIMESTAMP >= @ip_upd_timestamp_from
  AND ZZ_UPD_TIMESTAMP <= @ip_upd_timestamp_to
GROUP BY
  MANDT,
  RETAILSTOREID,
  BUSINESSDAYDATE,
  ZZ_UPD_TIMESTAMP,
  TRANSCURRENCY
