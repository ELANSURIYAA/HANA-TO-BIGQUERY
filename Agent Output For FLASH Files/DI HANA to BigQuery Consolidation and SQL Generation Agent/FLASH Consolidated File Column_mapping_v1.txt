--   @ip_week_ending_from   STRING  YYYYMMDD  – prior-week start date (replaces V_WEEK_ENDING_FROM_DATE)
--   @ip_week_ending_to     STRING  YYYYMMDD  – prior-week end date   (replaces V_WEEK_ENDING_TO_DATE)
--   @ip_upd_timestamp_from NUMERIC           – update-timestamp lower bound (replaces V_UPDATE_TIMESTAMP_FROM)

-- =====================================================================
-- CTE 1: FLASH_SALES_BASE
-- Logic: Reads raw flash-sales data from the physical CAR table,
--        applies the update-timestamp window filter from the procedure
--        parameters, and computes the week-ending date from BUSINESSDAYDATE
--        using HANA weekday semantics (Mon=0 … Sun=6).
-- Source: FLASH_SALES_VT_CAR  (REQUIRES VALIDATION: table not in Columns_Mapping_v3)
-- Filters: ZZ_UPD_TIMESTAMP BETWEEN @ip_upd_timestamp_from AND @ip_upd_timestamp_to
--          RETAILSTOREID NOT BETWEEN '0000020000' AND '0000024999'
--          BUSINESSDAYDATE BETWEEN @ip_week_ending_from AND @ip_week_ending_to
-- Calculation: CAL_WEEK_ENDING_DATE – next Saturday relative to BUSINESSDAYDATE
-- =====================================================================
FLASH_SALES_BASE AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    FS_SALESAMOUNT,
    REDUCTIONAMOUNT,
    RX_SALESAMOUNT,
    CAL_FS_UNITS,
    EMP_REDUCTIONAMOUNT,
    CAL_RX_CNT_GE84_NS,
    CAL_RX_CNT_GE84_RE,
    CAL_RX_CNT_NS,
    CAL_RX_CNT_RE,
    RX_UPD_TIMESTAMP,
    FS_UPD_TIMESTAMP,
    SCRIPTS_UPD_TIMESTAMP,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY,
    CVD_AMOUNT,
    CAL_CVD_UNITS,
    -- Week-ending date: advance BUSINESSDAYDATE to the coming Saturday
    -- HANA weekday: Mon=0 … Sun=6  →  BigQuery MOD(EXTRACT(DAYOFWEEK)+5,7)
    REPLACE(
      FORMAT_DATE(
        '%Y%m%d',
        DATE_ADD(
          PARSE_DATE('%Y%m%d', BUSINESSDAYDATE),
          INTERVAL (
            5 - CASE
                  WHEN MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7) = 6
                  THEN -1
                  ELSE MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7)
                END
          ) DAY
        )
      ),
      '-', ''
    ) AS CAL_WEEK_ENDING_DATE
  FROM `PROJECT.DATASET.FLASH_SALES_VT_CAR`   -- REQUIRES VALIDATION: map to BQ_FULL_TABLE
  WHERE ZZ_UPD_TIMESTAMP >= @ip_upd_timestamp_from
    AND ZZ_UPD_TIMESTAMP <= @ip_upd_timestamp_to
    AND (RETAILSTOREID < '0000020000' OR RETAILSTOREID > '0000024999')
    AND BUSINESSDAYDATE >= @ip_week_ending_from
    AND BUSINESSDAYDATE <= @ip_week_ending_to
),

-- Source: FLASH_SALES_BASE, RETAIL_CALENDAR
-- Join: LEFT JOIN RETAIL_CALENDAR ON CAL_WEEK_ENDING_DATE = zrwenddate
-- =====================================================================
WEEK AS (
  SELECT
    rc.zzweek,

    rc.zrperiod,

    ON f.CAL_WEEK_ENDING_DATE = rc.zrwenddate

-- =====================================================================
-- CTE 18: SCRIPT_KPIs
-- Logic: Derives calculated KPI columns from scripts data:
--        CAL_FS_UNITS, CAL_RX_CNT_GE84_NS/RE, CAL_RX_CNT_NS/RE,
--        CAL_CVD_UNITS, SCRIPTS_UPD_TIMESTAMP.
-- Source: Joined_Scripts_With_Currency
-- Calculation: CASE-based derivations from ZZ_CUSTTYPE and WORKSTATIONID
-- =====================================================================
SCRIPT_KPIs AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_RX_CNT_NS,
    ZZ_RX_CNT_REFILL,
    ZZ_RX_CNT_GE84_NS,
    ZZ_RX_CNT_GE84_RE,
    ZZ_RX_MCRX_GE84_NS,
    ZZ_RX_MCRX_GE84_RE,
    ZZ_CUSTTYPE,
    CURRENCY,
    ZZ_UPD_TIMESTAMP   AS SCRIPTS_UPD_TIMESTAMP,
    WORKSTATIONID,
    CASE
      WHEN ZZ_CUSTTYPE IN ('F', 'M', 'R') THEN 1
      WHEN ZZ_CUSTTYPE IN ('X', 'Y', 'Z') THEN -1
      ELSE 0
    END AS CAL_FS_UNITS,
    CASE
      WHEN ZZ_RX_CNT_GE84_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_NS AS INT64)
    END AS CAL_RX_CNT_GE84_NS,
    CASE
      WHEN ZZ_RX_CNT_GE84_RE = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_GE84_RE AS INT64)
    END AS CAL_RX_CNT_GE84_RE,
    CASE
      WHEN ZZ_RX_CNT_NS = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_NS AS INT64)
    END AS CAL_RX_CNT_NS,
    CASE
      WHEN ZZ_RX_CNT_REFILL = '' THEN 0
      ELSE CAST(ZZ_RX_CNT_REFILL AS INT64)
    END AS CAL_RX_CNT_RE,
    CASE
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'C' THEN 1
      WHEN WORKSTATIONID = '0000000555' AND ZZ_CUSTTYPE = 'W' THEN -1
      ELSE 0
    END AS CAL_CVD_UNITS
  FROM Joined_Scripts_With_Currency
),

-- =====================================================================
-- CTE 31: COMBINE_DATA
-- Logic: Aggregates the unioned flash-sales inputs by retail-day grain
--        plus TRANSCURRENCY and ZZ_UPD_TIMESTAMP.
-- Source: Combined_Flash_Sales_Inputs
-- Aggregation: SUM for amounts/counts/units; MAX for timestamps;
--              COUNT for raw RX counter fields
-- =====================================================================
COMBINE_DATA AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY,
    ZZ_UPD_TIMESTAMP,
    SUM(FS_SALESAMOUNT)       AS FS_SALESAMOUNT,
    SUM(REDUCTIONAMOUNT)      AS REDUCTIONAMOUNT,
    SUM(RX_SALESAMOUNT)       AS RX_SALESAMOUNT,
    COUNT(ZZ_RX_CNT_NS)       AS ZZ_RX_CNT_NS,
    COUNT(ZZ_RX_CNT_REFILL)   AS ZZ_RX_CNT_REFILL,
    COUNT(ZZ_RX_CNT_GE84_NS)  AS ZZ_RX_CNT_GE84_NS,
    COUNT(ZZ_RX_CNT_GE84_RE)  AS ZZ_RX_CNT_GE84_RE,
    COUNT(ZZ_RX_MCRX_GE84_NS) AS ZZ_RX_MCRX_GE84_NS,
    COUNT(ZZ_RX_MCRX_GE84_RE) AS ZZ_RX_MCRX_GE84_RE,
    SUM(CAL_FS_UNITS)         AS CAL_FS_UNITS,
    SUM(EMP_REDUCTIONAMOUNT)  AS EMP_REDUCTIONAMOUNT,
    SUM(CAL_RX_CNT_GE84_NS)   AS CAL_RX_CNT_GE84_NS,
    SUM(CAL_RX_CNT_GE84_RE)   AS CAL_RX_CNT_GE84_RE,
    SUM(CAL_RX_CNT_NS)        AS CAL_RX_CNT_NS,
    SUM(CAL_RX_CNT_RE)        AS CAL_RX_CNT_RE,
    MAX(FS_UPD_TIMESTAMP)     AS FS_UPD_TIMESTAMP,
    MAX(RX_UPD_TIMESTAMP)     AS RX_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP) AS SCRIPTS_UPD_TIMESTAMP,
    SUM(CVD_AMOUNT)           AS CVD_AMOUNT,
    SUM(CAL_CVD_UNITS)        AS CAL_CVD_UNITS
  FROM Combined_Flash_Sales_Inputs
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    TRANSCURRENCY,
    ZZ_UPD_TIMESTAMP
),

-- =====================================================================
-- CTE 32: MERGED
-- Logic: Inner-joins NAVIX to COMBINE_DATA on the retail-day grain keys
--        to restrict to days that exist in the NAVIX transaction log.
-- Source: NAVIX, COMBINE_DATA
-- Join: INNER JOIN COMBINE_DATA ON MANDT, RETAILSTOREID, BUSINESSDAYDATE
-- =====================================================================
MERGED AS (
  SELECT
    n.MANDT,
    n.RETAILSTOREID,
    n.BUSINESSDAYDATE,
    c.FS_SALESAMOUNT,
    c.REDUCTIONAMOUNT,
    c.RX_SALESAMOUNT,
    c.CAL_FS_UNITS,
    c.EMP_REDUCTIONAMOUNT,
    c.CAL_RX_CNT_GE84_NS,
    c.CAL_RX_CNT_GE84_RE,
    c.CAL_RX_CNT_NS,
    c.CAL_RX_CNT_RE,
    c.TRANSCURRENCY,
    c.RX_UPD_TIMESTAMP,
    c.FS_UPD_TIMESTAMP,
    c.SCRIPTS_UPD_TIMESTAMP,
    c.ZZ_UPD_TIMESTAMP,
    c.CVD_AMOUNT,
    c.CAL_CVD_UNITS
  FROM NAVIX n
  INNER JOIN COMBINE_DATA c
    ON n.MANDT          = c.MANDT
   AND n.RETAILSTOREID  = c.RETAILSTOREID
   AND n.BUSINESSDAYDATE= c.BUSINESSDAYDATE
),

-- =====================================================================
-- CTE 33: FLASH
-- Logic: Aggregates MERGED to the final CV_BASE_FIN_FLASH_SALES_CAR
--        output grain (MANDT, RETAILSTOREID, BUSINESSDAYDATE,
--        ZZ_UPD_TIMESTAMP, TRANSCURRENCY).
-- Source: MERGED
-- Aggregation: SUM for amounts/units; MAX for timestamps
-- =====================================================================
FLASH AS (
  SELECT
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY,
    SUM(FS_SALESAMOUNT)        AS FS_SALESAMOUNT,
    SUM(REDUCTIONAMOUNT)       AS REDUCTIONAMOUNT,
    SUM(RX_SALESAMOUNT)        AS RX_SALESAMOUNT,
    SUM(CAL_FS_UNITS)          AS CAL_FS_UNITS,
    SUM(EMP_REDUCTIONAMOUNT)   AS EMP_REDUCTIONAMOUNT,
    SUM(CAL_RX_CNT_GE84_NS)    AS CAL_RX_CNT_GE84_NS,
    SUM(CAL_RX_CNT_GE84_RE)    AS CAL_RX_CNT_GE84_RE,
    SUM(CAL_RX_CNT_NS)         AS CAL_RX_CNT_NS,
    SUM(CAL_RX_CNT_RE)         AS CAL_RX_CNT_RE,
    MAX(RX_UPD_TIMESTAMP)      AS RX_UPD_TIMESTAMP,
    MAX(FS_UPD_TIMESTAMP)      AS FS_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP) AS SCRIPTS_UPD_TIMESTAMP,
    SUM(CVD_AMOUNT)            AS CVD_AMOUNT,
    SUM(CAL_CVD_UNITS)         AS CAL_CVD_UNITS,
    -- Week-ending date computed from BUSINESSDAYDATE (same formula as FLASH_SALES_BASE)
    REPLACE(
      FORMAT_DATE(
        '%Y%m%d',
        DATE_ADD(
          PARSE_DATE('%Y%m%d', BUSINESSDAYDATE),
          INTERVAL (
            5 - CASE
                  WHEN MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7) = 6
                  THEN -1
                  ELSE MOD(EXTRACT(DAYOFWEEK FROM PARSE_DATE('%Y%m%d', BUSINESSDAYDATE)) + 5, 7)
                END
          ) DAY
        )
      ),
      '-', ''
    ) AS CAL_WEEK_ENDING_DATE
  FROM MERGED
  GROUP BY
    MANDT,
    RETAILSTOREID,
    BUSINESSDAYDATE,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY
),

-- =====================================================================
-- CTE 34: WEEK_COMP
-- Logic: Left-joins FLASH (CV_BASE_FIN_FLASH_SALES_CAR output) to the
--        retail calendar on CAL_WEEK_ENDING_DATE = zrwenddate.
-- Source: FLASH, RETAIL_CALENDAR
-- Join: LEFT JOIN RETAIL_CALENDAR ON CAL_WEEK_ENDING_DATE = zrwenddate
-- =====================================================================
WEEK_COMP AS (
  SELECT
    rc.zzweek,
    rc.zryear,
    rc.zrperiod,
    rc.zryrp,
    rc.zrwenddate,
    rc.zrwstrtdate,
    f.MANDT,
    f.BUSINESSDAYDATE,
    f.FS_SALESAMOUNT,
    f.REDUCTIONAMOUNT,
    f.RX_SALESAMOUNT,
    f.CAL_FS_UNITS,
    f.EMP_REDUCTIONAMOUNT,
    f.CAL_RX_CNT_GE84_NS,
    f.CAL_RX_CNT_GE84_RE,
    f.CAL_WEEK_ENDING_DATE,
    f.CAL_RX_CNT_NS,
    f.CAL_RX_CNT_RE,
    f.RETAILSTOREID          AS RETAILSTOREID_CAR,
    f.RX_UPD_TIMESTAMP,
    f.FS_UPD_TIMESTAMP,
    f.SCRIPTS_UPD_TIMESTAMP,
    f.ZZ_UPD_TIMESTAMP,
    f.TRANSCURRENCY,
    f.CVD_AMOUNT,
    f.CAL_CVD_UNITS
  FROM FLASH f
  LEFT JOIN RETAIL_CALENDAR rc
    ON f.CAL_WEEK_ENDING_DATE = rc.zrwenddate

-- =====================================================================
-- CTE 38: CALCULATIONS
-- Logic: Derives CAL_SCRIPTS_90AS3, CAL_COMP_FLAG, and CAL_WEEK_NUMBER
--        from the comp-flag-enriched data.
-- Source: Joined_Comp_Flag_Data_Comp
-- Calculation:
--   CAL_SCRIPTS_90AS3 = CAL_RX_CNT_NS + CAL_RX_CNT_RE
--                       + (CAL_RX_CNT_GE84_NS + CAL_RX_CNT_GE84_RE) * 2
--   CAL_COMP_FLAG     = GREATEST(fs_comp_wk, rx_comp_wk) or '0' if NULL
--   CAL_WEEK_NUMBER   = RIGHT(zzweek, 2)
-- =====================================================================
CALCULATIONS AS (
  SELECT
    CAL_RX_CNT_RE,
    CAL_RX_CNT_NS,
    CAL_RX_CNT_GE84_RE,
    CAL_RX_CNT_GE84_NS,
    EMP_REDUCTIONAMOUNT,
    CAL_FS_UNITS,
    RX_SALESAMOUNT,
    REDUCTIONAMOUNT,
    FS_SALESAMOUNT,
    rep_mkt_desc,
    rep_mkt_code,
    emerg_mkt_ind,
    rx_comp_wk,
    fs_comp_wk,
    CAL_WEEK_ENDING_DATE,
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    prctr,
    RETAILSTOREID,
    MANDT,
    division_code,
    division_desc,
    region_code,
    region_desc,
    district_code,
    district_desc,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    fs_open_dat,
    rx_open_dat,
    RX_UPD_TIMESTAMP,
    FS_UPD_TIMESTAMP,
    SCRIPTS_UPD_TIMESTAMP,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY,
    area_code,
    area_desc,
    CVD_AMOUNT,
    CAL_CVD_UNITS,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    CAL_RX_CNT_NS + CAL_RX_CNT_RE
      + (CAL_RX_CNT_GE84_NS + CAL_RX_CNT_GE84_RE) * 2   AS CAL_SCRIPTS_90AS3,
    CASE
      WHEN GREATEST(fs_comp_wk, rx_comp_wk) IS NULL THEN '0'
      ELSE GREATEST(fs_comp_wk, rx_comp_wk)
    END                                                    AS CAL_COMP_FLAG,
    RIGHT(zzweek, 2)                                       AS CAL_WEEK_NUMBER
  FROM Joined_Comp_Flag_Data_Comp
),

-- =====================================================================
-- CTE 39: EMPLOYEE_DISCOUNT
-- Logic: Aggregates employee-discount rows by dimensional grain and
--        derives CAL_PRCTR (synthetic profit centre for comp/non-comp).
-- Source: CALCULATIONS
-- Aggregation: SUM(EMP_REDUCTIONAMOUNT), MAX(RX/FS/SCRIPTS_UPD_TIMESTAMP)
-- Calculation: CAL_PRCTR from CAL_COMP_FLAG
-- =====================================================================
EMPLOYEE_DISCOUNT AS (
  SELECT
    SUM(EMP_REDUCTIONAMOUNT)    AS EMP_REDUCTIONAMOUNT,
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    CAL_WEEK_ENDING_DATE,
    CAL_COMP_FLAG,
    CAL_WEEK_NUMBER,
    MANDT,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    district_code,
    district_desc,
    region_code,
    region_desc,
    rx_comp_wk,
    fs_comp_wk,
    fs_open_dat,
    rx_open_dat,
    prctr                       AS PRCTR_ORG,
    MAX(RX_UPD_TIMESTAMP)       AS RX_UPD_TIMESTAMP,
    MAX(FS_UPD_TIMESTAMP)       AS FS_UPD_TIMESTAMP,
    MAX(SCRIPTS_UPD_TIMESTAMP)  AS SCRIPTS_UPD_TIMESTAMP,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY,
    area_code,
    area_desc,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code,
    CASE
      WHEN CAL_COMP_FLAG = '1' THEN '0000562075'
      WHEN CAL_COMP_FLAG = '0' THEN '0000562076'
      ELSE NULL
    END                         AS CAL_PRCTR
  FROM CALCULATIONS
  GROUP BY
    zryrp,
    zrperiod,
    zryear,
    zzweek,
    BUSINESSDAYDATE,
    CAL_WEEK_ENDING_DATE,
    CAL_COMP_FLAG,
    CAL_WEEK_NUMBER,
    MANDT,
    zrwenddate,
    zrwstrtdate,
    city,
    state,
    district_code,
    district_desc,
    region_code,
    region_desc,
    rx_comp_wk,
    fs_comp_wk,
    fs_open_dat,
    rx_open_dat,
    prctr,
    ZZ_UPD_TIMESTAMP,
    TRANSCURRENCY,
    area_code,
    area_desc,
    rx_division_code,
    rx_area_code,
    rx_region_code,
    rx_district_code
),

-- =====================================================================
-- CTE 40: COMP_NONCOMP_STORE_ATTR

-- =====================================================================
-- CTE 41: Joined_Employee_Discount_Store_Attributes

Joined_Employee_Discount_Store_Attributes AS (

-- =====================================================================

Combined_Flash_And_Employee_Discount AS (

    RETAILSTOREID,
    CAL_WEEK_ENDING_DATE,
    fs_comp_wk,
    rx_comp_wk,

    division_desc,

    rx_comp_wk,

    rep_mkt_desc,
    CAL_COMP_FLAG,
    division_code,
    division_desc,
    CAST(NULL AS NUMERIC)   AS FS_SALESAMOUNT,

    CAST(NULL AS INT64)     AS CAL_RX_CNT_RE,
    MANDT,
    prctr,
    BUSINESSDAYDATE,
    zzweek,
    zryear,
    zrperiod,
    zryrp,
    region_code,

-- =====================================================================
